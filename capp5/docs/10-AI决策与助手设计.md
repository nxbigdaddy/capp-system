## 一、AI能力概述

CAPP系统通过对接MaxKB专业版智能体，实现三大AI核心能力：

| AI能力 | 说明 | 核心价值 |
|--------|------|---------|
| AI工艺决策 | 基于产品特征自动推荐最优工艺方案 | 减少人工决策时间，提高工艺合理性 |
| AI辅助编制 | 在编制过程中提供智能建议和自动填充 | 提升编制效率，减少输入错误 |
| AI问答助手 | 随时回答工艺相关问题 | 即时的知识获取，降低经验依赖 |

## 二、AI工艺决策

### 2.1 决策流程

```
用户输入产品特征
  ↓
特征提取：产品类型、材质、规格、精度要求、批量
  ↓
MaxKB智能体分析
  ├── 知识库检索：匹配相似产品工艺
  ├── 规则推理：应用工艺选择规则
  └── 混合决策：综合生成最优方案
  ↓
输出推荐方案
  ├── 推荐工序序列
  ├── 每个工序的推荐设备和工装
  ├── 关键工艺参数参考
  └── 预估工时
  ↓
用户确认 / 调整
  ↓
一键生成工艺路线
```

### 2.2 核心实现

```php
// src/AI/AiDecisionService.php

class AiDecisionService {
    private $maxKb;
    private $recommendService;

    public function __construct() {
        $this->maxKb = new MaxKbClient(config('maxkb'));
        $this->recommendService = new AiRecommendService();
    }

    /**
     * AI工艺决策 - 推荐最优工艺方案
     */
    public function decideOptimalProcess($productFeatures) {
        $applicationId = config('maxkb.applications.process_decision.id');
        
        // 1. 先从知识库检索相似工艺
        $similarRoutes = $this->recommendService->recommendSimilarRoutes($productFeatures);
        
        // 2. 构建决策Prompt
        $prompt = $this->buildDecisionPrompt($productFeatures, $similarRoutes);
        
        // 3. 调用智能体进行决策分析
        $response = $this->maxKb->chatWithAgent($applicationId, $prompt, [], [
            'temperature' => 0.2,  // 低温度确保决策稳定性
        ]);
        
        // 4. 解析AI返回的结构化工艺方案
        return $this->parseDecisionResponse($response['data']['content'] ?? '');
    }

    /**
     * 构建决策Prompt
     */
    private function buildDecisionPrompt($features, $similarRoutes) {
        $prompt = "请根据以下产品特征，推荐最优的工艺路线方案：\n\n";
        $prompt .= "## 产品特征\n";
        $prompt .= "- 产品类型：{$features['product_type']}\n";
        $prompt .= "- 材质：{$features['material_type']}\n";
        $prompt .= "- 规格尺寸：{$features['specification']}\n";
        $prompt .= "- 精度等级：{$features['precision_level']}\n";
        $prompt .= "- 生产批量：{$features['batch_size']}\n";
        $prompt .= "- 毛坯类型：{$features['blank_type']}\n";
        
        if (!empty($similarRoutes)) {
            $prompt .= "\n## 参考相似工艺\n";
            foreach (array_slice($similarRoutes, 0, 3) as $route) {
                $prompt .= "- {$route['title']}（相似度: {$route['relevance']}）\n";
            }
        }
        
        $prompt .= "\n请按以下JSON格式返回工艺方案：\n";
        $prompt .= "{\n";
        $prompt .= '  "process_summary": "工艺方案概述",' . "\n";
        $prompt .= '  "operations": [' . "\n";
        $prompt .= '    {' . "\n";
        $prompt .= '      "operation_name": "工序名称",' . "\n";
        $prompt .= '      "work_center_type": "工作中心类型",' . "\n";
        $prompt .= '      "description": "工序说明",' . "\n";
        $prompt .= '      "key_params": {"参数名": "推荐值"},' . "\n";
        $prompt .= '      "estimated_time_min": 预估工时分钟数' . "\n";
        $prompt .= '    }' . "\n";
        $prompt .= '  ],' . "\n";
        $prompt .= '  "total_estimated_time_min": 总预估工时,' . "\n";
        $prompt .= '  "key_notes": ["关键注意事项"]' . "\n";
        $prompt .= '}';
        
        return $prompt;
    }

    /**
     * 解析AI决策响应
     */
    private function parseDecisionResponse($content) {
        // 提取JSON部分
        if (preg_match('/\{[\s\S]*\}/', $content, $matches)) {
            $json = json_decode($matches[0], true);
            if ($json) {
                return [
                    'success' => true,
                    'data'    => $json,
                ];
            }
        }
        
        // JSON解析失败，返回原始文本
        return [
            'success' => true,
            'data'    => ['raw_response' => $content],
        ];
    }

    /**
     * 将AI决策方案一键转换为工艺路线草稿
     */
    public function applyDecisionToRoute($decisionData) {
        $routeService = new ProcessRouteService();
        
        // 创建工艺路线主记录
        $routeId = $routeService->create([
            'name'        => $decisionData['product_name'] ?? 'AI推荐工艺路线',
            'product_id'  => $decisionData['product_id'],
            'status'      => 'drafting',
        ]);
        
        // 添加工序
        $seq = 10;
        foreach ($decisionData['operations'] as $op) {
            $routeOpService = new RouteOperationService();
            $routeOpService->create([
                'route_id'        => $routeId,
                'operation_no'    => (string)$seq,
                'operation_name'  => $op['operation_name'],
                'work_center_type'=> $op['work_center_type'],
                'description'     => $op['description'],
                'standard_time'   => $op['estimated_time_min'],
                'parameters'      => $op['key_params'] ?? [],
                'sort_order'      => $seq,
            ]);
            $seq += 10;
        }
        
        return $routeId;
    }
}
```

## 三、AI辅助编制

### 3.1 辅助场景

| 场景 | AI行为 | 触发时机 |
|------|--------|---------|
| 工序推荐 | 根据当前工序序列推荐下一道工序 | 添加新工序时 |
| 参数建议 | 根据工序/材质/设备推荐工艺参数 | 编辑工序参数时 |
| 工时预估 | 基于历史数据估算工序工时 | 填写工时定额时 |
| 设备推荐 | 根据工序要求推荐合适的设备和工装 | 配置工序资源时 |
| 材料推荐 | 根据产品特征推荐替代材料方案 | 编制材料定额时 |
| 检验项推荐 | 根据工序类型推荐检验项目和标准 | 编制检验卡片时 |

### 3.2 工序推荐实现

```php
// src/AI/AiRecommendService.php (扩展)

class AiRecommendService {
    /**
     * AI推荐下一道工序
     */
    public function recommendNextOperation($currentOperations, $productInfo) {
        $applicationId = config('maxkb.applications.process_assistant.id');
        
        // 描述当前的工序序列
        $opsDesc = '';
        foreach ($currentOperations as $op) {
            $opsDesc .= "- {$op['operation_no']}: {$op['operation_name']} (工作中心: {$op['work_center_name']})\n";
        }
        
        $prompt = "当前工艺路线已有以下工序：\n{$opsDesc}\n"
                . "产品类型：{$productInfo['product_type']}\n"
                . "材质：{$productInfo['material_type']}\n\n"
                . "请推荐下一道最适合的工序。返回JSON格式：\n"
                . '{"operation_name": "...", "reason": "...", "estimated_work_center": "..."}';
        
        $response = $this->maxKb->chatWithAgent($applicationId, $prompt, [], ['temperature' => 0.3]);
        
        return $this->parseJsonResponse($response['data']['content'] ?? '');
    }

    /**
     * AI预估工序工时
     */
    public function estimateOperationTime($operationInfo) {
        $applicationId = config('maxkb.applications.process_assistant.id');
        
        $prompt = "请根据以下信息估算该工序的标准工时（分钟）：\n"
                . "- 工序名称：{$operationInfo['operation_name']}\n"
                . "- 工作中心：{$operationInfo['work_center_name']}\n"
                . "- 材质：{$operationInfo['material_type']}\n"
                . "- 加工尺寸：{$operationInfo['dimensions']}\n\n"
                . "请返回JSON格式：{\"estimated_time_min\": 数字, \"breakdown\": {\"准备时间\": X, \"加工时间\": Y, \"辅助时间\": Z}, \"confidence\": \"高/中/低\"}";
        
        $response = $this->maxKb->chatWithAgent($applicationId, $prompt, [], ['temperature' => 0.2]);
        
        return $this->parseJsonResponse($response['data']['content'] ?? '');
    }
}
```

```php
// src/AI/AiAssistantService.php

class AiAssistantService {
    /**
     * 上下文感知问答
     */
    public function contextAwareChat($userId, $message, $context) {
        $applicationId = config('maxkb.applications.process_assistant.id');
        
        // 构建丰富的上下文
        $contextPrompt = $this->buildContextPrompt($context);
        $fullMessage = $contextPrompt ? "{$contextPrompt}\n\n用户问题：{$message}" : $message;
        
        // 获取最近的对话历史
        $history = $this->getRecentHistory($userId, 5);
        
        $response = $this->maxKb->chatWithAgent($applicationId, $fullMessage, $history, [
            'temperature' => 0.3,
        ]);
        
        // 记录问答日志(用于优化)
        $this->logQaInteraction($userId, $message, $context, $response);
        
        return [
            'answer'   => $response['data']['content'] ?? '',
            'sources'  => $response['data']['references'] ?? [],
        ];
    }

    /**
     * 构建上下文Prompt
     */
    private function buildContextPrompt($context) {
        $parts = [];
        
        if (!empty($context['route_id'])) {
            $route = (new ProcessRouteService())->getById($context['route_id']);
            $parts[] = "当前正在编辑的工艺路线：{$route['product_code']} - {$route['product_name']}";
        }
        
        if (!empty($context['operation_id'])) {
            $op = (new RouteOperationService())->getById($context['operation_id']);
            $parts[] = "当前编辑的工序：{$op['operation_name']}，工作中心：{$op['work_center_name']}";
        }
        
        if (!empty($context['material_type'])) {
            $parts[] = "加工材质：{$context['material_type']}";
        }
        
        if (empty($parts)) return '';
        
        return "【当前工作上下文】\n" . implode("\n", $parts);
    }

    /**
     * 快捷提问模板
     */
    public function getQuickQuestionTemplates($context) {
        $templates = [
            '通用' => [
                '这个产品的工艺难点是什么？',
                '推荐哪种毛坯类型？',
                '关键工序有哪些？',
            ],
            '机加工' => [
                '推荐切削参数范围？',
                '选择哪种刀具最合适？',
                '如何保证形位公差？',
            ],
            '焊接' => [
                '推荐哪种焊接方法？',
                '预热温度应该多少？',
                '焊接参数如何选择？',
            ],
            '热处理' => [
                '推荐哪种热处理工艺？',
                '热处理温度和时间？',
                '冷却方式怎么选？',
            ],
        ];
        
        // 根据当前工序类型过滤模板
        if (!empty($context['operation_type'])) {
            $type = $context['operation_type'];
            return $templates[$type] ?? $templates['通用'];
        }
        
        return $templates['通用'];
    }

    private function logQaInteraction($userId, $question, $context, $response) {
        $db = Database::getInstance();
        $db->table('capp_ai_qa_log')->insert([
            'user_id'     => $userId,
            'question'    => $question,
            'context'     => json_encode($context),
            'answer'      => $response['data']['content'] ?? '',
            'sources'     => json_encode($response['data']['references'] ?? []),
            'response_time_ms' => $response['elapsed_ms'] ?? 0,
            'helpful'     => null,  // 待用户反馈
            'created_at'  => date('Y-m-d H:i:s'),
        ]);
    }
}
```

### 4.3 前端集成代码

```javascript
// templates/assets/js/ai-assistant.js

class AiAssistant {
    constructor(containerId) {
        this.container = document.getElementById(containerId);
        this.context = {}; // 当前编辑上下文
        this.init();
    }

    init() {
        // 渲染助手面板
        this.render();
        
        // 绑定事件
        this.bindEvents();
    }

    setContext(context) {
        this.context = context;
        // 更新快捷问题
        this.updateQuickQuestions();
    }

    async sendMessage(message) {
        // 显示用户消息
        this.appendMessage('user', message);
        
        // 显示加载状态
        this.showTyping();
        
        try {
            const response = await fetch('/api/ai/chat', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({
                    message: message,
                    context: this.context
                })
            });
            
            const data = await response.json();
            
            // 隐藏加载状态
            this.hideTyping();
            
            // 显示AI回复
            this.appendMessage('assistant', data.answer, data.sources);
            
        } catch (error) {
            this.hideTyping();
            this.appendMessage('assistant', '抱歉，AI服务暂时不可用，请稍后重试。');
        }
    }

    appendMessage(role, content, sources) {
        const msgDiv = document.createElement('div');
        msgDiv.className = `ai-message ai-message-${role}`;
        
        const avatar = role === 'user' ? '👤' : '🤖';
        msgDiv.innerHTML = `
            <div class="ai-message-avatar">${avatar}</div>
            <div class="ai-message-content">
                <div class="ai-message-text">${this.formatContent(content)}</div>
                ${sources ? this.renderSources(sources) : ''}
                ${role === 'assistant' ? this.renderFeedback() : ''}
            </div>
        `;
        
        this.messagesContainer.appendChild(msgDiv);
        this.scrollToBottom();
    }

    renderSources(sources) {
        if (!sources || sources.length === 0) return '';
        return `
            <div class="ai-message-sources">
                📚 参考来源：
                ${sources.map(s => `<span class="source-tag">${s}</span>`).join('')}
            </div>
        `;
    }

    renderFeedback() {
        return `
            <div class="ai-message-feedback">
                <button onclick="this.rateAnswer('helpful')" title="有帮助">👍</button>
                <button onclick="this.rateAnswer('not_helpful')" title="无帮助">👎</button>
                <button onclick="this.copyAnswer(this)" title="复制">📋</button>
            </div>
        `;
    }
}
```

## 五、AI决策数据表

```sql
-- AI决策记录表
CREATE TABLE capp_ai_decision_record (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    product_id INT,
    decision_type VARCHAR(30),       -- process_route/operation/parameter
    input_features JSON,             -- 输入的产品特征
    ai_response JSON,                -- AI原始响应
    parsed_result JSON,              -- 解析后的结构化结果
    is_applied TINYINT DEFAULT 0,    -- 是否已采用
    applied_route_id INT,            -- 应用生成的路线ID
    user_feedback VARCHAR(20),       -- 用户反馈: adopted/modified/rejected
    feedback_note TEXT,              -- 反馈备注
    response_time_ms INT,            -- 响应时间(毫秒)
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user_product (user_id, product_id),
    INDEX idx_decision_type (decision_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- AI问答日志表
CREATE TABLE capp_ai_qa_log (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    question TEXT NOT NULL,
    context JSON,
    answer TEXT,
    sources JSON,
    response_time_ms INT,
    helpful TINYINT,                 -- null=未反馈, 1=有帮助, 0=无帮助
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user (user_id),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```
