# CAPP系统 - 企业微信集成设计

> 版本：V1.0 | 日期：2026-06-07 | 状态：初稿

## 一、概述

CAPP系统深度集成企业微信，实现三个核心能力：
1. **统一身份认证**：企业微信扫码登录，免密认证
2. **组织用户同步**：自动同步企业微信通讯录（人员、部门）
3. **消息通知**：审批待办、变更通知、系统公告等通过企业微信推送

### 2.2 核心代码实现

#### 2.2.1 构造授权URL

```php
// src/Integration/WeWork/WeWorkClient.php

class WeWorkClient {
    private $corpId;
    private $agentId;
    private $secret;

    public function __construct($config) {
        $this->corpId  = $config['corp_id'];
        $this->agentId = $config['agent_id'];
        $this->secret  = $config['secret'];
    }

    /**
     * 构造企业微信OAuth2.0授权URL
     */
    public function getAuthUrl($redirectUri, $state = '') {
        $params = [
            'appid'        => $this->corpId,
            'redirect_uri' => urlencode($redirectUri),
            'response_type'=> 'code',
            'scope'        => 'snsapi_base',  // 静默授权或 snsapi_privateinfo
            'state'        => $state,
        ];
        return 'https://open.weixin.qq.com/connect/oauth2/authorize?' 
               . http_build_query($params) . '#wechat_redirect';
    }

    /**
     * 用code换取用户身份信息
     */
    public function getUserInfoByCode($code) {
        // 第一步：获取access_token
        $tokenUrl = "https://qyapi.weixin.qq.com/cgi-bin/gettoken?"
                  . "corpid={$this->corpId}&corpsecret={$this->secret}";
        $tokenRes = $this->httpGet($tokenUrl);
        $accessToken = YOUR_SECRET'access_token'];

        // 第二步：用code获取userid
        $userIdUrl = "https://qyapi.weixin.qq.com/cgi-bin/auth/getuserinfo?"
                   . "access_token=YOUR_SECRET}&code={$code}";
        $userRes = $this->httpGet($userIdUrl);
        
        if ($userRes['errcode'] !== 0) {
            throw new Exception('获取用户信息失败: ' . $userRes['errmsg']);
        }

        // 第三步：获取用户详细信息
        return $this->getUserDetail($accessToken, $userRes['UserId']);
    }
}
```

#### 2.2.2 回调处理

```php
// src/Modules/Auth/Controllers/LoginController.php

class LoginController extends Controller {
    public function weworkCallback() {
        $code = $_GET['code'] ?? '';
        if (empty($code)) {
            $this->redirect('/login');
            return;
        }

        $weworkClient = ServiceContainer::get('wework');
        
        try {
            // 获取企业微信用户信息
            $wxUser = $weworkClient->getUserInfoByCode($code);
            
            // 在本系统查找/创建用户
            $userService = new UserService();
            $user = $userService->findOrCreateByWeWork($wxUser);
            
            // 更新登录信息
            $userService->updateLoginInfo($user['id'], $_SERVER['REMOTE_ADDR']);
            
            // 生成JWT Token
            $jwt = JwtHelper::encode([
                'user_id'   => $user['id'],
                'user_name' => $user['name'],
                'exp'       => time() + 7200,  // 2小时过期
            ]);
            
            // 设置Cookie并跳转
            setcookie('capp_token', $jwt, time() + 7200, '/', '', true, true);
            $this->redirect('/dashboard');
            
        } catch (Exception $e) {
            Logger::error('企微登录失败', ['error' => $e->getMessage()]);
            $this->redirect('/login?error=wework_fail');
        }
    }
}
```

## 三、企业微信用户同步

### 3.1 同步策略

| 策略 | 说明 |
|------|------|
| 全量同步 | 首次部署/每月执行一次，完整拉取通讯录 |
| 增量同步 | 每日定时执行（如凌晨2:00），仅同步变更 |
| 实时同步 | 企业微信成员变更回调事件触发 |

### 3.2 同步数据映射

| 企业微信字段 | CAPP字段 | 说明 |
|-------------|---------|------|
| UserId | wework_user_id | 唯一标识符 |
| Name | name | 姓名 |
| Mobile | mobile | 手机号 |
| Email | email | 邮箱 |
| Department | department | 所属部门(取主部门) |
| Position | position | 职位 |
| Avatar | avatar | 头像URL |
| Status (1/2/4) | status (1/0) | 1=已激活, 2/4=禁用 |

### 3.3 同步流程

```
定时任务触发 (scripts/sync_wework_users.php)
  │
  ├─ Step 1: 获取AccessToken
  │
  ├─ Step 2: 拉取部门列表
  │    GET /cgi-bin/department/list
  │    → 同步到 capp_department 表
  │
  ├─ Step 3: 遍历部门，拉取成员列表
  │    GET /cgi-bin/user/list?department_id={id}&fetch_child=1
  │    → 与本地 capp_user 比对
  │
  ├─ Step 4: 数据处理
  │    ├─ 新用户 → INSERT
  │    ├─ 已存在用户 → UPDATE (name/department/position/mobile)
  │    ├─ 已离职用户 → status=0
  │    └─ 记录变更日志
  │
  └─ Step 5: 同步结果记录到 capp_sync_log
```

### 3.4 核心代码

```php
// scripts/sync_wework_users.php

require_once __DIR__ . '/../src/bootstrap.php';

$weworkClient = new WeWorkClient(config('wework'));
$userService  = new UserService();
$syncLog      = ['sync_type' => 'wework', 'direction' => 'in'];

// 1. 获取部门列表
$departments = $weworkClient->getDepartmentList();

// 2. 遍历部门同步用户
$totalCount = $successCount = $failCount = 0;
foreach ($departments as $dept) {
    $users = $weworkClient->getUserList($dept['id']);
    foreach ($users as $wxUser) {
        $totalCount++;
        try {
            $userService->syncWeWorkUser($wxUser);
            $successCount++;
        } catch (Exception $e) {
            $failCount++;
            Logger::error('同步用户失败', ['user' => $wxUser['userid'], 'error' => $e->getMessage()]);
        }
    }
}

// 3. 记录同步日志
$syncLog['total_count']   = $totalCount;
$syncLog['success_count'] = $successCount;
$syncLog['fail_count']    = $failCount;

$db->table('capp_sync_log')->insert($syncLog);
```

### 4.4 消息回调（接收企业微信事件）

```php
// src/Integration/WeWork/WeWorkCallback.php

class WeWorkCallback {
    /**
     * 处理企业微信回调事件
     */
    public function handle($xmlData) {
        $event = $this->parseXml($xmlData);

        switch ($event['Event']) {
            case 'change_contact':
                $this->handleContactChange($event); // 通讯录变更
                break;
            case 'template_card_event':
                $this->handleCardClick($event);      // 卡片点击
                break;
        }
    }

    private function handleContactChange($event) {
        switch ($event['ChangeType']) {
            case 'create_user':
            case 'update_user':
                $wework = ServiceContainer::get('wework');
                $userInfo = $wework->getUserDetailByToken($event['UserID']);
                (new UserService())->syncWeWorkUser($userInfo);
                break;
            case 'delete_user':
                (new UserService())->disableUserByWeWorkId($event['UserID']);
                break;
        }
    }
}
```
