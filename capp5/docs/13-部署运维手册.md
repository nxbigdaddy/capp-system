## 五、备份与恢复

### 5.1 数据库备份脚本

```php
// scripts/backup_db.php

class DbBackup {
    public function run() {
        $config = config('database');
        $backupDir = STORAGE_PATH . '/backups/database/';
        
        if (!is_dir($backupDir)) {
            mkdir($backupDir, 0755, true);
        }
        
        $filename = 'capp_backup_' . date('Ymd_His') . '.sql.gz';
        $filepath = $backupDir . $filename;
        
        // 使用mysqldump备份
        $command = sprintf(
            'mysqldump -h%s -u%s -p%s %s --single-transaction --routines --triggers | gzip > %s',
            escapeshellarg($config['host']),
            escapeshellarg($config['username']),
            escapeshellarg($config['password']),
            escapeshellarg($config['database']),
            escapeshellarg($filepath)
        );
        
        exec($command, $output, $returnCode);
        
        if ($returnCode === 0) {
            // 清理30天前的备份
            $this->cleanOldBackups($backupDir, 30);
            echo "备份成功: {$filename}\n";
        } else {
            echo "备份失败，错误码: {$returnCode}\n";
        }
    }
    
    private function cleanOldBackups($dir, $days) {
        $files = glob($dir . 'capp_backup_*.sql.gz');
        $cutoff = time() - ($days * 86400);
        
        foreach ($files as $file) {
            if (filemtime($file) < $cutoff) {
                unlink($file);
                echo "清理旧备份: " . basename($file) . "\n";
            }
        }
    }
}

(new DbBackup())->run();
```

### 6.1 健康检查接口

```php
// GET /health - 健康检查
public function health() {
    $checks = [
        'app'      => ['status' => 'ok', 'version' => '1.0.0'],
        'database' => $this->checkDatabase(),
        'redis'    => $this->checkRedis(),
        'storage'  => $this->checkStorage(),
    ];
    
    $isHealthy = true;
    foreach ($checks as $key => $check) {
        if (($check['status'] ?? 'ok') !== 'ok' && $key !== 'app') {
            $isHealthy = false;
        }
    }
    
    http_response_code($isHealthy ? 200 : 503);
    return $this->json([
        'status'    => $isHealthy ? 'healthy' : 'unhealthy',
        'checks'    => $checks,
        'timestamp' => date('c'),
    ]);
}

private function checkDatabase() {
    try {
        $db = Database::getInstance();
        $db->query("SELECT 1");
        return ['status' => 'ok'];
    } catch (Exception $e) {
        return ['status' => 'error', 'message' => $e->getMessage()];
    }
}

private function checkRedis() {
    try {
        $cache = ServiceContainer::get('cache');
        $cache->set('health_check', 'ok', 10);
        return ['status' => 'ok'];
    } catch (Exception $e) {
        return ['status' => 'error', 'message' => $e->getMessage()];
    }
}

private function checkStorage() {
    $isWritable = is_writable(STORAGE_PATH);
    return [
        'status'      => $isWritable ? 'ok' : 'error',
        'free_space'  => disk_free_space(STORAGE_PATH) / 1024 / 1024 . 'MB'
    ];
}
```

## 七、故障处理

### 7.1 常见问题

| 问题 | 可能原因 | 解决方案 |
|------|---------|---------|
| 502 Bad Gateway | PHP-FPM未启动/进程耗尽 | 检查PHP-FPM状态，调整pm.max_children |
| 数据库连接失败 | MySQL未启动/连接数满 | 检查MySQL状态，调整max_connections |
| 登录页面空白 | PHP错误/文件权限 | 查看error log，检查storage权限 |
| 企业微信登录失败 | 回调域名配置错误 | 检查企业微信后台可信域名配置 |
| 文件生成失败 | 存储空间不足/权限问题 | 检查磁盘空间，调整storage权限 |
| Redis连接失败 | Redis未启动/密码错误 | 检查Redis状态和配置 |
| PLM同步失败 | SQL Server连接失败 | 检查网络/防火墙/连接字符串 |
