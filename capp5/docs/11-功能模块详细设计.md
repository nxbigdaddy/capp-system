## 二、系统管理模块 (system)

### 2.1 用户管理

**路由**: `/admin/users`

**功能列表**:
- 用户列表（分页、搜索、按部门筛选）
- 新增/编辑用户（含角色分配）
- 用户状态启用/禁用
- 企业微信同步状态查看
- 密码重置（本地登录用户）

**核心业务逻辑**:
```php
// src/Modules/System/Controllers/UserController.php

class UserController extends Controller {
    // GET /admin/users - 用户列表
    public function index() {
        $page     = $_GET['page'] ?? 1;
        $keyword  = $_GET['keyword'] ?? '';
        $department = $_GET['department'] ?? '';
        $status   = $_GET['status'] ?? '';
        
        $userService = new UserService();
        $result = $userService->getList($page, $keyword, $department, $status);
        
        return $this->view('system/users/index', [
            'users'      => $result['data'],
            'pagination' => $result['pagination'],
        ]);
    }

    // POST /admin/users - 创建用户
    public function store() {
        $input = $this->validate([
            'name'        => 'required|max:50',
            'employee_no' => 'required|unique:capp_user',
            'mobile'      => 'required|mobile',
            'department'  => 'required',
            'role_ids'    => 'required|array',
        ]);
        
        $userService = new UserService();
        $userId = $userService->create($input);
        
        // 分配角色
        $userService->assignRoles($userId, $input['role_ids']);
        
        // 记录操作日志
        Logger::audit('user', 'create', $userId, '创建用户: ' . $input['name']);
        
        return $this->redirect('/admin/users')->with('success', '用户创建成功');
    }
}
```

### 4.3 工序编辑弹窗

```php
// 工序编辑表单（弹窗）
// POST /api/process/route-operation/save
public function saveOperation() {
    $input = $this->validate([
        'route_id'          => 'required|int',
        'operation_no'      => 'required',
        'operation_name'    => 'required|max:100',
        'std_operation_id'  => 'int',        // 引用标准工序
        'work_center_id'    => 'required|int',
        'description'       => 'max:2000',
        'equipment_ids'     => 'array',       // 选择的设备ID列表
        'tooling_ids'       => 'array',
        'cutting_tool_ids'  => 'array',
        'gauge_ids'         => 'array',
        'standard_time'     => 'numeric|min:0',
        'parameters'        => 'json',        // 工艺参数JSON
        'is_key_point'      => 'bool',
    ]);
    
    // 如果引用标准工序，自动填充默认值
    if (!empty($input['std_operation_id'])) {
        $stdOp = (new StdOperationService())->getById($input['std_operation_id']);
        if ($stdOp) {
            $input['operation_name']   = $input['operation_name'] ?: $stdOp['name'];
            $input['description']      = $input['description'] ?: $stdOp['description'];
            $input['standard_time']    = $input['standard_time'] ?: $stdOp['default_duration'];
            $input['work_center_id']   = $input['work_center_id'] ?: $stdOp['default_work_center_id'];
            if (empty($input['parameters']) && !empty($stdOp['parameters_schema'])) {
                $input['parameters'] = json_decode($stdOp['parameters_schema'], true);
            }
        }
    }
    
    $service = new RouteOperationService();
    $id = $service->save($input);
    
    return $this->json(['code' => 0, 'data' => ['id' => $id]]);
}
```

### 4.4 版本管理

**版本号规则**: `V{major}.{minor}`
- 主版本号(major)：审批通过后自动+1
- 次版本号(minor)：修订保存时+1，发布后重置为0

**版本操作**:
- 查看任意历史版本（只读）
- 版本对比（双栏对比视图，差异高亮）
- 版本回退（管理员权限，基于历史版本创建新草稿）

## 六、审批流程模块 (approval)

### 6.1 流程定义配置

**路由**: `/admin/approval-flows`

流程设计器（以工艺路线审批为例）：

```
节点1: 工艺组长审核 (single)  →  节点2: 工艺总师审批 (single)  →  节点3: 技术副总批准 (single)

配置项:
- 节点类型: 单人审批 / 多人会签(全部通过) / 多人或签(一人通过)
- 审批人来源: 指定人 / 角色 / 部门负责人
- 超时设置: 24小时自动催办
- 通过条件: 所有节点通过
- 驳回策略: 退回至发起人 / 退回至上一节点
```

### 6.2 审批操作

```php
// src/Modules/Approval/Controllers/ApprovalController.php

class ApprovalController extends Controller {
    /**
     * 提交审批
     */
    public function submit() {
        $input = $this->validate([
            'business_type' => 'required',  // route/doc/change
            'business_id'   => 'required|int',
            'flow_id'       => 'required|int',
        ]);
        
        // 1. 校验业务对象状态
        $this->validateBusinessStatus($input);
        
        // 2. 创建审批流实例
        $instanceId = $this->createApprovalInstance($input);
        
        // 3. 锁定业务对象为"审核中"
        $this->lockBusiness($input['business_type'], $input['business_id']);
        
        // 4. 推送待办通知到下一个审批节点的审批人
        $this->notifyNextApprover($instanceId);
        
        return $this->json(['code' => 0, 'message' => '提交成功']);
    }

    /**
     * 审批处理
     */
    public function handle() {
        $input = $this->validate([
            'instance_id' => 'required|int',
            'action'      => 'required',      // approve/reject/transfer
            'comment'     => 'max:500',
        ]);
        
        $instance = $this->getInstance($input['instance_id']);
        $this->checkApprovalPermission($instance);
        
        switch ($input['action']) {
            case 'approve':
                $this->handleApprove($instance, $input['comment']);
                break;
            case 'reject':
                $this->handleReject($instance, $input['comment']);
                break;
            case 'transfer':
                $this->handleTransfer($instance, $input['transfer_to'], $input['comment']);
                break;
        }
        
        return $this->json(['code' => 0, 'message' => '操作成功']);
    }

    private function handleApprove($instance, $comment) {
        // 记录审批记录
        $this->recordApproval($instance, 'approve', $comment);
        
        // 判断是否有下一节点
        $nextNode = $this->getNextNode($instance);
        
        if ($nextNode) {
            // 推进到下一节点
            $this->advanceToNextNode($instance, $nextNode);
            // 通知下一节点审批人
            $this->notifyNextApprover($instance['id']);
        } else {
            // 所有节点通过，完成审批
            $this->completeApproval($instance);
            // 解锁业务对象，状态变更为"已发布"
            $this->publishBusiness($instance['business_type'], $instance['business_id']);
            // 通知发起人审批通过
            $this->notifyResult($instance, 'approved');
        }
    }
}
```

## 七、文件输出模块 (output)

### 7.1 输出模板

使用HTML模板 + TCPDF/mPDF生成PDF文件：

```php
// src/Modules/Output/Services/PdfGenerator.php

class PdfGenerator {
    /**
     * 生成工序卡片PDF
     */
    public function generateOperationCardPdf($routeOperationId) {
        $operationCard = (new OperationCardService())->getByRouteOperationId($routeOperationId);
        
        // 渲染HTML模板
        $html = $this->renderTemplate('output/operation_card', [
            'card' => $operationCard,
        ]);
        
        // 生成PDF
        $pdf = new \TCPDF(PDF_PAGE_ORIENTATION, PDF_UNIT, 'A4', true, 'UTF-8');
        $pdf->SetCreator('CAPP System');
        $pdf->SetTitle('工序卡片 - ' . $operationCard['operation_name']);
        $pdf->AddPage();
        
        // 添加水印
        $pdf->SetAlpha(0.1);
        $pdf->SetFont('helvetica', 'B', 40);
        $pdf->RotatedText(60, 200, "版本V{$operationCard['version']} " . date('Y-m-d'), 45);
        $pdf->SetAlpha(1);
        
        $pdf->writeHTML($html, true, false, true, false, '');
        
        $outputPath = STORAGE_PATH . '/exports/' . uniqid('card_') . '.pdf';
        $pdf->Output($outputPath, 'F');
        
        return $outputPath;
    }

    /**
     * 批量生成完整工艺文件包
     */
    public function generateProcessPackage($routeId) {
        $route = (new ProcessRouteService())->getFullRouteById($routeId);
        $zip = new \ZipArchive();
        $zipPath = STORAGE_PATH . '/exports/process_' . $route['code'] . '_' . time() . '.zip';
        
        if ($zip->open($zipPath, \ZipArchive::CREATE) !== true) {
            throw new Exception('无法创建压缩文件');
        }
        
        // 1. 工艺路线清单
        $zip->addFile(
            $this->generateRouteListPdf($routeId),
            '01-工艺路线清单.pdf'
        );
        
        // 2. 工序卡片
        foreach ($route['operations'] as $op) {
            $zip->addFile(
                $this->generateOperationCardPdf($op['id']),
                "02-工序卡片-{$op['operation_no']}{$op['operation_name']}.pdf"
            );
        }
        
        // 3. 材料定额汇总
        $zip->addFile(
            $this->generateMaterialSummaryPdf($routeId),
            '03-材料定额汇总.pdf'
        );
        
        // 4. 工时定额汇总
        $zip->addFile(
            $this->generateTimeSummaryPdf($routeId),
            '04-工时定额汇总.pdf'
        );
        
        $zip->close();
        
        return $zipPath;
    }
}
```
