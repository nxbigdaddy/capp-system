## 一、集成概述

### 1.1 集成目标

- **统一身份认证**：通过企业微信扫码登录，免密访问 CAPP 系统
- **组织架构同步**：自动同步企业微信的部门、人员信息
- **消息推送**：审批待办、变更通知、催办等实时推送

### 1.2 集成范围

| 功能 | 优先级 | 说明 |
|------|--------|------|
| OAuth2.0 扫码登录 | P0 | 用户免密登录 |
| 通讯录同步 | P0 | 部门、人员信息同步 |
| 审批待办推送 | P0 | 审批消息卡片推送 |
| 变更通知推送 | P0 | 工艺变更 Markdown 通知 |
| 超时催办 | P1 | 审批超时提醒 |
| 应用首页 | P1 | 企业微信工作台入口 |

### 1.3 前置条件

- 已注册企业微信企业号
- 已创建自建应用（CAPP 应用）
- 已配置可信域名、回调域名
- 已获取 CorpID、AgentID、Secret

```php
    /**
     * 批量同步用户到本地数据库
     */
    private function batchSyncUsers($users) {
        $db = Database::getMysql();
        $created = 0;
        $updated = 0;
        $failed = 0;
        
        $db->beginTransaction();
        try {
            $findStmt = $db->prepare("SELECT id FROM capp_user WHERE wework_user_id = ? AND is_deleted = 0");
            $insertStmt = $db->prepare("
                INSERT INTO capp_user (wework_user_id, name, mobile, department, position, email, avatar, status)
                VALUES (?, ?, ?, ?, ?, ?, ?, 1)
            ");
            $updateStmt = $db->prepare("
                UPDATE capp_user SET name = ?, mobile = ?, department = ?, position = ?, email = ?, avatar = ?, updated_at = NOW()
                WHERE wework_user_id = ?
            ");
            
            foreach ($users as $user) {
                $findStmt->execute([$user['userid']]);
                $exists = $findStmt->fetch();
                
                if ($exists) {
                    $updateStmt->execute([
                        $user['name'], $user['mobile'], $user['department'],
                        $user['position'], $user['email'], $user['avatar'], $user['userid']
                    ]);
                    $updated++;
                } else {
                    $insertStmt->execute([
                        $user['userid'], $user['name'], $user['mobile'],
                        $user['department'], $user['position'], $user['email'], $user['avatar']
                    ]);
                    $created++;
                }
            }
            
            $db->commit();
        } catch (Exception $e) {
            $db->rollBack();
            throw $e;
        }
        
        return [
            'total' => count($users),
            'created' => $created,
            'updated' => $updated,
            'failed' => $failed,
        ];
    }

        $response = curl_exec($ch);

    private function httpPost($url, $data) {
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
        $response = curl_exec($ch);
        curl_close($ch);
        return json_decode($response, true);
    }
    
    private function getChangeNoticeReceivers($notice) {
```

## 四、消息回调处理

### 4.1 回调配置

企业微信后台配置接收消息 URL：`https://capp.company.com/api/wework/callback`

### 4.2 回调处理代码

```php
// src/Integration/WeWork/CallbackController.php

class CallbackController extends Controller {
    
    /**
     * 企业微信消息回调
     */
    public function handle() {
        $config = config('wework');
        
        // 1. 验证签名（首次配置时）
        if ($_SERVER['REQUEST_METHOD'] === 'GET') {
            return $this->verifyUrl($config);
        }
        
        // 2. 处理消息
        $xml = file_get_contents('php://input');
        $data = $this->decryptMessage($xml, $config);
        
        $msgType = $data['MsgType'] ?? '';
        $event = $data['Event'] ?? '';
        
        // 处理不同事件
        switch ($event) {
            case 'change_contact':
                $this->handleContactChange($data);
                break;
            case 'enter_agent':
                $this->handleEnterAgent($data);
                break;
            default:
                break;
        }
        
        // 返回成功响应
        echo 'success';
    }
    
    /**
     * 验证 URL（企业微信首次配置回调时）
     */
    private function verifyUrl($config) {
        $signature = $_GET['msg_signature'] ?? '';
        $timestamp = $_GET['timestamp'] ?? '';
        $nonce = $_GET['nonce'] ?? '';
        $echostr = $_GET['echostr'] ?? '';
        
        // 验证签名并解密
        $verify = $this->verifySignature($signature, $timestamp, $nonce, $echostr, $config);
        if ($verify) {
            echo $echostr;
        } else {
            http_response_code(403);
            echo 'verify failed';
        }
        exit;
    }
    
    /**
     * 处理通讯录变更事件
     */
    private function handleContactChange($data) {
        $changeType = $data['ChangeType'] ?? '';
        $userId = $data['UserID'] ?? '';
        
        switch ($changeType) {
            case 'create_user':
            case 'update_user':
                // 同步单个用户
                $userInfo = (new WeWorkService())->getUserDetail($userId);
                (new UserService())->findOrCreateByWework($userInfo);
                break;
            case 'delete_user':
                // 禁用用户
                (new UserService())->disableByWeworkId($userId);
                break;
            case 'create_party':
            case 'update_party':
                // 部门变更（可选处理）
                break;
        }
    }
    
    /**
     * 处理用户进入应用事件
     */
    private function handleEnterAgent($data) {
        $userId = $data['FromUserName'] ?? '';
        // 记录用户访问日志
        AuditLog::log('wework', 0, 'enter_agent', 0, null, ['wework_user_id' => $userId]);
    }
    
    private function verifySignature($signature, $timestamp, $nonce, $echostr, $config) {
        // 使用企业微信 SDK 验证签名
        // 简化示例，实际需使用官方加密库
        $token = YOUR_SECRET'token'];
        $tmpArr = [$token, $timestamp, $nonce, $echostr];
        sort($tmpArr, SORT_STRING);
        $tmpStr = implode($tmpArr);
        $tmpStr = sha1($tmpStr);
        return $tmpStr === $signature;
    }
    
    private function decryptMessage($xml, $config) {
        // 使用企业微信官方解密库解密消息
        // 简化示例
```

```php
// scripts/sync_wework_users.php

require_once __DIR__ . '/../src/bootstrap.php';

    $result = (new WeWorkService())->syncUsers();
    $logger->info('同步完成', $result);
    
    // 记录同步日志
    $db = Database::getMysql();
    $stmt = $db->prepare("
        INSERT INTO capp_sync_log (sync_type, direction, total_count, success_count, fail_count, started_at, completed_at)
        VALUES ('wework', 'in', ?, ?, ?, NOW(), NOW())
    ");
    $stmt->execute([
        $result['total'],
        $result['created'] + $result['updated'],
        $result['failed'],
    ]);
} catch (Exception $e) {
    $logger->error('同步失败: ' . $e->getMessage());
}
```
