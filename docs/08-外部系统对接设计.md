## 一、集成总览

### 1.1 集成系统

| 系统 | 厂商 | 对接方式 | 数据流向 | 优先级 |
|------|------|---------|---------|--------|
| CAXA PLM | 数码大方 | SQL Server 只读 + Web API | PLM→CAPP：产品结构/EBOM/图纸 | P0 |
| 用友 U8 | 用友 | SQL Server 只读 + EAI WebService | U8→CAPP：物料/库存/订单；CAPP→U8：定额/工艺 | P0 |
| MES | 定制 | RESTful API（双向） | CAPP→MES：工艺/SOP/变更；MES→CAPP：报工/质量 | P0 |

### 1.2 集成架构

```
┌─────────────────────────────────────────────────────────────┐
│                         CAPP 系统                            │
```

## 二、CAXA PLM 对接

### 2.1 对接方式

- **数据库**：SQL Server 只读连接，读取产品结构、EBOM、物料、图纸信息
- **频率**：每日凌晨同步，支持手动触发
- **数据范围**：已发布的产品和零部件

### 2.4 同步服务

```php
// src/Integration/Caxa/CaxaSyncService.php

class CaxaSyncService {
    private $db;
    private $cappDb;
    
    public function __construct() {
        $this->db = Database::getSqlServer('caxa');
        $this->cappDb = Database::getMysql();
    }
    
    /**
     * 同步产品/零部件
     */
    public function syncProducts() {
        $config = config('caxa.sync');
        
        // 获取 PLM 中已发布的产品
        $stmt = $this->db->query("
            SELECT ProductID, ProductCode, ProductName, Specification, 
                   Material, DrawingNo, Unit, Weight, ParentID, Status
            FROM {$config['products_table']}
            WHERE Status = 'Published'
        ");
        $products = $stmt->fetchAll(PDO::FETCH_ASSOC);
        
        $synced = 0;
        $failed = 0;
        
        $this->cappDb->beginTransaction();
        try {
            $findStmt = $this->cappDb->prepare("SELECT id FROM capp_product WHERE source_system = 'plm' AND source_id = ?");
            $insertStmt = $this->cappDb->prepare("
                INSERT INTO capp_product (code, name, specification, material_type, drawing_no, unit, weight, parent_id, source_system, source_id, status)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'plm', ?, 1)
            ");
            $updateStmt = $this->cappDb->prepare("
                UPDATE capp_product SET code = ?, name = ?, specification = ?, material_type = ?, drawing_no = ?, unit = ?, weight = ?, parent_id = ?, updated_at = NOW()
                WHERE source_system = 'plm' AND source_id = ?
            ");
            
            foreach ($products as $product) {
                try {
                    $findStmt->execute([$product['ProductID']]);
                    $exists = $findStmt->fetch();
                    
                    if ($exists) {
                        $updateStmt->execute([
                            $product['ProductCode'],
                            $product['ProductName'],
                            $product['Specification'],
                            $product['Material'],
                            $product['DrawingNo'],
                            $product['Unit'],
                            $product['Weight'],
                            $product['ParentID'],
                            $product['ProductID'],
                        ]);
                    } else {
                        $insertStmt->execute([
                            $product['ProductCode'],
                            $product['ProductName'],
                            $product['Specification'],
                            $product['Material'],
                            $product['DrawingNo'],
                            $product['Unit'],
                            $product['Weight'],
                            $product['ParentID'],
                            $product['ProductID'],
                        ]);
                    }
                    $synced++;
                } catch (Exception $e) {
                    $failed++;
                    Logger::error('PLM产品同步失败: ' . $e->getMessage(), ['product' => $product]);
                }
            }
            
            $this->cappDb->commit();
        } catch (Exception $e) {
            $this->cappDb->rollBack();
            throw $e;
        }
        
        return ['synced' => $synced, 'failed' => $failed];
    }
    
    /**
     * 同步 BOM 结构
     */
    public function syncBOM() {
        $config = config('caxa.sync');
        
        $stmt = $this->db->query("
            SELECT ParentProductID, ChildProductID, Quantity, SortOrder
            FROM {$config['bom_table']}
        ");
        $boms = $stmt->fetchAll(PDO::FETCH_ASSOC);
        
        foreach ($boms as $bom) {
            // 更新 CAPP 中的 parent_id 关系
            $this->cappDb->prepare("
                UPDATE capp_product p1
                INNER JOIN capp_product p2 ON p2.source_id = ? AND p2.source_system = 'plm'
                SET p1.parent_id = p2.id
                WHERE p1.source_id = ? AND p1.source_system = 'plm'
            ")->execute([$bom['ParentProductID'], $bom['ChildProductID']]);
        }
        
        return ['synced' => count($boms)];
    }
    
    /**
     * 获取图纸信息
     */
    public function getDrawingInfo($plmProductId) {
        $config = config('caxa.sync');
        
        $stmt = $this->db->prepare("
            SELECT DrawingNo, DrawingName, FilePath, Version
            FROM {$config['drawing_table']}
            WHERE ProductID = ?
        ");
        $stmt->execute([$plmProductId]);
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
}
```

### 3.4 同步服务

```php
// src/Integration/U8/U8SyncService.php

class U8SyncService {
    private $db;
    private $cappDb;
    
    public function __construct() {
        $this->db = Database::getSqlServer('u8');
        $this->cappDb = Database::getMysql();
    }
    
    /**
     * 同步物料档案
     */
    public function syncMaterials() {
        $config = config('u8.sync');
        
        $stmt = $this->db->query("
            SELECT cInvCode, cInvName, cInvStd, cInvCCode, cComUnitCode, iInvWeight
            FROM {$config['inventory_table']}
            WHERE bService = 0
        ");
        $materials = $stmt->fetchAll(PDO::FETCH_ASSOC);
        
        $synced = 0;
        $this->cappDb->beginTransaction();
        try {
            $findStmt = $this->cappDb->prepare("SELECT id FROM capp_material WHERE source_system = 'u8' AND source_id = ?");
            $insertStmt = $this->cappDb->prepare("
                INSERT INTO capp_material (code, name, specification, category_id, unit, source_system, source_id, status)
                VALUES (?, ?, ?, ?, ?, 'u8', ?, 1)
            ");
            $updateStmt = $this->cappDb->prepare("
                UPDATE capp_material SET code = ?, name = ?, specification = ?, category_id = ?, unit = ?, updated_at = NOW()
                WHERE source_system = 'u8' AND source_id = ?
            ");
            
            foreach ($materials as $material) {
                $findStmt->execute([$material['cInvCode']]);
                $exists = $findStmt->fetch();
                
                if ($exists) {
                    $updateStmt->execute([
                        $material['cInvCode'],
                        $material['cInvName'],
                        $material['cInvStd'],
                        $material['cInvCCode'],
                        $material['cComUnitCode'],
                        $material['cInvCode'],
                    ]);
                } else {
                    $insertStmt->execute([
                        $material['cInvCode'],
                        $material['cInvName'],
                        $material['cInvStd'],
                        $material['cInvCCode'],
                        $material['cComUnitCode'],
                        $material['cInvCode'],
                    ]);
                }
                $synced++;
            }
            
            $this->cappDb->commit();
        } catch (Exception $e) {
            $this->cappDb->rollBack();
            throw $e;
        }
        
        return ['synced' => $synced];
    }
    
    /**
     * 回写材料定额到 U8
     */
    public function writeBackMaterialQuota($routeId) {
        $route = (new ProcessRouteService())->getDetail($routeId);
        $quotas = (new QuotaService())->getMaterialByRoute($routeId);
        
        $xml = '<?xml version="1.0" encoding="utf-8"?>';
        $xml .= '<ufinterface roottag="inventory" billtype="" replace="Y">';
        
        foreach ($quotas as $quota) {
            $xml .= '<inventory>';
            $xml .= '<header>';
            $xml .= '<cInvCode>' . htmlspecialchars($quota['material_code']) . '</cInvCode>';
            $xml .= '</header>';
            $xml .= '<body>';
            $xml .= '<iGrossWeight>' . $quota['gross_weight'] . '</iGrossWeight>';
            $xml .= '<iNetWeight>' . $quota['net_weight'] . '</iNetWeight>';
            $xml .= '<iUtilizationRate>' . $quota['utilization_rate'] . '</iUtilizationRate>';
            $xml .= '</body>';
            $xml .= '</inventory>';
        }
        
        $xml .= '</ufinterface>';
        
        return $this->callEai($xml);
    }
    
    /**
     * 回写工艺路线到 U8
     */
    public function writeBackProcessRoute($routeId) {
        $route = (new ProcessRouteService())->getDetail($routeId);
        
        $xml = '<?xml version="1.0" encoding="utf-8"?>';
        $xml .= '<ufinterface roottag="routing" billtype="" replace="Y">';
        $xml .= '<routing>';
        $xml .= '<cInvCode>' . htmlspecialchars($route['product_code']) . '</cInvCode>';
        $xml .= '<cVersion>' . $route['version'] . '</cVersion>';
        
        foreach ($route['operations'] as $op) {
            $xml .= '<operation>';
            $xml .= '<iOpSeq>' . $op['operation_no'] . '</iOpSeq>';
            $xml .= '<cOpName>' . htmlspecialchars($op['operation_name']) . '</cOpName>';
            $xml .= '<iLaborTime>' . ($op['standard_time'] ?? 0) . '</iLaborTime>';
            $xml .= '</operation>';
        }
        
        $xml .= '</routing>';
        $xml .= '</ufinterface>';
        
        return $this->callEai($xml);
    }
    
    /**
     * 调用 U8 EAI 接口
     */
    private function callEai($xml) {
        $config = config('u8.eai');
        
        $ch = curl_init($config['url']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $xml);
        curl_setopt($ch, CURLOPT_HTTPHEADER, [
            'Content-Type: text/xml; charset=utf-8',
        ]);
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        if ($httpCode !== 200) {
            throw new Exception('U8 EAI 调用失败，HTTP状态码: ' . $httpCode);
        }
        
        // 解析响应
        $result = simplexml_load_string($response);
        if ((string)$result['success'] !== '1') {
            throw new Exception('U8 EAI 处理失败: ' . (string)$result['errormessage']);
        }
        
        return true;
    }
}
```

## 四、MES 对接

### 4.1 对接方式

- **协议**：RESTful API（双向 HTTP/HTTPS）
- **认证**：AppKey + HMAC-SHA256 签名
- **数据格式**：JSON
- **频率**：工艺路线实时下发；报工数据定时拉取

### 4.2 配置

```php
// config/mes.php

return [
    'enabled' => true,
    'base_url' => 'http://mes.company.com/api',
    'app_key' => 'CAPP_APP_KEY',
    'app_secret' => 'CAPP_APP_SECRET',
    'timeout' => 30,
    'retry_times' => 3,
];
```

### 4.3 认证机制

```php
// src/Integration/Mes/MesClient.php

class MesClient {
    private $config;
    
    public function __construct() {
        $this->config = config('mes');
    }
    
    /**
     * 生成请求签名
     */
    private function generateSign($params, $timestamp) {
        $stringToSign = $this->config['app_key'] . $timestamp . json_encode($params);
        return hash_hmac('sha256', $stringToSign, $this->config['app_secret']);
    }
    
    /**
     * 发送请求
     */
    public function request($method, $path, $data = []) {
        $url = rtrim($this->config['base_url'], '/') . $path;
        $timestamp = time();
        $sign = $this->generateSign($data, $timestamp);
        
        $headers = [
            'Content-Type: application/json',
            'X-App-Key: ' . $this->config['app_key'],
            'X-Timestamp: ' . $timestamp,
            'X-Sign: ' . $sign,
        ];
        
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        curl_setopt($ch, CURLOPT_TIMEOUT, $this->config['timeout']);
        
        if ($method === 'POST') {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
        }
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        if ($httpCode !== 200) {
            throw new Exception('MES 接口调用失败，HTTP状态码: ' . $httpCode);
        }
        
        $result = json_decode($response, true);
        if (empty($result['success'])) {
            throw new Exception('MES 接口返回错误: ' . ($result['message'] ?? '未知错误'));
        }
        
        return $result['data'];
    }
    
    public function get($path) {
        return $this->request('GET', $path);
    }
    
    public function post($path, $data) {
        return $this->request('POST', $path, $data);
    }
}
```

### 4.4 下发工艺路线到 MES

```php
// src/Integration/Mes/MesSyncService.php

class MesSyncService {
    private $client;
    
    public function __construct() {
        $this->client = new MesClient();
    }
    
    /**
     * 下发工艺路线到 MES
     */
    public function pushProcessRoute($routeId) {
        $route = (new ProcessRouteService())->getDetail($routeId);
        
        if ($route['status'] !== 'published') {
            throw new Exception('仅已发布工艺路线可下发到 MES', 40001);
        }
        
        $data = [
            'route_code' => $route['code'],
            'product_code' => $route['product_code'],
            'product_name' => $route['product_name'],
            'version' => $route['version'],
            'operations' => [],
        ];
        
        foreach ($route['operations'] as $op) {
            $data['operations'][] = [
                'operation_no' => $op['operation_no'],
                'operation_name' => $op['operation_name'],
                'work_center_code' => $op['work_center_name'] ?? '',
                'standard_time' => $op['standard_time'],
                'description' => $op['description'],
                'is_key_point' => $op['is_key_point'],
                'sop_content' => $this->getSopContent($op['id']),
            ];
        }
        
        return $this->client->post('/v1/process-routes', $data);
    }
    
    /**
     * 下发变更通知到 MES
     */
    public function pushChangeNotice($ecnNo) {
        $notice = (new ChangeService())->getNoticeByNo($ecnNo);
        
        $data = [
            'ecn_no' => $ecnNo,
            'route_code' => $notice['route_code'],
            'change_content' => $notice['change_content'],
            'effective_time' => $notice['issued_at'],
        ];
        
        return $this->client->post('/v1/change-notices', $data);
    }
    
    /**
     * 从 MES 拉取报工数据
     */
    public function pullWorkReports($startDate, $endDate) {
        $data = [
            'start_date' => $startDate,
            'end_date' => $endDate,
        ];
        
        $reports = $this->client->post('/v1/work-reports/query', $data);
        
        // 保存到本地
        foreach ($reports as $report) {
            $this->saveWorkReport($report);
        }
        
        return count($reports);
    }
    
    /**
     * 保存报工数据
     */
    private function saveWorkReport($report) {
        $db = Database::getMysql();
        
        // 更新工时定额的实际工时
        $stmt = $db->prepare("
            INSERT INTO capp_quota_time (route_operation_id, actual_time)
            VALUES ((SELECT id FROM capp_route_operation WHERE route_id = ? AND operation_no = ?), ?)
            ON DUPLICATE KEY UPDATE actual_time = VALUES(actual_time)
        ");
        $stmt->execute([
            $report['route_id'],
            $report['operation_no'],
            $report['actual_time'],
        ]);
    }
    
    /**
     * 获取 SOP 内容
     */
    private function getSopContent($routeOperationId) {
        $stmt = Database::getMysql()->prepare("SELECT operation_steps FROM capp_operation_card WHERE route_operation_id = ?");
        $stmt->execute([$routeOperationId]);
        return $stmt->fetchColumn() ?? '';
    }
}
```

```php
            INSERT INTO capp_quality_report (route_id, operation_no, inspection_item, result, inspector, inspect_time)
            VALUES (?, ?, ?, ?, ?, ?)
        ");
        $stmt->execute([
            $data['route_id'],
            $data['operation_no'],
            $data['inspection_item'],
            $data['result'],
            $data['inspector'],
            $data['inspect_time'],
        ]);
        
        Response::success(['received' => true]);
    }
    
    /**
     * 验证 MES 请求签名
     */
    private function verifyMesSign() {
        $config = config('mes');
        $appKey = $_SERVER['HTTP_X_APP_KEY'] ?? '';
        $timestamp = $_SERVER['HTTP_X_TIMESTAMP'] ?? '';
        $sign = $_SERVER['HTTP_X_SIGN'] ?? '';
        
        if ($appKey !== $config['app_key']) {
            Response::error(40301, 'AppKey 无效');
        }
        
        // 检查时间戳（5分钟内有效）
        if (abs(time() - (int)$timestamp) > 300) {
            Response::error(40301, '请求已过期');
        }
        
        $body = file_get_contents('php://input');
        $stringToSign = $appKey . $timestamp . $body;
        $expectedSign = hash_hmac('sha256', $stringToSign, $config['app_secret']);
        
        if (!hash_equals($expectedSign, $sign)) {
            Response::error(40301, '签名无效');
        }
    }
```

## 五、同步任务调度

```php
// scripts/sync_external_data.php

require_once __DIR__ . '/../src/bootstrap.php';

$logger = new Logger('sync');
$logger->info('开始外部系统数据同步');

$results = [];

// 1. 同步 PLM 产品数据
try {
    $logger->info('同步 PLM 产品数据...');
    $result = (new CaxaSyncService())->syncProducts();
    $results['plm_products'] = $result;
    $logger->info('PLM 产品同步完成', $result);
} catch (Exception $e) {
    $logger->error('PLM 产品同步失败: ' . $e->getMessage());
    $results['plm_products'] = ['error' => $e->getMessage()];
}

// 2. 同步 PLM BOM
try {
    $logger->info('同步 PLM BOM...');
    $result = (new CaxaSyncService())->syncBOM();
    $results['plm_bom'] = $result;
    $logger->info('PLM BOM 同步完成', $result);
} catch (Exception $e) {
    $logger->error('PLM BOM 同步失败: ' . $e->getMessage());
    $results['plm_bom'] = ['error' => $e->getMessage()];
}

// 3. 同步 U8 物料
try {
    $logger->info('同步 U8 物料...');
    $result = (new U8SyncService())->syncMaterials();
    $results['u8_materials'] = $result;
    $logger->info('U8 物料同步完成', $result);
} catch (Exception $e) {
    $logger->error('U8 物料同步失败: ' . $e->getMessage());
    $results['u8_materials'] = ['error' => $e->getMessage()];
}

// 4. 同步 MES 报工数据
try {
    $logger->info('同步 MES 报工数据...');
    $startDate = date('Y-m-d', strtotime('-7 days'));
    $endDate = date('Y-m-d');
    $result = (new MesSyncService())->pullWorkReports($startDate, $endDate);
    $results['mes_work_reports'] = ['synced' => $result];
    $logger->info('MES 报工数据同步完成', ['synced' => $result]);
} catch (Exception $e) {
    $logger->error('MES 报工数据同步失败: ' . $e->getMessage());
    $results['mes_work_reports'] = ['error' => $e->getMessage()];
}

// 记录同步日志
$db = Database::getMysql();
foreach ($results as $type => $result) {
    $stmt = $db->prepare("
        INSERT INTO capp_sync_log (sync_type, direction, total_count, success_count, fail_count, error_detail, started_at, completed_at)
        VALUES (?, ?, ?, ?, ?, ?, NOW(), NOW())
    ");
    $stmt->execute([
        $type,
        'in',
        $result['synced'] ?? 0,
        $result['synced'] ?? 0,
        $result['failed'] ?? 0,
        $result['error'] ?? null,
    ]);
}

$logger->info('外部系统数据同步全部完成');
```
