### 1.1 模块职责

- 用户登录/登出

### 1.2 核心类设计

```php
// src/Modules/Auth/LoginService.php

class LoginService {
    private $userService;
    private $weworkService;
    
    public function __construct() {
        $this->userService = new UserService();
        $this->weworkService = new WeWorkService();
    }
    
    /**
     * 账号密码登录
     * @param string $username 工号或手机号
     * @param string $password 密码
     * @return array {token, user}
     */
    public function loginByPassword($username, $password) {
        $user = $this->userService->findByUsername($username);
        if (!$user) {
            throw new Exception('账号或密码错误', 40101);
        }
        
        if (!password_verify($password, $user['password_hash'])) {
            throw new Exception('账号或密码错误', 40101);
        }
        
        if ($user['status'] != 1) {
            throw new Exception('账号已禁用', 40102);
        }
        
        // 更新登录时间
        $this->userService->updateLoginInfo($user['id']);
        
        $token = YOUR_SECRET
            'user_id' => $user['id'],
            'user_name' => $user['name'],
        ]);
        
        return ['token' => $token, 'user' => $user];
    }
    
    /**
     * 企业微信扫码登录回调
     * @param string $code 授权码
     * @return array {token, user}
     */
    public function loginByWeWork($code) {
        // 1. 获取用户身份信息
        $userInfo = $this->weworkService->getUserInfo($code);
        
        // 2. 查找或创建本地用户
        $user = $this->userService->findOrCreateByWework($userInfo);
        
        // 3. 更新登录信息
        $this->userService->updateLoginInfo($user['id']);
        
        $token = YOUR_SECRET
            'user_id' => $user['id'],
            'user_name' => $user['name'],
        ]);
        
        return ['token' => $token, 'user' => $user];
    }
    
    /**
     * 登出
     * @param string $token
     */
    public function logout($token) {
        // Token 加入黑名单（Redis 中设置过期时间）
        RedisCache::set('token_blacklist:' . md5($token), 1, 7200);
    }
    
    /**
     * 刷新 Token
     * @param string $refreshToken
     * @return string 新 token
     */
    public function refreshToken($refreshToken) {
        $payload = JwtHelper::decode($refreshToken);
        if (!$payload) {
            throw new Exception('Token 无效', 40101);
        }
        
        $user = $this->userService->findById($payload['user_id']);
        if (!$user || $user['status'] != 1) {
            throw new Exception('用户不存在或已禁用', 40102);
        }
        
        return JwtHelper::encode([
            'user_id' => $user['id'],
            'user_name' => $user['name'],
        ]);
    }
}
```

## 二、系统管理模块 (System)

### 2.1 用户管理

```php
// src/Modules/System/UserService.php

class UserService {
    private $db;
    
    public function __construct() {
        $this->db = Database::getMysql();
    }
    
    /**
     * 搜索用户
     */
    public function search($filters, $page, $pageSize) {
        $where = ['is_deleted = 0'];
        $params = [];
        
        if (!empty($filters['keyword'])) {
            $where[] = '(name LIKE ? OR employee_no LIKE ? OR mobile LIKE ?)';
            $keyword = '%' . $filters['keyword'] . '%';
            $params = array_merge($params, [$keyword, $keyword, $keyword]);
        }
        
        if (!empty($filters['status'])) {
            $where[] = 'status = ?';
            $params[] = $filters['status'];
        }
        
        $whereStr = implode(' AND ', $where);
        
        // 总数
        $countStmt = $this->db->prepare("SELECT COUNT(*) FROM capp_user WHERE {$whereStr}");
        $countStmt->execute($params);
        $total = $countStmt->fetchColumn();
        
        // 列表
        $offset = ($page - 1) * $pageSize;
        $stmt = $this->db->prepare("SELECT id, name, employee_no, mobile, department, position, status, last_login_at FROM capp_user WHERE {$whereStr} ORDER BY id DESC LIMIT {$offset}, {$pageSize}");
        $stmt->execute($params);
        $list = $stmt->fetchAll();
        
        return ['list' => $list, 'total' => $total];
    }
    
    /**
     * 根据企业微信信息查找或创建用户
     */
    public function findOrCreateByWework($weworkUser) {
        $stmt = $this->db->prepare("SELECT * FROM capp_user WHERE wework_user_id = ? AND is_deleted = 0");
        $stmt->execute([$weworkUser['userid']]);
        $user = $stmt->fetch();
        
        if ($user) {
            // 更新信息
            $stmt = $this->db->prepare("UPDATE capp_user SET name = ?, department = ?, position = ?, avatar = ?, updated_at = NOW() WHERE id = ?");
            $stmt->execute([$weworkUser['name'], $weworkUser['department'], $weworkUser['position'], $weworkUser['avatar'], $user['id']]);
            return array_merge($user, [
                'name' => $weworkUser['name'],
                'department' => $weworkUser['department'],
                'position' => $weworkUser['position'],
            ]);
        }
        
        // 创建新用户
        $stmt = $this->db->prepare("INSERT INTO capp_user (wework_user_id, name, mobile, department, position, avatar, status) VALUES (?, ?, ?, ?, ?, ?, 1)");
        $stmt->execute([
            $weworkUser['userid'],
            $weworkUser['name'],
            $weworkUser['mobile'] ?? null,
            $weworkUser['department'],
            $weworkUser['position'],
            $weworkUser['avatar'] ?? null,
        ]);
        
        return $this->findById($this->db->lastInsertId());
    }
    
    /**
     * 更新用户角色
     */
    public function updateRoles($userId, $roleIds) {
        $this->db->beginTransaction();
        try {
            // 删除旧角色
            $stmt = $this->db->prepare("DELETE FROM capp_user_role WHERE user_id = ?");
            $stmt->execute([$userId]);
            
            // 添加新角色
            $stmt = $this->db->prepare("INSERT INTO capp_user_role (user_id, role_id) VALUES (?, ?)");
            foreach ($roleIds as $roleId) {
                $stmt->execute([$userId, $roleId]);
            }
            
            $this->db->commit();
            
            // 清除权限缓存
            RedisCache::del('user_permissions:' . $userId);
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    /**
     * 获取用户权限列表
     */
    public function getPermissions($userId) {
        // 先从缓存获取
        $cacheKey = 'user_permissions:' . $userId;
        $perms = RedisCache::get($cacheKey);
        if ($perms) {
            return json_decode($perms, true);
        }
        
        $stmt = $this->db->prepare("
            SELECT DISTINCT p.code 
            FROM capp_permission p
            INNER JOIN capp_role_permission rp ON p.id = rp.permission_id
            INNER JOIN capp_user_role ur ON rp.role_id = ur.role_id
            WHERE ur.user_id = ?
        ");
        $stmt->execute([$userId]);
        $perms = $stmt->fetchAll(PDO::FETCH_COLUMN);
        
        // 缓存1小时
        RedisCache::set($cacheKey, json_encode($perms), 3600);
        
        return $perms;
    }
    
    /**
     * 检查权限
     */
    public function checkPermission($userId, $permissionCode) {
        $perms = $this->getPermissions($userId);
        return in_array($permissionCode, $perms);
    }
}
```

## 三、工艺路线模块 (ProcessRoute)

```php
// src/Modules/ProcessRoute/ProcessRouteService.php

class ProcessRouteService {
    private $db;
    private $approvalService;
    private $qrGenerator;
    
    public function __construct() {
        $this->db = Database::getMysql();
        $this->approvalService = new ApprovalService();
        $this->qrGenerator = new QrGenerator();
    }
    
    /**
     * 创建工艺路线
     */
    public function create($data, $userId) {
        $this->db->beginTransaction();
        try {
            // 生成编码
            $code = $this->generateCode();
            
            $stmt = $this->db->prepare("
                INSERT INTO capp_process_route 
                (code, product_id, product_code, product_name, name, version, status, description, created_by)
                VALUES (?, ?, ?, ?, ?, 'V1.0', 'drafting', ?, ?)
            ");
            $stmt->execute([
                $code,
                $data['product_id'],
                $data['product_code'],
                $data['product_name'],
                $data['name'],
                $data['description'] ?? null,
                $userId,
            ]);
            
            $routeId = $this->db->lastInsertId();
            
            // 添加工序
            if (!empty($data['operations'])) {
                $this->saveOperations($routeId, $data['operations']);
            }
            
            $this->db->commit();
            
            // 记录日志
            AuditLog::log('process_route', $routeId, 'create', $userId, null, ['code' => $code]);
            
            return $routeId;
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    /**
     * 更新工艺路线（仅草稿状态可编辑）
     */
    public function update($routeId, $data, $userId) {
        $route = $this->getById($routeId);
        if ($route['status'] !== 'drafting') {
            throw new Exception('仅草稿状态可编辑', 40001);
        }
        
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare("
                UPDATE capp_process_route 
                SET product_id = ?, product_code = ?, product_name = ?, name = ?, description = ?, updated_at = NOW()
                WHERE id = ?
            ");
            $stmt->execute([
                $data['product_id'],
                $data['product_code'],
                $data['product_name'],
                $data['name'],
                $data['description'] ?? null,
                $routeId,
            ]);
            
            // 更新工序
            if (!empty($data['operations'])) {
                $this->saveOperations($routeId, $data['operations']);
            }
            
            $this->db->commit();
            
            AuditLog::log('process_route', $routeId, 'update', $userId);
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    /**
     * 保存工序列表
     */
    private function saveOperations($routeId, $operations) {
        // 删除旧工序
        $stmt = $this->db->prepare("DELETE FROM capp_route_operation WHERE route_id = ?");
        $stmt->execute([$routeId]);
        
        // 插入新工序
        $stmt = $this->db->prepare("
            INSERT INTO capp_route_operation 
            (route_id, operation_no, operation_name, std_operation_id, work_center_id, description, 
             equipment_ids, tooling_ids, standard_time, sort_order, predecessor_no, is_key_point, parameters)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        ");
        
        foreach ($operations as $index => $op) {
            $stmt->execute([
                $routeId,
                $op['operation_no'],
                $op['operation_name'],
                $op['std_operation_id'] ?? null,
                $op['work_center_id'] ?? null,
                $op['description'] ?? null,
                !empty($op['equipment_ids']) ? implode(',', $op['equipment_ids']) : null,
                !empty($op['tooling_ids']) ? implode(',', $op['tooling_ids']) : null,
                $op['standard_time'] ?? null,
                $index,
                $op['predecessor_no'] ?? null,
                $op['is_key_point'] ?? 0,
                !empty($op['parameters']) ? json_encode($op['parameters']) : null,
            ]);
        }
        
        // 更新总工时
        $this->updateTotalTime($routeId);
    }
    
    /**
     * 更新总工时
     */
    private function updateTotalTime($routeId) {
        $stmt = $this->db->prepare("SELECT SUM(standard_time) FROM capp_route_operation WHERE route_id = ?");
        $stmt->execute([$routeId]);
        $totalTime = $stmt->fetchColumn() ?? 0;
        
        $stmt = $this->db->prepare("UPDATE capp_process_route SET total_time = ? WHERE id = ?");
        $stmt->execute([$totalTime, $routeId]);
    }
    
    /**
     * 提交审核
     */
    public function submitForApproval($routeId, $userId) {
        $route = $this->getById($routeId);
        if ($route['status'] !== 'drafting') {

            AuditLog::log('process_route', $routeId, 'submit', $userId);

            $this->db->rollBack();
            throw $e;
        }
    }
    
    /**
     * 审批通过
     */
    public function approve($routeId, $userId, $data) {
        $route = $this->getById($routeId);
        if ($route['status'] !== 'submitted') {
            throw new Exception('工艺路线状态不正确', 40001);
        }
        
        $this->db->beginTransaction();
        try {
            // 记录审批
            $this->approvalService->approve($routeId, $userId, $data['comment'] ?? '');
            
            // 检查是否所有节点已通过
            if ($this->approvalService->isCompleted($routeId)) {
                // 更新为已批准
                $stmt = $this->db->prepare("UPDATE capp_process_route SET status = 'approved', approved_by = ?, approved_at = NOW() WHERE id = ?");
                $stmt->execute([$userId, $routeId]);
            }
            
            $this->db->commit();
            
            AuditLog::log('process_route', $routeId, 'approve', $userId);
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    /**
     * 审批驳回
     */
    public function reject($routeId, $userId, $data) {
        $this->db->beginTransaction();
        try {
            // 记录审批

            $stmt->execute([$routeId]);
            
            $this->db->commit();
            
            AuditLog::log('process_route', $routeId, 'reject', $userId);
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }

    /**
     * 搜索
     */
    public function search($filters, $page, $pageSize) {
        $where = ['r.is_deleted = 0'];
        $params = [];
        
        if (!empty($filters['keyword'])) {
            $where[] = '(r.code LIKE ? OR r.name LIKE ? OR r.product_code LIKE ? OR r.product_name LIKE ?)';

        }
        
        if (!empty($filters['status'])) {
            $where[] = 'r.status = ?';
            $params[] = $filters['status'];
```

```php
            ");
            
            foreach ($items as $item) {

                           + ($item['assist_time'] ?? 0) + ($item['allowance_time'] ?? 0);

                ]);
            }

    /**
     * 基于历史数据自动计算定额
     */
    public function calculate($data) {
        $routeOperationId = $data['route_operation_id'];
        $method = $data['method'] ?? 'estimate';
        
        if ($method === 'statistic') {
            // 基于 MES 报工数据统计
            return $this->calculateByStatistics($routeOperationId);
        }
        
        if ($method === 'technical') {
            // 基于工艺参数技术计算
            return $this->calculateByTechnical($data);
        }
        
        // 默认估算法
        return [
            'prepare_time' => $data['prepare_time'] ?? 0,
            'process_time' => $data['process_time'] ?? 0,
            'assist_time' => $data['assist_time'] ?? 0,
            'allowance_time' => $data['allowance_time'] ?? 0,
        ];
    }
    
    /**
     * 基于 MES 历史数据统计定额
     */
    private function calculateByStatistics($routeOperationId) {
        // 从 MES 获取历史报工数据
        $mesData = (new MesService())->getWorkReportByOperation($routeOperationId, 30);
        
        if (empty($mesData)) {
            throw new Exception('无历史数据', 40001);
        }
        
        $times = array_column($mesData, 'actual_time');
        sort($times);
        
        // 去掉最高和最低 10%，取平均值
        $count = count($times);
        $start = (int)($count * 0.1);
        $end = (int)($count * 0.9);
        $validTimes = array_slice($times, $start, $end - $start);
        
        $avgTime = array_sum($validTimes) / count($validTimes);
        
        return [
            'prepare_time' => 0,
            'process_time' => round($avgTime, 2),
            'assist_time' => 0,
            'allowance_time' => round($avgTime * 0.1, 2), // 10% 宽放
            'sample_count' => $count,
        ];
    }
    
    /**
     * 基于工艺参数技术计算
     */
    private function calculateByTechnical($data) {
        // 示例：车削加工时间计算
        // T = L / (f * n) + 换刀时间 + 测量时间
        $length = $data['length'] ?? 0; // 加工长度 mm
        $feed = $data['feed'] ?? 0; // 进给量 mm/r
        $speed = $data['speed'] ?? 0; // 转速 r/min
        $toolChangeTime = $data['tool_change_time'] ?? 0;
        $measureTime = $data['measure_time'] ?? 0;
        
        if ($feed > 0 && $speed > 0) {
            $processTime = ($length / ($feed * $speed)) * 60; // 转换为分钟
        } else {
            $processTime = 0;
        }
        
        return [
            'prepare_time' => $toolChangeTime,
            'process_time' => round($processTime, 2),
            'assist_time' => $measureTime,
            'allowance_time' => round($processTime * 0.15, 2),
        ];
    }
}
```

```php
        $stmt = $this->db->prepare("SELECT * FROM capp_change_request WHERE id = ?");

        // 获取关联的工艺路线
        $stmt = $this->db->prepare("SELECT * FROM capp_process_route WHERE id = ?");
        $stmt->execute([$ecr['business_id']]);
        $route = $stmt->fetch();
        
        return ['route' => $route];
    }
    
    private function applyChange($ecrId) {

        $stmt->execute([$ecrId]);
        $ecr = $stmt->fetch();
        
        // 变更工艺路线状态为需要更新
        $stmt = $this->db->prepare("UPDATE capp_process_route SET status = 'drafting' WHERE id = ?");
        $stmt->execute([$ecr['business_id']]);
    }

        // 发送企业微信通知给相关人员
```
