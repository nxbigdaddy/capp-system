### 4.1 统一入口与路由

```php
// public/index.php
require_once __DIR__ . '/../src/bootstrap.php';

$uri = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);
$uri = trim($uri, '/');

// 路由映射表
$routes = [
    // 认证
    'login' => ['Auth\LoginController', 'index'],
    'auth/wework/callback' => ['Auth\LoginController', 'weworkCallback'],
    'api/auth/login' => ['Auth\ApiController', 'login'],
    'api/auth/refresh' => ['Auth\ApiController', 'refresh'],
    'api/auth/user-info' => ['Auth\ApiController', 'userInfo'],
    
    // 系统管理
    'admin/users' => ['System\UserController', 'index'],
    'api/admin/users' => ['System\UserApiController', 'index'],
    
    // 工艺路线
    'process/routes' => ['ProcessRoute\RouteController', 'index'],
    'api/process/routes' => ['ProcessRoute\RouteApiController', 'index'],
    'api/process/routes/(\d+)' => ['ProcessRoute\RouteApiController', 'detail'],
    
    // ... 其他路由
];

// 匹配路由并执行
$matched = false;
foreach ($routes as $pattern => $handler) {
    if (preg_match('#^' . $pattern . '$#', $uri, $matches)) {
        array_shift($matches);
        $controller = new $handler[0]();
        call_user_func_array([$controller, $handler[1]], $matches);
        $matched = true;
        break;
    }
}

if (!$matched) {
    http_response_code(404);
    echo 'Page Not Found';
}
```

```php
     * 获取 SQL Server 连接（只读）
```

```php
        exit;
```

### 4.4 JWT 认证中间件

```php
// src/Common/JwtHelper.php

class JwtHelper {
    private static $secret = null;
    
    private static function getSecret() {
        if (self::$secret === null) {
            self::$secret = config('app.jwt_secret');
        }
        return self::$secret;
    }
    
    public static function encode($payload, $expiry = 7200) {
        $header = json_encode(['typ' => 'JWT', 'alg' => 'HS256']);
        $payload['iat'] = time();
        $payload['exp'] = time() + $expiry;
        
        $base64Header = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($header));
        $base64Payload = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode(json_encode($payload)));
        
        $signature = hash_hmac('sha256', $base64Header . "." . $base64Payload, self::getSecret(), true);
        $base64Signature = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($signature));
        
        return $base64Header . "." . $base64Payload . "." . $base64Signature;
    }
    
    public static function decode($token) {
        $parts = explode('.', $token);
        if (count($parts) !== 3) {
            return null;
        }
        
        $payload = json_decode(base64_decode(str_replace(['-', '_'], ['+', '/'], $parts[1])), true);
        
        if (!$payload || !isset($payload['exp']) || $payload['exp'] < time()) {
            return null;
        }
        
        // 验证签名
        $signature = hash_hmac('sha256', $parts[0] . "." . $parts[1], self::getSecret(), true);
        $base64Signature = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($signature));
        
        if (!hash_equals($base64Signature, $parts[2])) {
            return null;
        }
        
        return $payload;
    }
}
```

### 4.6 二维码生成

```php
// src/Common/QrGenerator.php

class QrGenerator {
    private $cacheDir;
    
    public function __construct() {
        $this->cacheDir = __DIR__ . '/../../public/qr/';
        if (!is_dir($this->cacheDir)) {
            mkdir($this->cacheDir, 0755, true);
        }
    }
    
    /**
     * 生成二维码图片
     * @param string $content 二维码内容（URL）
     * @param string $filename 文件名（不含扩展名）
     * @param int $size 尺寸（像素）
     * @return string 二维码图片 URL
     */
    public function generate($content, $filename, $size = 200) {
        $filepath = $this->cacheDir . $filename . '.png';
        
        // 如果已存在且未过期（24小时），直接返回
        if (file_exists($filepath) && (time() - filemtime($filepath)) < 86400) {
            return '/qr/' . $filename . '.png';
        }
        
        // 使用 phpqrcode 生成
        require_once __DIR__ . '/../../vendor/phpqrcode/qrlib.php';
        QRcode::png($content, $filepath, QR_ECLEVEL_L, $size / 25);
        
        return '/qr/' . $filename . '.png';
    }
    
    /**
     * 生成工艺卡片二维码 URL
     */
    public function generateRouteQr($routeId, $version) {
        $url = config('app.url') . '/mobile/route/' . $routeId . '?v=' . $version;
        return $this->generate($url, 'route_' . $routeId . '_' . $version, 150);
    }
    
    /**
     * 生成工序卡片二维码 URL
     */
    public function generateOperationQr($routeId, $operationNo) {
        $url = config('app.url') . '/mobile/operation/' . $routeId . '/' . $operationNo;
        return $this->generate($url, 'op_' . $routeId . '_' . $operationNo, 120);
    }
    
    /**
     * 生成标签二维码 URL
     */
    public function generateLabelQr($productCode, $batchNo) {
```

## 六、安全设计

### 6.1 认证安全

- **JWT Token**：有效期 2 小时，支持刷新
- **Cookie 安全**：HttpOnly + Secure + SameSite=Strict
- **密码策略**：本地登录用户密码最小 8 位，含字母+数字

### 6.2 接口安全

- **CORS 限制**：仅允许公司域名
- **请求限流**：单 IP 60 秒内最多 100 次请求
- **SQL 注入防护**：全部使用 PDO 预处理语句
- **XSS 防护**：输出时 HTML 实体编码

### 6.3 数据安全

- **操作审计**：所有数据变更记录操作日志（操作人、时间、前后数据）
- **软删除**：关键业务表使用 `is_deleted` 标记，不物理删除
- **数据备份**：每日自动备份，保留 30 天
