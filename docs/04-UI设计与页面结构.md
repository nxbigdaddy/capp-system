# CAPP 系统 - UI 设计与页面结构

> 版本：V1.0 | 日期：2026-06-07 | 状态：初稿

## 一、设计原则

### 1.1 整体风格

- **布局**：左侧固定导航栏（200px）+ 右侧内容区（自适应）
- **配色**：主色 #1890ff（蓝色），辅色 #52c41a（绿色），警告 #faad14，错误 #f5222d
- **字体**：系统默认字体栈，中文优先 `PingFang SC, Microsoft YaHei`
- **间距**：基础单位 8px，组件间距 16px/24px
- **响应式**：PC 端最小宽度 1280px，移动端 H5 独立适配

### 1.2 交互规范

- 列表页：表格展示，支持分页（每页 20 条）、排序、筛选
- 详情页：卡片式布局，分区块展示
- 表单页：标签页分组，必填项红色星号标记
- 操作反馈：成功绿色提示，错误红色提示，加载中骨架屏
- 确认操作：删除/提交等关键操作需二次确认弹窗

```php
// Auth/LoginController.php
class LoginController extends Controller {
    
    // GET /login - 显示登录页
    public function index() {
        if ($this->currentUser) {
            header('Location: /dashboard');
            exit;
        }
        
        $weworkQrUrl = (new WeWorkService())->getQrConnectUrl();
        $this->view('auth/login', ['weworkQrUrl' => $weworkQrUrl]);
    }
    
    // POST /api/auth/login - 账号密码登录
    public function login() {
        $data = $this->getJsonBody();
        $username = $data['username'] ?? '';
        $password = YOUR_PASSWORD'password'] ?? '';
        
        $user = (new UserService())->findByUsername($username);
        if (!$user || !password_verify($password, $user['password_hash'])) {
            Response::error(40101, '账号或密码错误');
        }
        
        $token = YOUR_SECRET'user_id' => $user['id'], 'user_name' => $user['name']]);
        Response::success(['token' => $token, 'user' => $user]);
    }
    
    // GET /auth/wework/callback - 企业微信回调
    public function weworkCallback() {
        $code = $_GET['code'] ?? '';
        $userInfo = (new WeWorkService())->getUserInfo($code);
        
        $user = (new UserService())->findOrCreateByWework($userInfo);
        $token = YOUR_SECRET'user_id' => $user['id'], 'user_name' => $user['name']]);
        
        setcookie('capp_token', $token, time() + 7200, '/', '', true, true);
        header('Location: /dashboard');
        exit;
    }
}
```

```javascript
function request(url, options = {}) {

        headers: {
            'Content-Type': 'application/json',
            'Authorization': 'Bearer ' + getToken()

    return fetch(url, { ...defaultOptions, ...options })
        .then(res => res.json())
        .then(data => {
            if (data.code !== 0) {
                throw new Error(data.message);
            }
            return data.data;
        });
}

    let html = '<table class="data-table"><thead><tr>';
```

```javascript
document.addEventListener('DOMContentLoaded', function() {
    // 检查登录状态
    if (!getToken() && !location.pathname.startsWith('/login')) {
        location.href = '/login';
        return;
    }
    
    // 初始化侧边栏菜单
    initSidebar();
    
    // 初始化面包屑
    initBreadcrumb();
});

function initSidebar() {
    // 高亮当前菜单
    const currentPath = location.pathname;
    document.querySelectorAll('.sidebar-menu a').forEach(link => {
        if (link.getAttribute('href') === currentPath) {
            link.parentElement.classList.add('active');
        }
    });
}

function initBreadcrumb() {
    // 根据当前路径生成面包屑
    const paths = location.pathname.split('/').filter(p => p);
    const breadcrumb = document.getElementById('breadcrumb');
    if (breadcrumb) {
        let html = '<a href="/dashboard">首页</a>';
        paths.forEach((path, index) => {
            html += ' / ' + path;
        });
        breadcrumb.innerHTML = html;
    }
}
```
