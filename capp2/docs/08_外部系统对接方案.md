### 1.4 对接原则

#### 1.4.1 数据一致性原则

- **最终一致性**：跨系统数据同步采用最终一致性模型，允许短暂的数据不一致窗口（通常 < 5分钟）
- **对账机制**：每日凌晨执行跨系统数据对账，生成差异报告
- **冲突解决**：以业务系统为准，CAPP系统作为工艺数据主数据源，PLM作为设计数据主数据源

#### 1.4.2 幂等性原则

- 所有同步接口必须实现幂等性，同一数据多次同步不产生副作用
- 幂等键设计：`系统标识 + 业务类型 + 业务主键 + 版本号/时间戳`
- 消息队列消费者必须实现幂等消费，防止重复处理

#### 1.4.3 异常处理原则

| 异常级别 | 处理策略 | 通知方式 | 重试机制 |
|:---|:---|:---|:---|
| 瞬时异常（网络超时） | 自动重试 | 无需通知 | 指数退避，最多5次 |
| 业务异常（数据校验失败） | 记录失败，人工介入 | 企业微信通知 | 不重试，待修复后手动触发 |
| 系统异常（服务不可用） | 熔断降级 | 短信+企业微信 | 定时探测恢复 |
| 数据冲突 | 记录冲突详情 | 企业微信通知 | 人工裁决后重试 |

#### 1.4.4 日志追踪原则

- 全链路追踪：每个跨系统请求分配唯一 `TraceID`，贯穿所有调用链路
- 日志内容：请求时间、响应时间、请求参数、响应结果、异常信息、耗时
- 日志保留：接口调用日志保留90天，错误日志保留180天
- 日志格式：`[TraceID] [时间] [系统] [接口] [状态] [耗时ms] [摘要]`

## 2. CAXA PLM对接方案

### 2.1 对接目标

| 目标编号 | 对接目标 | 业务价值 | 优先级 |
|:---|:---|:---|:---:|
| PLM-01 | 图纸同步 | 自动获取最新设计图纸，避免人工传递 | 高 |
| PLM-02 | EBOM同步 | 同步设计BOM至CAPP，作为工艺BOM基础 | 高 |
| PLM-03 | 设计变更通知 | 实时接收设计变更，触发工艺变更流程 | 高 |
| PLM-04 | 工艺状态回写 | 将工艺编制状态回写至PLM | 中 |

### 2.2 CAXA PLM接口调研

CAXA PLM提供以下对接方式：

| 对接方式 | 技术说明 | 适用场景 | 推荐度 |
|:---|:---|:---|:---:|
| **REST API** | CAXA PLM 2023+ 版本提供标准REST接口 | 常规数据查询、同步 | ★★★★★ |
| **WebService** | SOAP协议接口，支持XML数据交换 | 复杂业务操作 | ★★★★ |
| **数据库直连** | 直接访问PLM底层SQL Server/Oracle数据库 | 大批量数据同步 | ★★★ |
| **中间文件** | XML/Excel文件交换 | 异构系统、老旧版本 | ★★ |

**推荐方案**：以REST API为主，数据库直连为辅（用于初始全量同步）。

### 2.5 PHP实现方案

#### 2.5.1 API调用封装类

```php
<?php
/**
 * CAXA PLM API客户端封装类
 * 文件位置：/services/integration/PlmClient.php
 */
class PlmClient
{
    private string $baseUrl;
    private string $apiKey;
    private string $apiSecret;
    private int $timeout;
    private string $traceId;

    public function __construct(array $config)
    {
        $this->baseUrl   = rtrim($config['base_url'], '/');
        $this->apiKey    = YOUR_SECRET'api_key'];
        $this->apiSecret = $config['api_secret'];
        $this->timeout   = $config['timeout'] ?? 30;
        $this->traceId   = $this->generateTraceId();
    }

    /**
     * 获取变更的BOM列表
     */
    public function getChangedBoms(string $since): array
    {
        return $this->request('GET', '/api/bom/changes', [
            'since' => $since,
            'page'  => 1,
            'size'  => 100
        ]);
    }

    /**
     * 获取BOM明细
     */
    public function getBomDetail(string $bomId): array
    {
        return $this->request('GET', "/api/bom/{$bomId}/detail");
    }

    /**
     * 获取图纸文件信息
     */
    public function getDrawingInfo(string $partId): array
    {
        return $this->request('GET', "/api/part/{$partId}/drawings");
    }

    /**
     * 下载图纸文件
     */
    public function downloadDrawing(string $drawingId): string
    {
        $response = $this->requestRaw('GET', "/api/drawing/{$drawingId}/download");
        $filePath = $this->saveTempFile($response);
        return $filePath;
    }

    /**
     * 通用HTTP请求方法
     */
    private function request(string $method, string $uri, array $params = []): array
    {
        $url = $this->baseUrl . $uri;
        $headers = $this->buildHeaders();

        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL            => $url . ($method === 'GET' ? '?' . http_build_query($params) : ''),
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT        => $this->timeout,
            CURLOPT_HTTPHEADER     => $headers,
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
        ]);

        if ($method === 'POST') {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($params));
        }

        $startTime = microtime(true);
        $response  = curl_exec($ch);
        $httpCode  = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $totalTime = round((microtime(true) - $startTime) * 1000, 2);
        curl_close($ch);

        // 记录接口调用日志
        $this->logApiCall($method, $uri, $params, $httpCode, $totalTime);

        if ($httpCode !== 200) {
            throw new PlmApiException("PLM API调用失败: HTTP {$httpCode}", $httpCode);
        }

        $data = json_decode($response, true);
        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new PlmApiException('PLM API返回数据解析失败');
        }

        return $data;
    }

    /**
     * 构建请求头
     */
    private function buildHeaders(): array
    {
        $timestamp = time();
        $nonce     = bin2hex(random_bytes(8));
        $sign      = $this->signRequest($timestamp, $nonce);

        return [
            'Content-Type: application/json',
            'X-Api-Key: ' . $this->apiKey,
            'X-Timestamp: ' . $timestamp,
            'X-Nonce: ' . $nonce,
            'X-Signature: ' . $sign,
            'X-Trace-Id: ' . $this->traceId,
        ];
    }

    /**
     * 请求签名
     */
    private function signRequest(int $timestamp, string $nonce): string
    {
        $stringToSign = $this->apiKey . $timestamp . $nonce;
        return hash_hmac('sha256', $stringToSign, $this->apiSecret);
    }

    /**
     * 生成TraceID
     */
    private function generateTraceId(): string
    {
        return 'plm-' . date('YmdHis') . '-' . uniqid();
    }

    /**
     * 记录API调用日志
     */
    private function logApiCall(string $method, string $uri, array $params, int $httpCode, float $totalTime): void
    {
        $logData = [
            'trace_id'   => $this->traceId,
            'system'     => 'PLM',
            'method'     => $method,
            'uri'        => $uri,
            'params'     => json_encode($params, JSON_UNESCAPED_UNICODE),
            'http_code'  => $httpCode,
            'total_time' => $totalTime,
            'created_at' => date('Y-m-d H:i:s'),
        ];
        // 写入日志表或文件
        IntegrationLog::write($logData);
    }
}
```

```php
<?php
/**
 * PLM增量同步服务
 * 文件位置：/services/integration/PlmSyncService.php
 */
class PlmSyncService
{
    private PlmClient $client;
    private SyncStateManager $stateManager;

    /**
     * 执行增量同步
     */
    public function syncIncremental(string $syncType): SyncResult
    {
        $lastSyncTime = $this->stateManager->getLastSyncTime('plm', $syncType);
        $currentTime  = date('Y-m-d H:i:s');

        try {
            switch ($syncType) {
                case 'bom':
                    $result = $this->syncBomChanges($lastSyncTime);
                    break;
                case 'drawing':
                    $result = $this->syncDrawingChanges($lastSyncTime);
                    break;
                case 'design_change':
                    $result = $this->syncDesignChanges($lastSyncTime);
                    break;
                default:
                    throw new InvalidArgumentException("未知同步类型: {$syncType}");
            }

            // 更新同步状态
            if ($result->isSuccess()) {
                $this->stateManager->updateLastSyncTime('plm', $syncType, $currentTime);
            }

            return $result;

        } catch (Exception $e) {
            $this->stateManager->recordSyncError('plm', $syncType, $e->getMessage());
            throw $e;
        }
    }

    /**
     * 同步BOM变更
     */
    private function syncBomChanges(string $since): SyncResult
    {
        $result = new SyncResult();

        // 1. 获取变更列表
        $changes = $this->client->getChangedBoms($since);

        foreach ($changes['items'] as $bom) {
            try {
                // 2. 获取BOM明细
                $detail = $this->client->getBomDetail($bom['BOM_ID']);

                // 3. 数据映射转换
                $mappedData = $this->mapBomData($detail);

                // 4. 幂等性检查
                if ($this->isAlreadySynced($mappedData['plm_bom_id'], $mappedData['plm_bom_version'])) {
                    $result->addSkipped($bom['BOM_ID'], '已同步过该版本');
                    continue;
                }

                // 5. 数据校验
                $validation = $this->validateBomData($mappedData);
                if (!$validation->isValid()) {
                    $result->addFailed($bom['BOM_ID'], '数据校验失败: ' . $validation->getErrors());
                    continue;
                }

                // 6. 写入数据库（事务）
                $this->saveBomData($mappedData);

                $result->addSuccess($bom['BOM_ID']);

            } catch (Exception $e) {
                $result->addFailed($bom['BOM_ID'], $e->getMessage());
            }
        }

        return $result;
    }

    /**
     * 检查是否已同步（幂等性）
     */
    private function isAlreadySynced(string $bomId, string $version): bool
    {
        $sql = "SELECT 1 FROM plm_bom_sync_log 
                WHERE plm_bom_id = ? AND plm_bom_version = ? AND sync_status = 'SUCCESS'";
        return Db::query($sql, [$bomId, $version])->fetch() !== false;
    }
}
```

### 2.7 异常处理与重试机制

```php
<?php
/**
 * 同步异常处理器
 * 文件位置：/services/integration/SyncExceptionHandler.php
 */
class SyncExceptionHandler
{
    private RetryPolicy $retryPolicy;
    private NotificationService $notifier;

    /**
     * 处理同步异常
     */
    public function handle(string $system, string $syncType, Exception $e, array $context): void
    {
        $errorType = $this->classifyError($e);

        switch ($errorType) {
            case 'TRANSIENT':
                $this->handleTransientError($system, $syncType, $e, $context);
                break;
            case 'BUSINESS':
                $this->handleBusinessError($system, $syncType, $e, $context);
                break;
            case 'SYSTEM':
                $this->handleSystemError($system, $syncType, $e, $context);
                break;
            default:
                $this->handleUnknownError($system, $syncType, $e, $context);
        }
    }

    /**
     * 处理瞬时异常（自动重试）
     */
    private function handleTransientError(string $system, string $syncType, Exception $e, array $context): void
    {
        $retryCount = $context['retry_count'] ?? 0;

        if ($retryCount < 5) {
            // 指数退避：1s, 2s, 4s, 8s, 16s
            $delay = pow(2, $retryCount) * 1000;

            // 写入延迟队列
            DelayedQueue::push([
                'system'      => $system,
                'sync_type'   => $syncType,
                'context'     => $context,
                'retry_count' => $retryCount + 1,
                'execute_at'  => time() + $delay,
            ]);
        } else {
            // 超过重试次数，转为人工处理
            $this->notifier->sendSyncFailureAlert($system, $syncType, $e, $context);
            SyncLog::recordPermanentFailure($system, $syncType, $e, $context);
        }
    }

    /**
     * 处理业务异常（记录失败，人工介入）
     */
    private function handleBusinessError(string $system, string $syncType, Exception $e, array $context): void
    {
        SyncLog::recordFailure($system, $syncType, $e, $context);
        $this->notifier->sendBusinessErrorAlert($system, $syncType, $e->getMessage());
    }

    /**
     * 异常分类
     */
    private function classifyError(Exception $e): string
    {
        if ($e instanceof NetworkException || $e instanceof TimeoutException) {
            return 'TRANSIENT';
        }
        if ($e instanceof ValidationException || $e instanceof DataMappingException) {
            return 'BUSINESS';
        }
        if ($e instanceof ServiceUnavailableException) {
            return 'SYSTEM';
        }
        return 'UNKNOWN';
    }
}
```

```php
     */

    /**
     * 获取AccessToken
     */
    private function getAccessToken(): void

        // 检查缓存的Token是否有效
        $cacheKey = 'u8_access_token';
        $cached = Redis::get($cacheKey);

        if ($cached) {
            $data = json_decode($cached, true);
            if ($data['expire_time'] > time() + 300) {
                $this->accessToken = YOUR_SECRET'token'];
                return;
            }
        }

        // 重新获取Token
        $response = $this->requestRaw('POST', '/api/auth/token', [
            'app_key'    => $this->appKey,
            'app_secret' => $this->appSecret,
            'grant_type' => 'client_credentials'
        ]);

        $data = json_decode($response, true);
        if ($data['code'] !== 0) {
            throw new U8ApiException('获取U8 AccessToken失败: ' . $data['message']);
        }

        $this->accessToken = YOUR_SECRET'data']['access_token'];
        $expireIn = $data['data']['expires_in'] ?? 7200;

        // 缓存Token
        Redis::setex($cacheKey, $expireIn - 300, json_encode([
            'token'       => $this->accessToken,
            'expire_time' => time() + $expireIn
        ]));
    }

    /**
     * 通用请求方法
     */
    private function request(string $method, string $uri, array $params = []): array
    {
        $headers = [
            'Content-Type: application/json',
            'Authorization: Bearer ' . $this->accessToken,
        ];

        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL            => $this->baseUrl . $uri,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT        => 30,
            CURLOPT_HTTPHEADER     => $headers,
            CURLOPT_POSTFIELDS     => json_encode($params),
        ]);

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($httpCode !== 200) {
            throw new U8ApiException("U8 API请求失败: HTTP {$httpCode}");
        }

        $data = json_decode($response, true);
        if ($data['code'] !== 0) {
            throw new U8ApiException("U8业务错误: {$data['message']}");
        }

        return $data['data'] ?? [];
    }
}
```

```php
    /**
     * 执行带事务的同步操作

    public function executeWithTransaction(callable $callback, string $syncId): mixed
    {
        $db = Db::getConnection();

        try {
            $db->beginTransaction();

            // 记录同步开始
            $this->recordSyncStart($syncId);

            // 执行同步逻辑
            $result = $callback($db);

            // 记录同步成功
            $this->recordSyncSuccess($syncId);

            $db->commit();
            return $result;

        } catch (Exception $e) {
            $db->rollBack();

            // 记录同步失败
            $this->recordSyncFailure($syncId, $e);

            throw $e;
        }
    }

    /**

        $syncId = $this->generateSyncId();

            $u8Client = new U8Client(Config::get('u8'));
            $result = $u8Client->uploadRoutingTime($routingData);

            // 3. 更新本地状态为已回传
            $this->updateUploadStatus($syncId, 'SUCCESS', $result);

        } catch (Exception $e) {
            // 4. 记录失败，进入补偿队列

                'data'    => $routingData,

            ]);

        }
    }
```

```php
        // 验证签名
        if (!$this->verifyWebhookSignature($data)) {
            throw new SecurityException('MES Webhook签名验证失败');
        }

        // 处理报工数据
        $reportData = $this->mapProductionReport($data);

    /**
     * 验证Webhook签名
     */
    private function verifyWebhookSignature(array $data): bool
    {
        $signature = $_SERVER['HTTP_X_MES_SIGNATURE'] ?? '';
        $timestamp = $_SERVER['HTTP_X_MES_TIMESTAMP'] ?? '';

        // 检查时间戳（5分钟内有效）
        if (abs(time() - intval($timestamp)) > 300) {
            return false;
        }

        $payload = json_encode($data);
        $expectedSign = hash_hmac('sha256', $timestamp . $payload, $this->apiSecret);

        return hash_equals($expectedSign, $signature);
    }

            ->where('routing_id', $routingId)
            ->where('operation_seq', $operationSeq)
            ->value('run_time');

        if ($standardTime > 0) {
            $deviation = abs($actualTime - $standardTime) / $standardTime;

            if ($deviation > 0.2) {

                    'routing_id'    => $routingId,
                    'operation_seq' => $operationSeq,
                    'standard_time' => $standardTime,
                    'actual_time'   => $actualTime,
                    'deviation'     => round($deviation * 100, 2) . '%',
                ]);
            }
        }
    }
```

```php
            'Authorization: Bearer ' . $this->apiKey,

        $ch = curl_init();
        $options = [
            CURLOPT_URL            => $url,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT        => $this->timeout,
            CURLOPT_HTTPHEADER     => $headers,
            CURLOPT_SSL_VERIFYPEER => false, // 内网部署可关闭
        ];

        if ($method === 'POST') {
            $options[CURLOPT_POST] = true;
            $options[CURLOPT_POSTFIELDS] = json_encode($params, JSON_UNESCAPED_UNICODE);
        } elseif ($method === 'PUT') {
            $options[CURLOPT_CUSTOMREQUEST] = 'PUT';
            $options[CURLOPT_POSTFIELDS] = json_encode($params, JSON_UNESCAPED_UNICODE);
        } elseif ($method === 'DELETE') {
            $options[CURLOPT_CUSTOMREQUEST] = 'DELETE';
        }

        curl_setopt_array($ch, $options);

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $error    = curl_error($ch);
        curl_close($ch);

        if ($error) {
            throw new MaxKbApiException("MaxKB请求失败: {$error}");
        }

        if ($httpCode !== 200) {
            throw new MaxKbApiException("MaxKB API错误: HTTP {$httpCode}, 响应: {$response}");
        }

        $data = json_decode($response, true);
        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new MaxKbApiException('MaxKB返回数据解析失败');
        }

        // MaxKB响应格式: {code: 200, data: ..., message: ...}
        if (($data['code'] ?? 0) !== 200) {
            throw new MaxKbApiException("MaxKB业务错误: {$data['message']}");
        }

        return $data['data'] ?? [];
    }

        }
        return mb_substr($content, 0, $length) . '...';
    }
```

#### 5.5.2 文档同步服务

```php
<?php
/**
 * MaxKB文档同步服务
 * 文件位置：/services/integration/MaxKbSyncService.php
 */
class MaxKbSyncService
{
    private MaxKbClient $client;
    private DocumentConverter $converter;
    private SyncStateManager $stateManager;

    public function __construct()
    {
        $this->client        = new MaxKbClient(Config::get('maxkb'));
        $this->converter     = new DocumentConverter();
        $this->stateManager  = new SyncStateManager();
    }

    /**
     * 同步单个工艺文档
     */
    public function syncDocument(int $documentId): SyncResult
    {
        $result = new SyncResult();

        try {
            // 1. 获取文档信息
            $doc = Db::table('process_document')
                ->where('id', $documentId)
                ->where('status', 'APPROVED')
                ->first();

            if (!$doc) {
                throw new InvalidArgumentException('文档不存在或未审批');
            }

            // 2. 检查是否需要同步
            $lastSync = $this->stateManager->getDocumentSyncState($documentId);
            if ($lastSync && $lastSync['sync_time'] >= $doc['update_time']) {
                $result->addSkipped($documentId, '文档无变更');
                return $result;
            }

            // 3. 读取并转换文档
            $filePath = $this->getDocumentFilePath($doc);
            $content  = $this->converter->convertToText($filePath);

            // 4. 构建MaxKB文档数据
            $maxkbDoc = [
                'title'       => $doc['doc_name'],
                'content'     => $content,
                'meta'        => json_encode([
                    'capp_doc_id'   => $documentId,
                    'doc_type'      => $doc['doc_type'],
                    'product_code'  => $doc['product_code'],
                    'process_code'  => $doc['process_code'],
                    'version'       => $doc['version'],
                    'author'        => $doc['author'],
                    'source_system' => 'CAPP',
                ]),
                'type'        => '0', // 0=普通文档
            ];

            // 5. 同步到MaxKB
            $datasetId = Config::get('maxkb.dataset_id');

            if ($lastSync && $lastSync['maxkb_doc_id']) {
                // 更新文档
                $response = $this->client->updateDocument(
                    $datasetId,
                    $lastSync['maxkb_doc_id'],
                    $maxkbDoc
                );
            } else {
                // 新增文档
                $response = $this->client->uploadDocument($datasetId, $maxkbDoc);
            }

            // 6. 记录同步状态
            $this->stateManager->saveDocumentSyncState([
                'capp_doc_id'   => $documentId,
                'maxkb_doc_id'  => $response['id'],
                'sync_time'     => date('Y-m-d H:i:s'),
                'sync_status'   => 'PENDING', // 等待解析完成
                'version'       => $doc['version'],
            ]);

            // 7. 轮询解析状态
            $this->waitForParseComplete($datasetId, $response['id'], $documentId);

            $result->addSuccess($documentId);

        } catch (Exception $e) {
            $result->addFailed($documentId, $e->getMessage());
            $this->stateManager->saveDocumentSyncState([
                'capp_doc_id'   => $documentId,
                'sync_time'     => date('Y-m-d H:i:s'),
                'sync_status'   => 'FAILED',
                'error_msg'     => $e->getMessage(),
            ]);
        }

        return $result;
    }

    /**
     * 批量同步文档
     */
    public function batchSyncDocuments(array $filters = []): SyncResult
    {
        $result = new SyncResult();

        $query = Db::table('process_document')
            ->where('status', 'APPROVED')
            ->where('is_synced', 0);

        if (!empty($filters['doc_type'])) {
            $query->where('doc_type', $filters['doc_type']);
        }
        if (!empty($filters['product_code'])) {
            $query->where('product_code', $filters['product_code']);
        }

        $documents = $query->limit(100)->get();

        foreach ($documents as $doc) {
            $docResult = $this->syncDocument($doc['id']);
            $result->merge($docResult);
        }

        return $result;
    }

    /**
     * 等待文档解析完成
     */
    private function waitForParseComplete(string $datasetId, string $docId, int $cappDocId): void
    {
        $maxAttempts = 30;
        $attempt = 0;

        while ($attempt < $maxAttempts) {
            sleep(2);
            $status = $this->client->getDocumentStatus($datasetId, $docId);

            if ($status['status'] === 'completed') {
                $this->stateManager->updateSyncStatus($cappDocId, 'SYNCED');
                return;
            }

            if ($status['status'] === 'failed') {
                throw new MaxKbApiException('MaxKB文档解析失败: ' . ($status['error_msg'] ?? '未知错误'));
            }

            $attempt++;
        }

        throw new MaxKbApiException('MaxKB文档解析超时');
    }
}
```

```php
/**
 * 同步失败重试队列处理器
 * 文件位置：/services/integration/RetryQueueProcessor.php
 */
class RetryQueueProcessor
{
    /**
     * 处理重试队列
     */
    public function process(): void
    {
        $failedTasks = Db::table('integration_sync_log')
            ->where('status', 'FAILED')
            ->where('retry_count', '<', 5)
            ->where('next_retry_time', '<=', date('Y-m-d H:i:s'))
            ->orderBy('next_retry_time')
            ->limit(50)
            ->get();

        foreach ($failedTasks as $task) {
            try {
                $service = $this->getSyncService($task['system_code']);
                $result = $service->retrySync($task['id']);

                if ($result->isSuccess()) {
                    $this->markSuccess($task['id']);
                } else {
                    $this->scheduleRetry($task['id'], $task['retry_count'] + 1);
                }
            } catch (Exception $e) {
                $this->scheduleRetry($task['id'], $task['retry_count'] + 1);
            }
        }
    }

    /**
     * 计算下次重试时间（指数退避）
     */
    private function calculateNextRetryTime(int $retryCount): string
    {
        $delayMinutes = pow(2, $retryCount); // 2, 4, 8, 16, 32分钟
        return date('Y-m-d H:i:s', strtotime("+{$delayMinutes} minutes"));
    }
```

## 7. 接口安全

### 7.1 认证机制

#### 7.1.1 API Key认证

适用于系统间服务调用：

```php
<?php
/**
 * API Key认证中间件
 * 文件位置：/middleware/ApiKeyAuth.php
 */
class ApiKeyAuth
{
    public function handle(Request $request, callable $next): Response
    {
        $apiKey = YOUR_SECRET'X-Api-Key');
        $signature = $request->header('X-Signature');
        $timestamp = $request->header('X-Timestamp');
        $nonce = $request->header('X-Nonce');

        // 验证参数完整性
        if (empty($apiKey) || empty($signature) || empty($timestamp) || empty($nonce)) {
            return Response::json(['code' => 401, 'message' => '缺少认证参数'], 401);
        }

        // 验证时间戳（5分钟内有效）
        if (abs(time() - intval($timestamp)) > 300) {
            return Response::json(['code' => 401, 'message' => '请求已过期'], 401);
        }

        // 验证API Key
        $config = IntegrationConfig::findByApiKey($apiKey);
        if (!$config || !$config->isEnabled()) {
            return Response::json(['code' => 401, 'message' => '无效的API Key'], 401);
        }

        // 验证签名
        $expectedSign = $this->signRequest($apiKey, $config->api_secret, $timestamp, $nonce, $request->body());
        if (!hash_equals($expectedSign, $signature)) {
            return Response::json(['code' => 401, 'message' => '签名验证失败'], 401);
        }

        // 验证nonce（防重放）
        if (Redis::exists("nonce:{$nonce}")) {
            return Response::json(['code' => 401, 'message' => '重复的请求'], 401);
        }
        Redis::setex("nonce:{$nonce}", 300, '1');

        return $next($request);
    }

    private function signRequest(string $apiKey, string $secret, int $timestamp, string $nonce, string $body): string
    {
        $stringToSign = $apiKey . $timestamp . $nonce . $body;
        return hash_hmac('sha256', $stringToSign, $secret);
    }
}
```

```php
 * 敏感数据加密工具
 * 文件位置：/utils/EncryptionUtil.php

class EncryptionUtil
{
    private static string $key;

    public static function init(string $masterKey): void
    {
        self::$key = hash('sha256', $masterKey, true);
    }

    /**
     * AES-256-GCM加密
     */
    public static function encrypt(string $plaintext): string
    {
        $iv = random_bytes(12);
        $tag = '';
        $ciphertext = openssl_encrypt(
            $plaintext,
            'AES-256-GCM',
            self::$key,
            OPENSSL_RAW_DATA,
            $iv,
            $tag,
            '',
            16
        );

        return base64_encode($iv . $tag . $ciphertext);
    }

    /**
     * AES-256-GCM解密
     */
    public static function decrypt(string $encrypted): string
    {
        $data = base64_decode($encrypted);
        $iv = substr($data, 0, 12);
        $tag = substr($data, 12, 16);
        $ciphertext = substr($data, 28);

        return openssl_decrypt(
            $ciphertext,
            'AES-256-GCM',
            self::$key,
            OPENSSL_RAW_DATA,
            $iv,
            $tag
        );
    }
}
```

#### 7.3.1 令牌桶限流实现

```php
<?php
/**
 * 令牌桶限流器
 * 文件位置：/middleware/RateLimiter.php
 */
class RateLimiter
{
    /**
     * 检查是否允许请求
     */
    public function allow(string $key, int $capacity, int $rate): bool
    {
        $redisKey = "rate_limit:{$key}";
        $now = microtime(true);

        $lua = <<<LUA
            local key = KEYS[1]
            local capacity = tonumber(ARGV[1])
            local rate = tonumber(ARGV[2])
            local now = tonumber(ARGV[3])
            local requested = 1

            local bucket = redis.call('HMGET', key, 'tokens', 'last_time')
            local tokens = tonumber(bucket[1]) or capacity
            local last_time = tonumber(bucket[2]) or now

            local elapsed = now - last_time
            local new_tokens = math.min(capacity, tokens + elapsed * rate)

            if new_tokens >= requested then
                new_tokens = new_tokens - requested
                redis.call('HMSET', key, 'tokens', new_tokens, 'last_time', now)
                redis.call('EXPIRE', key, 60)
                return 1
            else
                redis.call('HMSET', key, 'tokens', new_tokens, 'last_time', now)
                redis.call('EXPIRE', key, 60)
                return 0
            end
LUA;

        $result = Redis::eval($lua, 1, $redisKey, $capacity, $rate, $now);
        return $result === 1;
    }
}
```

### 7.4 IP白名单

```php
<?php
/**
 * IP白名单中间件
 * 文件位置：/middleware/IpWhitelist.php
 */
class IpWhitelist
{
    private array $whitelist;

    public function __construct()
    {
        $this->whitelist = Config::get('integration.ip_whitelist', []);
    }

    public function handle(Request $request, callable $next): Response
    {
        $clientIp = $request->getClientIp();

        if (!$this->isAllowed($clientIp)) {
            SecurityLog::record('IP_BLOCKED', [
                'ip'       => $clientIp,
                'uri'      => $request->uri(),
                'time'     => date('Y-m-d H:i:s'),
            ]);
            return Response::json(['code' => 403, 'message' => 'IP不在白名单中'], 403);
        }

        return $next($request);
    }

    private function isAllowed(string $ip): bool
    {
        foreach ($this->whitelist as $allowed) {
            if ($this->matchIp($ip, $allowed)) {
                return true;
            }
        }
        return false;
    }

    private function matchIp(string $ip, string $pattern): bool
    {
        // 支持CIDR格式（如 203.0.113.10/24）
        if (strpos($pattern, '/') !== false) {
            return $this->matchCidr($ip, $pattern);
        }
        return $ip === $pattern;
    }
}
```
