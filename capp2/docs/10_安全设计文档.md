## 1. 安全设计概述

### 1.1 安全设计目标

| 安全目标 | 定义 | CAPP系统具体要求 |
|:---|:---|:---|
| **机密性（Confidentiality）** | 确保信息仅被授权用户访问 | 工艺数据、图纸文件、客户信息加密存储与传输；严格的访问控制 |
| **完整性（Integrity）** | 确保信息在传输和存储过程中不被篡改 | 数据校验、数字签名、操作审计；防止非法修改工艺参数 |
| **可用性（Availability）** | 确保授权用户能够及时访问系统和数据 | 高可用架构、DDoS防护、灾难恢复；保障生产连续性 |
| **可追溯性（Accountability）** | 确保所有操作可被追踪和审计 | 完整的操作日志、审计追踪；满足质量体系和合规要求 |

### 1.2 安全设计原则

| 原则 | 说明 | 实施要点 |
|:---|:---|:---|
| **最小权限原则** | 用户和进程仅拥有完成工作所需的最小权限 | RBAC权限模型；API接口最小化授权；数据库账号权限分离 |
| **纵深防御原则** | 多层安全控制，单点失效不导致整体失守 | 网络隔离 + 应用安全 + 数据安全 + 审计监控多层防护 |
| **安全默认原则** | 系统默认配置应为最安全状态 | 默认拒绝访问；强密码策略；安全头部默认启用 |
| ** fail-safe原则** | 故障时系统应进入安全状态 | 认证失败默认拒绝；异常时关闭敏感接口 |
| **职责分离原则** | 关键操作需要多人协作完成 | 工艺审批流程；敏感操作二次确认 |

**JWT Token设计**：

```php
<?php
/**
 * JWT Token管理类
 * 文件位置：/services/auth/JwtManager.php
 */
class JwtManager
{
    private string $secretKey;
    private int $accessTokenTtl;   // 访问Token有效期：2小时
    private int $refreshTokenTtl;  // 刷新Token有效期：7天

    public function __construct()
    {
        $this->secretKey = YOUR_SECRET'jwt.secret');
        $this->accessTokenTtl = 7200;   // 2小时
        $this->refreshTokenTtl = 604800; // 7天
    }

    /**
     * 生成访问Token
     */
    public function generateAccessToken(array $user): string
    {
        $header = json_encode(['typ' => 'JWT', 'alg' => 'HS256']);
        $time = time();
        $payload = json_encode([
            'iss' => 'capp-system',           // 签发者
            'aud' => 'capp-users',            // 受众
            'iat' => $time,                   // 签发时间
            'nbf' => $time,                   // 生效时间
            'exp' => $time + $this->accessTokenTtl,  // 过期时间
            'sub' => $user['id'],             // 用户ID
            'jti' => uniqid('jwt_', true),    // Token唯一标识
            'data' => [
                'user_id'    => $user['id'],
                'username'   => $user['username'],
                'real_name'  => $user['real_name'],
                'dept_id'    => $user['dept_id'],
                'roles'      => $user['roles'],
                'permissions'=> $user['permissions'],
            ]
        ]);

        $base64Header = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($header));
        $base64Payload = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($payload));

        $signature = hash_hmac('sha256', $base64Header . "." . $base64Payload, $this->secretKey, true);
        $base64Signature = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($signature));

        return $base64Header . "." . $base64Payload . "." . $base64Signature;
    }

    /**
     * 验证Token
     */
    public function verifyToken(string $token): array
    {
        $parts = explode('.', $token);
        if (count($parts) !== 3) {
            throw new JwtException('Invalid token format');
        }

        // 验证签名
        $signature = hash_hmac('sha256', $parts[0] . "." . $parts[1], $this->secretKey, true);
        $base64Signature = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($signature));

        if (!hash_equals($base64Signature, $parts[2])) {
            throw new JwtException('Invalid token signature');
        }

        $payload = json_decode(base64_decode(str_replace(['-', '_'], ['+', '/'], $parts[1])), true);

        // 验证过期时间
        if ($payload['exp'] < time()) {
            throw new JwtException('Token expired');
        }

        // 验证Token是否被吊销
        if ($this->isTokenRevoked($payload['jti'])) {
            throw new JwtException('Token revoked');
        }

        return $payload;
    }

    /**
     * 吊销Token（用户登出、密码修改等场景）
     */
    public function revokeToken(string $jti, int $exp): void
    {
        $ttl = $exp - time();
        if ($ttl > 0) {
            Redis::setex("jwt:revoked:{$jti}", $ttl, '1');
        }
    }

    private function isTokenRevoked(string $jti): bool
    {
        return Redis::exists("jwt:revoked:{$jti}");
    }
}
```

### 2.2 登录安全

#### 2.2.1 密码策略

| 策略项 | 要求 | 实现方式 |
|:---|:---|:---|
| 最小长度 | 8位 | 前端 + 后端双重校验 |
| 复杂度要求 | 大写、小写、数字、特殊字符至少3种 | 正则表达式校验 |
| 密码有效期 | 90天强制更换 | 定时任务检查 |
| 历史密码 | 禁止复用最近5次密码 | 历史密码哈希存储 |
| 密码存储 | bcrypt哈希，cost=12 | password_hash() |
| 传输加密 | HTTPS传输 | 强制HTTPS |

```php
<?php
/**
 * 密码策略验证
 * 文件位置：/services/auth/PasswordPolicy.php
 */
class PasswordPolicy
{
    /**
     * 验证密码复杂度
     */
    public function validate(string $password): ValidationResult
    {
        $result = new ValidationResult();

        if (strlen($password) < 8) {
            $result->addError('密码长度至少8位');
        }

        $complexity = 0;
        if (preg_match('/[A-Z]/', $password)) $complexity++;
        if (preg_match('/[a-z]/', $password)) $complexity++;
        if (preg_match('/[0-9]/', $password)) $complexity++;
        if (preg_match('/[^A-Za-z0-9]/', $password)) $complexity++;

        if ($complexity < 3) {
            $result->addError('密码必须包含大写字母、小写字母、数字、特殊字符中的至少3种');
        }

        // 检查常见弱密码
        if ($this->isCommonPassword($password)) {
            $result->addError('密码过于简单，请使用更复杂的密码');
        }

        return $result;
    }

    /**
     * 检查是否为历史密码
     */
    public function isHistoryPassword(int $userId, string $password): bool
    {
        $history = Db::table('user_password_history')
            ->where('user_id', $userId)
            ->orderBy('created_at', 'DESC')
            ->limit(5)
            ->column('password_hash');

        foreach ($history as $hash) {
            if (password_verify($password, $hash)) {
                return true;
            }
        }
        return false;
    }

    /**
     * 密码哈希
     */
    public function hash(string $password): string
    {
        return password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
    }
}
```

#### 2.2.2 登录失败锁定

| 失败次数 | 锁定策略 | 自动解锁 |
|:---:|:---|:---:|
| 3次 | 要求输入验证码 | - |
| 5次 | 锁定账户15分钟 | 是 |
| 10次 | 锁定账户1小时 | 是 |
| 20次 | 锁定账户24小时，通知管理员 | 否 |

```php
<?php
/**
 * 登录失败锁定机制
 * 文件位置：/services/auth/LoginLockService.php
 */
class LoginLockService
{
    /**
     * 记录登录失败
     */
    public function recordFailure(string $username, string $ip): void
    {
        $key = "login:fail:{$username}";
        $count = Redis::incr($key);
        Redis::expire($key, 3600); // 1小时过期

        // IP维度统计
        $ipKey = "login:fail:ip:{$ip}";
        $ipCount = Redis::incr($ipKey);
        Redis::expire($ipKey, 3600);

        // 触发锁定
        if ($count >= 5) {
            $lockTime = $this->calculateLockTime($count);
            Redis::setex("login:lock:{$username}", $lockTime, '1');

            if ($count >= 20) {
                AlertService::send('账户暴力破解告警', [
                    'username' => $username,
                    'ip'       => $ip,
                    'count'    => $count,
                ]);
            }
        }

        // IP黑名单（单IP超过50次失败）
        if ($ipCount >= 50) {
            Redis::setex("ip:blacklist:{$ip}", 86400, '1');
            Firewall::blockIp($ip);
        }
    }

    /**
     * 检查是否被锁定
     */
    public function isLocked(string $username): bool
    {
        return Redis::exists("login:lock:{$username}");
    }

    /**
     * 计算锁定时间
     */
    private function calculateLockTime(int $failCount): int
    {
        if ($failCount >= 20) return 86400;    // 24小时
        if ($failCount >= 10) return 3600;      // 1小时
        return 900;                              // 15分钟
    }
}
```

#### 2.2.3 验证码机制

```php
<?php
/**
 * 验证码服务
 * 文件位置：/services/auth/CaptchaService.php
 */
class CaptchaService
{
    /**
     * 生成图形验证码
     */
    public function generate(): array
    {
        $code = $this->generateCode(4);
        $captchaId = uniqid('captcha_', true);

        // 存储验证码（5分钟有效）

        // 生成图片（使用GD库）
        $image = $this->createImage($code);
        ob_start();
        imagepng($image);
        $imageData = ob_get_clean();
        imagedestroy($image);

        return [
            'captcha_id'   => $captchaId,
            'captcha_image'=> 'data:image/png;base64,' . base64_encode($imageData),
        ];
    }

    /**
     * 验证验证码
     */
    public function verify(string $captchaId, string $inputCode): bool
    {
        $key = "captcha:{$captchaId}";
        $storedCode = Redis::get($key);

        if (!$storedCode) {
            return false;
        }

        // 验证后立即删除（一次性使用）
        Redis::del($key);

        return hash_equals($storedCode, strtolower($inputCode));
    }

    /**
     * 生成随机验证码
     */
    private function generateCode(int $length): string
    {
        $chars = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';
        $code = '';
        for ($i = 0; $i < $length; $i++) {
            $code .= $chars[random_int(0, strlen($chars) - 1)];
        }
        return $code;
    }
}
```

#### 2.2.4 会话管理

| 会话策略 | 配置 |
|:---|:---|
| 会话超时 | 无操作30分钟自动过期 |
| 并发登录 | 允许，但限制最大并发数3个 |
| 会话绑定 | 绑定IP + User-Agent，异常时要求重新认证 |
| 登出处理 | 服务端吊销JWT，清除客户端Token |
| 密码修改 | 强制所有会话失效，要求重新登录 |

```php
<?php
/**
 * 会话管理
 * 文件位置：/services/auth/SessionManager.php
 */
class SessionManager
{
    /**
     * 创建会话
     */
    public function createSession(int $userId, array $tokenData): void
    {
        $sessionId = uniqid('sess_', true);
        $sessionData = [
            'user_id'     => $userId,
            'token_jti'   => $tokenData['jti'],
            'ip'          => $_SERVER['REMOTE_ADDR'],
            'user_agent'  => $_SERVER['HTTP_USER_AGENT'] ?? '',
            'created_at'  => time(),
            'last_active' => time(),
        ];

        // 存储会话
        Redis::hset("user:sessions:{$userId}", $sessionId, json_encode($sessionData));

        // 限制并发会话数
        $this->limitConcurrentSessions($userId, 3);
    }

    /**
     * 验证会话合法性
     */
    public function validateSession(int $userId, string $sessionId): bool
    {
        $sessionJson = Redis::hget("user:sessions:{$userId}", $sessionId);
        if (!$sessionJson) {
            return false;
        }

        $session = json_decode($sessionJson, true);

        // 检查IP和User-Agent是否一致（可选，根据安全策略）
        if ($session['ip'] !== $_SERVER['REMOTE_ADDR']) {
            // 记录异常登录
            SecurityLog::record('SESSION_IP_MISMATCH', $session);
            // 根据策略决定是否拒绝
        }

        // 更新最后活跃时间
        $session['last_active'] = time();
        Redis::hset("user:sessions:{$userId}", $sessionId, json_encode($session));

        return true;
    }

    /**
     * 限制并发会话数
     */
    private function limitConcurrentSessions(int $userId, int $maxSessions): void
    {
        $sessions = Redis::hgetall("user:sessions:{$userId}");

        if (count($sessions) > $maxSessions) {
            // 按最后活跃时间排序，踢出最早的会话
            uasort($sessions, function ($a, $b) {
                $aData = json_decode($a, true);
                $bData = json_decode($b, true);
                return $aData['last_active'] <=> $bData['last_active'];
            });

            $toRemove = array_slice(array_keys($sessions), 0, count($sessions) - $maxSessions);
            foreach ($toRemove as $sessionId) {
                Redis::hdel("user:sessions:{$userId}", $sessionId);
            }
        }
    }

    /**
     * 销毁用户所有会话（密码修改等场景）
     */
    public function destroyAllSessions(int $userId): void
    {
        Redis::del("user:sessions:{$userId}");
    }
}
```

```php
 * 中间件管道实现
 * 文件位置：/core/MiddlewarePipeline.php

class MiddlewarePipeline
{
    private array $middlewares = [];

    public function add($middleware): self
    {
        $this->middlewares[] = $middleware;
        return $this;
    }

    public function handle(Request $request): Response
    {
        $pipeline = array_reduce(
            array_reverse($this->middlewares),
            function ($next, $middleware) {
                return function ($request) use ($next, $middleware) {
                    if (is_string($middleware)) {
                        $middleware = new $middleware();
                    }
                    return $middleware->handle($request, $next);
                };
            },
            function ($request) {
                return Router::dispatch($request);
            }
        );

        return $pipeline($request);
    }
```

```php
            16
        );

        // 格式: base64(iv + tag + ciphertext)
        return base64_encode($iv . $tag . $ciphertext);
    }

    /**
     * AES-256-GCM解密
     */
    public function decrypt(string $encrypted): string
    {
        $data = base64_decode($encrypted);
        $iv = substr($data, 0, 12);
        $tag = substr($data, 12, 16);
        $ciphertext = substr($data, 28);

        $plaintext = openssl_decrypt(
            $ciphertext,
            'AES-256-GCM',
            $this->masterKey,
            OPENSSL_RAW_DATA,
            $iv,
            $tag
        );

        if ($plaintext === false) {
            throw new SecurityException('解密失败，数据可能已损坏');
        }

        return $plaintext;
    }
}
```

```php
class SecureDb
{
    private PDO $pdo;

    public function __construct(PDO $pdo)
    {
        $this->pdo = $pdo;
        // 禁用模拟预处理
        $this->pdo->setAttribute(PDO::ATTR_EMULATE_PREPARES, false);
    }

    /**
     * 安全的查询方法
     */
    public function query(string $sql, array $params = []): PDOStatement
    {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt;
    }

    /**
     * 安全的插入方法
     */
    public function insert(string $table, array $data): int
    {
        $columns = array_keys($data);
        $placeholders = array_map(fn($col) => ":{$col}", $columns);

        $sql = sprintf(
            "INSERT INTO %s (%s) VALUES (%s)",
            $this->quoteIdentifier($table),
            implode(', ', array_map([$this, 'quoteIdentifier'], $columns)),
            implode(', ', $placeholders)
        );

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($data);

        return (int) $this->pdo->lastInsertId();
    }

    /**
     * 安全的更新方法
     */
    public function update(string $table, array $data, string $where, array $whereParams): int
    {
        $sets = [];
        foreach ($data as $column => $value) {
            $sets[] = $this->quoteIdentifier($column) . " = :set_{$column}";
        }

        $sql = sprintf(
            "UPDATE %s SET %s WHERE %s",
            $this->quoteIdentifier($table),
            implode(', ', $sets),
            $where
        );

        $params = [];
        foreach ($data as $column => $value) {
            $params[":set_{$column}"] = $value;
        }
        $params = array_merge($params, $whereParams);

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);

        return $stmt->rowCount();
    }

    /**
     * 标识符转义
     */
    private function quoteIdentifier(string $identifier): string
    {
        return '`' . str_replace('`', '``', $identifier) . '`';
    }
```

```php
<?php
/**
 * 安全文件上传
 * 文件位置：/services/security/SecureUpload.php
 */
class SecureUpload
{
    private array $allowedTypes = [
        'application/pdf'                               => 'pdf',
        'application/msword'                            => 'doc',
        'application/vnd.openxmlformats-officedocument.wordprocessingml.document' => 'docx',
        'application/vnd.ms-excel'                      => 'xls',
        'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' => 'xlsx',
        'image/jpeg'                                    => 'jpg',
        'image/png'                                     => 'png',
        'image/gif'                                     => 'gif',
        'image/svg+xml'                                 => 'svg',
        'text/plain'                                    => 'txt',
        'application/dwg'                               => 'dwg',
        'application/octet-stream'                      => 'dwg', // CAXA图纸
    ];

    private int $maxSize = 50 * 1024 * 1024; // 50MB
    private string $uploadPath;

    public function __construct()
    {
        $this->uploadPath = '/var/www/capp/uploads/';
    }

    /**
     * 安全上传文件
     */
    public function upload(array $file): array
    {
        // 1. 检查上传错误
        if ($file['error'] !== UPLOAD_ERR_OK) {
            throw new UploadException('文件上传失败: ' . $this->getUploadError($file['error']));
        }

        // 2. 检查文件大小
        if ($file['size'] > $this->maxSize) {
            throw new UploadException('文件大小超过限制（最大50MB）');
        }

        // 3. 检查MIME类型（使用finfo，而非客户端提供的type）
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $realMime = $finfo->file($file['tmp_name']);

        if (!isset($this->allowedTypes[$realMime])) {
            throw new UploadException('不支持的文件类型: ' . $realMime);
        }

        $extension = $this->allowedTypes[$realMime];

        // 4. 检查文件扩展名
        $originalExt = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        $dangerousExts = ['php', 'php3', 'php4', 'php5', 'phtml', 'pl', 'py', 'jsp', 'asp', 'aspx', 'sh', 'cgi', 'exe', 'bat', 'cmd'];

        if (in_array($originalExt, $dangerousExts)) {
            throw new UploadException('禁止上传可执行文件');
        }

        // 5. 生成安全文件名（不使用原始文件名）
        $safeName = $this->generateSafeName($extension);
        $targetPath = $this->uploadPath . date('Y/m/') . $safeName;

        // 6. 确保目录存在且安全
        $dir = dirname($targetPath);
        if (!is_dir($dir)) {
            mkdir($dir, 0755, true);
        }

        // 7. 移动文件
        if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
            throw new UploadException('文件保存失败');
        }

        // 8. 设置文件权限
        chmod($targetPath, 0644);

        // 9. 病毒扫描（如有杀毒软件）
        $this->virusScan($targetPath);

        // 10. 存储到MinIO（生产环境）
        $minioUrl = $this->storeToMinIO($targetPath, $safeName);

        // 11. 删除本地临时文件
        unlink($targetPath);

        return [
            'original_name' => $file['name'],
            'safe_name'     => $safeName,
            'extension'     => $extension,
            'mime_type'     => $realMime,
            'size'          => $file['size'],
            'url'           => $minioUrl,
        ];
    }

    /**
     * 生成安全文件名
     */
    private function generateSafeName(string $extension): string
    {
        return date('YmdHis') . '_' . bin2hex(random_bytes(8)) . '.' . $extension;
    }

    /**
     * 病毒扫描
     */
    private function virusScan(string $filePath): void
    {
        if (function_exists('clamdscan')) {

            if (strpos($result, 'FOUND') !== false) {
                unlink($filePath);
                throw new UploadException('文件包含病毒，已删除');
```

```php
{
    /**
     * 生成请求签名
     */
    public static function sign(array $params, string $secret, string $timestamp, string $nonce): string
    {
        // 1. 按key排序
        ksort($params);

        // 2. 拼接参数
        $stringToSign = http_build_query($params);
        $stringToSign = $timestamp . '&' . $nonce . '&' . $stringToSign;

        // 3. HMAC-SHA256签名
        return hash_hmac('sha256', $stringToSign, $secret);
    }

    /**
     * 验证请求签名
     */
    public static function verify(array $params, string $secret, string $timestamp, string $nonce, string $signature): bool
    {
        $expected = self::sign($params, $secret, $timestamp, $nonce);
        return hash_equals($expected, $signature);
    }
}
```

```php
     * AES解密消息
     */

        $aesKey = base64_decode($encodingAesKey . '=');

        // AES-256-CBC解密
        $iv = substr($aesKey, 0, 16);
        $decrypted = openssl_decrypt(
            base64_decode($encryptedMsg),
            'AES-256-CBC',
            $aesKey,
            OPENSSL_RAW_DATA,
            $iv
        );

        if ($decrypted === false) {
            throw new SecurityException('消息解密失败');
        }

        // 去除填充
        $pad = ord(substr($decrypted, -1));
        $result = substr($decrypted, 0, -$pad);

        // 去除随机字符串和网络字节序长度
        $content = substr($result, 16);
        $lenList = unpack('N', substr($content, 0, 4));
        $xmlLen = $lenList[1];
        $xmlContent = substr($content, 4, $xmlLen);

        return $xmlContent;
    }
```

```php
<?php
/**
 * 智能体对话安全过滤
 * 文件位置：/services/integration/MaxKbChatSecurity.php

class MaxKbChatSecurity
{
    /**
     * 过滤对话内容
     */
    public function filterConversation(string $message): array
    {
        // 1. 敏感信息检测
        $sensitivePatterns = [
            '/\b\d{17}[\dXx]\b/' => '身份证号',
            '/\b1[3-9]\d{9}\b/' => '手机号',
            '/\b\d{16,19}\b/' => '银行卡号',
            '/密码\s*[:=]\s*\S+/i' => '密码',
        ];

        foreach ($sensitivePatterns as $pattern => $type) {
            if (preg_match($pattern, $message)) {
                return [
                    'allowed' => false,
                    'reason'  => "输入内容可能包含{$type}，请移除后重试",
                ];
            }
        }

        // 2. 长度限制
        if (mb_strlen($message) > 2000) {
            return [
                'allowed' => false,
                'reason'  => '输入内容超过2000字符限制',
            ];
        }

        return ['allowed' => true];
    }

    /**
     * 审计对话内容
     */
    public function auditConversation(int $userId, string $sessionId, string $question, string $answer): void
    {
        AuditLog::record('MAXKB_CHAT', [
            'user_id'    => $userId,
            'session_id' => $sessionId,
            'question'   => $this->truncateForAudit($question),
            'answer'     => $this->truncateForAudit($answer),
            'ip'         => $_SERVER['REMOTE_ADDR'],
        ]);
    }

    private function truncateForAudit(string $text, int $maxLength = 500): string
    {
        if (mb_strlen($text) <= $maxLength) {
            return $text;
        }
        return mb_substr($text, 0, $maxLength) . '...';
    }
}
```

### 11.4 知识库同步安全

```php
<?php
/**
 * 知识库同步安全校验
 * 文件位置：/services/integration/MaxKbSyncSecurity.php
 */
class MaxKbSyncSecurity
{
    /**
     * 验证同步文档安全性
     */
    public function validateDocument(array $document): array
    {
        // 1. 文件类型白名单
        $allowedTypes = ['doc', 'docx', 'pdf', 'txt', 'md'];
        $ext = pathinfo($document['title'], PATHINFO_EXTENSION);
        if (!in_array(strtolower($ext), $allowedTypes)) {
            return ['valid' => false, 'error' => '不支持的文件类型'];
        }

        // 2. 内容安全检查
        $content = $document['content'] ?? '';

        // 检查是否包含敏感信息
        if ($this->containsSensitiveData($content)) {
            return ['valid' => false, 'error' => '文档内容包含敏感信息，无法同步'];
        }

        // 3. 文件大小检查
        if (strlen($content) > 10 * 1024 * 1024) { // 10MB
            return ['valid' => false, 'error' => '文档内容超过10MB限制'];
        }

        // 4. 完整性校验
        $document['checksum'] = hash('sha256', $content);

        return ['valid' => true, 'document' => $document];
    }

    /**
     * 验证同步完整性
     */
    public function verifySyncIntegrity(string $documentId, string $expectedChecksum): bool
    {
        $document = Db::table('process_document')->where('id', $documentId)->first();
        if (!$document) {
            return false;
        }

        $actualChecksum = hash('sha256', $document['content']);
        return hash_equals($expectedChecksum, $actualChecksum);
    }
}
```
