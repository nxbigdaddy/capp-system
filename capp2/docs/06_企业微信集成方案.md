## 1. 企业微信集成概述

### 1.1 集成目标

本方案旨在实现CAPP系统与企业微信的深度集成，达成以下核心目标：

| 目标 | 说明 | 业务价值 |
|------|------|---------|
| 统一身份认证 | 通过企业微信扫码登录CAPP系统 | 减少密码管理成本，提升登录体验 |
| 消息通知 | 工艺审批、变更等关键事件推送至企业微信 | 及时触达，提升审批效率 |
| 组织架构同步 | 企业微信通讯录与CAPP系统用户体系同步 | 保持人员信息一致性，降低维护成本 |
| 移动办公支持 | 通过企业微信接收待办、查看工艺信息 | 支持移动端审批与查询 |

### 1.2 企业微信应用架构

```
+------------------+         +------------------+         +------------------+
|   企业微信客户端  | <-----> |   企业微信服务器  | <-----> |   CAPP系统服务器  |
|  (PC/移动端)     |         |  (微信官方)      |         |  (PHP原生)       |
+------------------+         +------------------+         +------------------+
        |                            |                            |
        | 1. 扫码登录                | 2. OAuth2.0授权             | 3. 用户绑定/登录
        | 4. 接收消息推送            | 5. 消息推送API              | 6. 发送通知
        | 7. 通讯录变更              | 8. 回调通知                 | 9. 同步处理
        +----------------------------+----------------------------+
```

**集成模块划分：**

| 模块 | 功能 | 技术方案 |
|------|------|---------|
| 身份认证模块 | 扫码登录、账号绑定 | OAuth2.0 + JWT |
| 用户同步模块 | 组织架构同步 | 通讯录API + 回调事件 |
| 消息推送模块 | 审批通知、任务提醒 | 应用消息API + RabbitMQ |
| 回调处理模块 | 事件接收与处理 | 回调URL + 消息加解密 |

## 3. 扫码登录方案

### 3.1 OAuth2.0授权流程

```mermaid
sequenceDiagram
    autonumber
    actor User as 用户
    participant Browser as 浏览器
    participant CAPP as CAPP系统
    participant Wechat as 企业微信服务器

    User->>Browser: 访问CAPP登录页
    Browser->>CAPP: GET /login
    CAPP->>Browser: 返回登录页（含企业微信扫码区域）
    
    User->>Browser: 点击"企业微信扫码登录"
    Browser->>CAPP: GET /api/v1/auth/wechat-qrcode
    CAPP->>CAPP: 生成state参数（防CSRF）
    CAPP->>Browser: 返回扫码登录URL
    Browser->>Wechat: 请求扫码登录页面（OAuth2.0授权）
    Wechat->>Browser: 展示登录二维码
    
    User->>Wechat: 使用企业微信APP扫码
    Wechat->>Wechat: 用户确认登录
    Wechat->>Browser: 302重定向到回调URL
    Browser->>CAPP: GET /api/v1/auth/wechat-callback?code=xxx&state=xxx
    CAPP->>CAPP: 验证state参数
    CAPP->>Wechat: GET /getuserinfo?code=xxx&access_token=xxx
    Wechat->>CAPP: 返回UserID
    
    alt 用户已绑定
        CAPP->>CAPP: 查询绑定的系统用户
        CAPP->>CAPP: 生成JWT Token
        CAPP->>Browser: 登录成功，返回Token
        Browser->>CAPP: 跳转首页
    else 用户未绑定
        CAPP->>Browser: 返回绑定页面
        User->>Browser: 输入系统账号密码完成绑定
        Browser->>CAPP: POST /api/v1/auth/wechat-bind
        CAPP->>CAPP: 验证账号密码，建立绑定关系
        CAPP->>CAPP: 生成JWT Token
        CAPP->>Browser: 绑定成功，返回Token
        Browser->>CAPP: 跳转首页
    end
```

### 3.2 PHP实现方案

#### 3.2.1 获取授权码（构造扫码登录URL）

```php
<?php
/**
 * 企业微信扫码登录 - 获取登录URL
 * File: /services/WechatAuthService.php
 */

class WechatAuthService
{
    private $corpId;
    private $agentId;
    private $redirectUri;
    
    public function __construct()
    {
        $config = $this->getWechatConfig();
        $this->corpId = $config['corp_id'];
        $this->agentId = $config['agent_id'];
        $this->redirectUri = urlencode('https://capp.company.com/api/v1/auth/wechat-callback');
    }
    
    /**
     * 生成企业微信扫码登录URL
     */
    public function generateLoginUrl(): array
    {
        // 生成防CSRF的state参数
        $state = $this->generateState();
        
        // 将state存入Redis/Session，有效期5分钟
        $this->storeState($state);
        
        // 构造授权URL（独立应用的扫码登录）
        $loginUrl = sprintf(
            'https://open.weixin.qq.com/connect/oauth2/authorize' .
            '?appid=%s&redirect_uri=%s&response_type=code&scope=snsapi_base&state=%s#wechat_redirect',
            $this->corpId,
            $this->redirectUri,
            $state
        );
        
        // 构造嵌入二维码的URL（网页端展示）
        $qrcodeUrl = sprintf(
            'https://open.work.weixin.qq.com/wwopen/sso/qrConnect' .
            '?appid=%s&agentid=%s&redirect_uri=%s&state=%s',
            $this->corpId,
            $this->agentId,
            $this->redirectUri,
            $state
        );
        
        return [
            'login_url' => $loginUrl,
            'qrcode_url' => $qrcodeUrl,
            'state' => $state
        ];
    }
    
    /**
     * 生成随机state
     */
    private function generateState(): string
    {
        return bin2hex(random_bytes(16));
    }
    
    /**
     * 存储state到Redis
     */
    private function storeState(string $state): void
    {
        $redis = RedisClient::getInstance();
        $redis->setex('wechat:state:' . $state, 300, time());
    }
    
    /**
     * 验证state
     */
    public function verifyState(string $state): bool
    {
        $redis = RedisClient::getInstance();
        return $redis->exists('wechat:state:' . $state);
    }
    
    /**
     * 清除state
     */
    public function clearState(string $state): void
    {
        $redis = RedisClient::getInstance();
        $redis->del('wechat:state:' . $state);
    }
}
```

#### 3.2.2 获取用户身份（通过code换取userid）

```php
<?php
/**
 * 通过授权码获取用户身份
 * File: /services/WechatAuthService.php
 */

class WechatAuthService
{
    /**
     * 通过code获取用户UserID
     */
    public function getUserIdByCode(string $code): array
    {
        $accessToken = YOUR_SECRET);
        
        $url = sprintf(
            'https://qyapi.weixin.qq.com/cgi-bin/user/getuserinfo?access_token=YOUR_SECRET',
            $accessToken,
            $code
        );
        
        $response = $this->httpGet($url);
        $data = json_decode($response, true);
        
        if (isset($data['errcode']) && $data['errcode'] !== 0) {
            throw new WechatApiException(
                $data['errmsg'],
                $data['errcode']
            );
        }
        
        // 返回UserID
        if (isset($data['UserId'])) {
            return [
                'user_id' => $data['UserId'],
                'device_id' => $data['DeviceId'] ?? null,
                'user_ticket' => $data['user_ticket'] ?? null
            ];
        }
        
        // 非企业成员（外部联系人）
        if (isset($data['OpenId'])) {
            throw new WechatApiException('非企业成员，无法登录', 40001);
        }
        
        throw new WechatApiException('获取用户信息失败', 50001);
    }
    
    /**
     * 获取用户详细信息（使用user_ticket）
     */
    public function getUserDetail(string $userTicket): array
    {
        $accessToken = YOUR_SECRET);
        
        $url = 'https://qyapi.weixin.qq.com/cgi-bin/user/getuserdetail?access_token=' . $accessToken;
        
        $postData = [
            'user_ticket' => $userTicket
        ];
        
        $response = $this->httpPost($url, json_encode($postData));
        $data = json_decode($response, true);
        
        if (isset($data['errcode']) && $data['errcode'] !== 0) {
            throw new WechatApiException($data['errmsg'], $data['errcode']);
        }
        
        return [
            'userid' => $data['userid'],
            'name' => $data['name'] ?? '',
            'department' => $data['department'] ?? [],
            'position' => $data['position'] ?? '',
            'mobile' => $data['mobile'] ?? '',
            'email' => $data['email'] ?? '',
            'avatar' => $data['avatar'] ?? '',
            'gender' => $data['gender'] ?? 0
        ];
    }
    
    /**
     * 获取AccessToken
     */
    public function getAccessToken(): string
    {
        // 优先从缓存获取
        $cacheKey = 'wechat:access_token';
        $redis = RedisClient::getInstance();
        $cachedToken = $redis->get($cacheKey);
        
        if ($cachedToken) {
            return $cachedToken;
        }
        
        // 从企业微信API获取
        $config = $this->getWechatConfig();
        $url = sprintf(
            'https://qyapi.weixin.qq.com/cgi-bin/gettoken?corpid=%s&corpsecret=%s',
            $config['corp_id'],
            $config['secret']
        );
        
        $response = $this->httpGet($url);
        $data = json_decode($response, true);
        
        if (isset($data['errcode']) && $data['errcode'] !== 0) {
            throw new WechatApiException($data['errmsg'], $data['errcode']);
        }
        
        $accessToken = YOUR_SECRET'access_token'];
        $expiresIn = $data['expires_in'] ?? 7200;
        
        // 缓存Token（提前5分钟过期）
        $redis->setex($cacheKey, $expiresIn - 300, $accessToken);
        
        // 更新数据库
        $this->updateTokenInDb($accessToken, time() + $expiresIn);
        
        return $accessToken;
    }
}
```

#### 3.2.3 系统账号绑定

```php
<?php
/**
 * 企业微信用户与系统用户绑定
 * File: /services/WechatBindService.php
 */

class WechatBindService
{
    /**
     * 检查企业微信用户是否已绑定
     */
    public function isBound(string $wechatUserId): bool
    {
        $sql = "SELECT id FROM users WHERE wechat_userid = ? AND status = 1 LIMIT 1";
        $stmt = Db::prepare($sql);
        $stmt->execute([$wechatUserId]);
        return (bool)$stmt->fetch();
    }
    
    /**
     * 获取绑定的系统用户
     */
    public function getBoundUser(string $wechatUserId): ?array
    {
        $sql = "SELECT id, username, real_name, department_id, status FROM users WHERE wechat_userid = ? LIMIT 1";
        $stmt = Db::prepare($sql);
        $stmt->execute([$wechatUserId]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$user) {
            return null;
        }
        
        if ($user['status'] !== 1) {
            throw new AuthException('账号已被禁用，请联系管理员', 4030);
        }
        
        return $user;
    }
    
    /**
     * 绑定企业微信用户与系统账号
     */
    public function bindUser(string $wechatUserId, string $username, string $password): array
    {
        // 验证系统账号密码
        $user = $this->verifySystemUser($username, $password);
        
        if (!$user) {
            throw new AuthException('用户名或密码错误', 4012);
        }
        
        // 检查是否已被其他企业微信账号绑定
        if ($user['wechat_userid'] && $user['wechat_userid'] !== $wechatUserId) {
            throw new AuthException('该账号已绑定其他企业微信', 4002);
```

```php
<?php
/**
 * JWT Token管理
 * File: /services/JwtService.php
 */

class JwtService
{
    private $secretKey;
    private $algorithm = 'HS256';
    
    public function __construct()
    {
        $this->secretKey = YOUR_SECRET'JWT_SECRET'];
    }
    
    /**
     * 生成JWT Token
     */
    public function generateToken(array $user, bool $remember = false): array
    {
        $issuedAt = time();
        $accessExpire = $remember ? 86400 * 7 : 28800; // 记住登录7天，否则8小时
        $refreshExpire = 86400 * 7; // Refresh Token 7天
        
        // Access Token
        $accessPayload = [
            'iss' => 'capp-system',
            'sub' => (string)$user['id'],
            'name' => $user['real_name'],
            'dept_id' => $user['department_id'] ?? 0,
            'iat' => $issuedAt,
            'exp' => $issuedAt + $accessExpire,
            'jti' => uniqid('token_', true),
            'login_type' => 'wechat'
        ];
        
        $accessToken = YOUR_SECRET);
        
        // Refresh Token
        $refreshPayload = [
            'iss' => 'capp-system',
            'sub' => (string)$user['id'],
            'iat' => $issuedAt,
            'exp' => $issuedAt + $refreshExpire,
            'jti' => uniqid('refresh_', true),
            'type' => 'refresh'
        ];
        
        $refreshToken = $this->encode($refreshPayload);
        
        // 将Token信息存入Redis（用于单点登录控制和登出）
        $redis = RedisClient::getInstance();
        $redis->setex('jwt:' . $accessPayload['jti'], $accessExpire, json_encode([
            'user_id' => $user['id'],
            'login_type' => 'wechat',
            'created_at' => $issuedAt
        ]));
        
        return [
            'access_token' => $accessToken,
            'refresh_token' => $refreshToken,
            'token_type' => 'Bearer',
            'expires_in' => $accessExpire
        ];
    }
    
    /**
     * 验证Token
     */
    public function verifyToken(string $token): array
    {
        try {
            $payload = $this->decode($token);
            
            // 检查Token是否在黑名单中
            $redis = RedisClient::getInstance();
            if ($redis->exists('jwt:blacklist:' . $payload['jti'])) {
                throw new AuthException('Token已失效', 4013);
            }
            
            // 检查Token是否在Redis中存在
            if (!$redis->exists('jwt:' . $payload['jti'])) {
                throw new AuthException('Token已过期', 4012);
            }
            
            return $payload;
        } catch (Exception $e) {
            throw new AuthException('Token验证失败: ' . $e->getMessage(), 4011);
        }
    }
    
    /**
     * 将Token加入黑名单（退出登录）
     */
    public function blacklistToken(string $token): void
    {
        $payload = $this->decode($token);
        $redis = RedisClient::getInstance();
        
        // 计算剩余有效期
        $ttl = $payload['exp'] - time();
        if ($ttl > 0) {
            $redis->setex('jwt:blacklist:' . $payload['jti'], $ttl, time());
            $redis->del('jwt:' . $payload['jti']);
        }
    }
    
    /**
     * JWT编码
     */
    private function encode(array $payload): string
    {
        $header = json_encode(['typ' => 'JWT', 'alg' => $this->algorithm]);
        $payload = json_encode($payload);
        
        $base64Header = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($header));
        $base64Payload = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($payload));
        
        $signature = hash_hmac('sha256', $base64Header . '.' . $base64Payload, $this->secretKey, true);
        $base64Signature = str_replace(['+', '/', '='], ['-', '_', ''], base64_encode($signature));
        
        return $base64Header . '.' . $base64Payload . '.' . $base64Signature;
    }
    
    /**
     * JWT解码
     */
    private function decode(string $token): array
    {
        $parts = explode('.', $token);
        if (count($parts) !== 3) {
            throw new Exception('Invalid token format');
        }
        
        $payload = json_decode(base64_decode(str_replace(['-', '_'], ['+', '/'], $parts[1])), true);
        
        if ($payload['exp'] < time()) {
            throw new Exception('Token expired');
        }
        
        return $payload;
    }
}
```

## 4. 用户同步方案

### 4.1 通讯录同步策略

系统采用**全量同步 + 增量同步**的双策略：

| 同步类型 | 触发时机 | 同步范围 | 说明 |
|---------|---------|---------|------|
| 全量同步 | 系统初始化、每日凌晨 | 全部部门和成员 | 确保数据完整性 |
| 增量同步 | 定时任务（每15分钟） | 变更的部门和成员 | 基于变更日志 |
| 实时同步 | 回调事件触发 | 单个部门或成员 | 企业微信主动推送 |

**企业微信部门字段与系统部门字段映射：**

| id | wechat_dept_id | 企业微信部门ID |
| name | name | 部门名称 |
| parentid | parent_wechat_id | 父部门ID |
| order | sort_order | 排序值 |

### 4.3 用户同步

**企业微信成员字段与系统用户字段映射：**

| 企业微信字段 | 系统字段 | 说明 |
|------------|---------|------|
| userid | wechat_userid | 企业微信用户ID |
| name | real_name | 姓名 |

```php
    /**
```

```php
    public function fullSync(): array

            $result['departments']['created'] = $deptStats['created'];
            $result['departments']['updated'] = $deptStats['updated'];
            
            // 2. 同步用户（从根部门递归获取）
            $users = $this->getDepartmentUsers(1); // 1为根部门ID
            $result['users']['total'] = count($users);
            
            // 获取详细用户信息

            $this->logSync('full', $result, $e->getMessage());
```

#### 5.4.1 消息队列（RabbitMQ）异步推送

```php
<?php
/**
 * 消息推送队列服务
 * File: /services/MessageQueueService.php
 */

class MessageQueueService
{
    private $connection;
    private $channel;
    private $queueName = 'wechat_message_queue';
    
    public function __construct()
    {
        $this->connection = new AMQPStreamConnection(
            $_ENV['RABBITMQ_HOST'],
            $_ENV['RABBITMQ_PORT'],
            $_ENV['RABBITMQ_USER'],
            $_ENV['RABBITMQ_PASS']
        );
        $this->channel = $this->connection->channel();
        $this->channel->queue_declare($this->queueName, false, true, false, false);
    }
    
    /**
     * 发送消息到队列
     */
    public function enqueue(array $message): void
    {
        $msg = new AMQPMessage(
            json_encode($message),
            ['delivery_mode' => AMQPMessage::DELIVERY_MODE_PERSISTENT]
        );
        $this->channel->basic_publish($msg, '', $this->queueName);
    }
    
    /**
     * 消费消息
     */
    public function consume(callable $callback): void
    {
        $this->channel->basic_qos(null, 1, null);
        $this->channel->basic_consume($this->queueName, '', false, false, false, false, $callback);
        
        while ($this->channel->is_consuming()) {
            $this->channel->wait();
        }
    }
    
    public function __destruct()
    {
        $this->channel->close();
        $this->connection->close();
    }
}
```

```php
     * 接收企业微信回调
     */
    public function handle(): void
    {
        $config = $this->getWechatConfig();
        
        // 获取回调参数
        $signature = $_GET['msg_signature'] ?? '';
        $timestamp = $_GET['timestamp'] ?? '';
        $nonce = $_GET['nonce'] ?? '';
        $echostr = $_GET['echostr'] ?? '';
        
        // URL验证（首次配置回调URL时）
        if ($echostr) {
            $this->verifyUrl($signature, $timestamp, $nonce, $echostr, $config);
            return;
        }
        
        // 接收并处理消息
        $postData = file_get_contents('php://input');
        
        try {
            // 1. 验证签名
            $this->verifySignature($signature, $timestamp, $nonce, $postData, $config);
            
            // 2. 解密消息
            $decryptedMsg = $this->decryptMessage($postData, $config);
            
            // 3. 解析XML
            $xml = simplexml_load_string($decryptedMsg);
            $eventType = (string)$xml->Event;
            $changeType = (string)$xml->ChangeType;
            
            // 4. 处理事件
            $this->handleEvent($changeType, $xml);
            
            // 5. 返回成功响应

        }

    /**
     * URL验证
     */
    private function verifyUrl(string $signature, string $timestamp, string $nonce, string $echostr, array $config): void
    {
        // 计算签名
        $token = YOUR_SECRET'token'];
        $tmpArr = [$token, $timestamp, $nonce, $echostr];
        sort($tmpArr, SORT_STRING);
        $tmpStr = implode($tmpArr);
        $computedSignature = sha1($tmpStr);
        
        if ($computedSignature !== $signature) {
            http_response_code(403);
            echo 'Invalid signature';
            return;
        }
        
        // 解密echostr
        $decrypted = $this->decrypt($echostr, $config);
        echo $decrypted;
    }
    
    /**
     * 验证消息签名
     */
    private function verifySignature(string $signature, string $timestamp, string $nonce, string $postData, array $config): void
    {
        // 提取加密消息体
        $xml = simplexml_load_string($postData);
        $encrypt = (string)$xml->Encrypt;
        
        // 计算签名
        $token = YOUR_SECRET'token'];
        $tmpArr = [$token, $timestamp, $nonce, $encrypt];
        sort($tmpArr, SORT_STRING);
        $tmpStr = implode($tmpArr);
        $computedSignature = sha1($tmpStr);
        
        if ($computedSignature !== $signature) {
            throw new Exception('Invalid message signature');
        }
    }
    
    /**
     * 解密消息
     */
    private function decryptMessage(string $postData, array $config): string
    {
        $xml = simplexml_load_string($postData);
        $encrypt = (string)$xml->Encrypt;
        
        return $this->decrypt($encrypt, $config);
    }
    
    /**
     * AES解密
     */
    private function decrypt(string $encrypted, array $config): string
    {
        $encodingAesKey = $config['encoding_aes_key'];
        $corpId = $config['corp_id'];
        
        // Base64解码AES Key
        $aesKey = base64_decode($encodingAesKey . '=');
        
        // Base64解码加密数据
        $ciphertext = base64_decode($encrypted);
        
        // AES-256-CBC解密
        $iv = substr($aesKey, 0, 16);
        $decrypted = openssl_decrypt($ciphertext, 'AES-256-CBC', $aesKey, OPENSSL_RAW_DATA | OPENSSL_ZERO_PADDING, $iv);
        
        // 去除PKCS7填充
        $pad = ord(substr($decrypted, -1));
        $decrypted = substr($decrypted, 0, -$pad);
        
        // 去除头部随机字符串和CorpID
        $content = substr($decrypted, 16);
        $msgLen = unpack('N', substr($content, 0, 4))[1];
        $msg = substr($content, 4, $msgLen);
        $fromCorpId = substr($content, 4 + $msgLen);
        
        if ($fromCorpId !== $corpId) {
            throw new Exception('CorpID mismatch');
        }
        
        return $msg;
    }
```

## 7. 安全设计

### 7.1 AccessToken安全管理

| 安全措施 | 说明 |
|---------|------|
| 集中缓存 | AccessToken存储在Redis中，全局共享 |
| 自动刷新 | Token过期前自动刷新，避免中断 |
| 并发控制 | 使用Redis分布式锁防止并发刷新 |
| 加密存储 | Secret密钥使用AES加密存储于数据库 |

```php
<?php
/**
 * AccessToken安全获取（带分布式锁）
 */
public function getAccessTokenWithLock(): string
{
    $redis = RedisClient::getInstance();
    $cacheKey = 'wechat:access_token';
    $lockKey = 'wechat:token_refresh_lock';
    
    // 先尝试从缓存获取
    $cachedToken = $redis->get($cacheKey);
    if ($cachedToken) {
        return $cachedToken;
    }
    
    // 获取分布式锁（防止并发刷新）
    $lockAcquired = $redis->set($lockKey, '1', ['NX', 'EX' => 10]);
    
    if (!$lockAcquired) {
        // 未获取到锁，等待后重试
        usleep(500000); // 等待500ms
        $cachedToken = $redis->get($cacheKey);
        if ($cachedToken) {
            return $cachedToken;
        }
        throw new Exception('获取AccessToken失败，请重试');
    }
    
    try {
        // 双重检查
        $cachedToken = $redis->get($cacheKey);
        if ($cachedToken) {
            return $cachedToken;
        }
        
        // 从企业微信API获取新Token
        $token = YOUR_SECRET);
        return $token;
        
    } finally {
        // 释放锁
        $redis->del($lockKey);
    }
}
```

### 8.1 企业微信API调用错误码处理

|--------|---------|---------|

| 0 | 请求成功 | 正常处理 |
| 40001 | 不合法的secret参数 | 检查配置，告警通知 |

### 8.2 重试机制设计

```php
<?php
/**
 * 带重试机制的HTTP请求
 * File: /utils/HttpClient.php
 */

class HttpClient
{
    /**
     * 发送请求（带重试）
     */
    public static function requestWithRetry(
        string $method,
        string $url,
        array $options = [],
        int $maxRetries = 3
    ): string {
        $lastException = null;
        
        for ($i = 0; $i < $maxRetries; $i++) {
            try {
                $response = self::request($method, $url, $options);
                $data = json_decode($response, true);
                
                // 检查业务错误码
                if (isset($data['errcode'])) {
                    if ($data['errcode'] === 0) {
                        return $response;
                    }
                    
                    // 可重试的错误
                    if (in_array($data['errcode'], [-1, 42001, 40014])) {
                        if ($i < $maxRetries - 1) {
                            $delay = pow(2, $i) * 1000; // 指数退避
                            usleep($delay * 1000);
                            continue;
                        }
                    }
                    
                    throw new WechatApiException($data['errmsg'], $data['errcode']);
                }
                
                return $response;
                
            } catch (WechatApiException $e) {
                $lastException = $e;
                
                // 可重试的错误
                if (in_array($e->getCode(), [-1, 42001, 40014]) && $i < $maxRetries - 1) {
                    $delay = pow(2, $i) * 1000;
                    usleep($delay * 1000);
                    continue;
                }
                
                throw $e;
            } catch (Exception $e) {
                $lastException = $e;
                
                if ($i < $maxRetries - 1) {
                    $delay = pow(2, $i) * 1000;
                    usleep($delay * 1000);
                    continue;
                }
            }
        }
        
        throw $lastException ?? new Exception('Request failed after retries');
    }
}
```
