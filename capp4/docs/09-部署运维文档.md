### 3.2 代码部署

#### 3.2.1 克隆代码

```bash
# 创建应用目录
mkdir -p /var/www/capp
cd /var/www/capp

# 克隆代码仓库

composer install --no-dev
```

#### 3.2.2 配置Nginx

```nginx
# /etc/nginx/conf.d/capp.conf
server {
    listen 80;
    server_name capp.example.com;
    root /var/www/capp/public;
    index index.php;

    # 强制HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl;
    server_name capp.example.com;
    root /var/www/capp/public;
    index index.php;

    # SSL配置
    ssl_certificate /etc/nginx/ssl/capp.crt;
    ssl_certificate_key /etc/nginx/ssl/capp.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # 安全头
    add_header X-Frame-Options DENY;
    add_header X-XSS-Protection "1; mode=block";
    add_header Content-Security-Policy "default-src 'self'";

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico)$ {
        expires 1d;
        add_header Cache-Control "public, immutable";
    }

    # PHP处理
    location ~ \.php$ {
        fastcgi_pass unix:/var/run/php80-php-fpm.sock;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
    }

    # 错误日志
    error_log /var/log/nginx/capp_error.log;
    access_log /var/log/nginx/capp_access.log;
}
```

### 4.3 备份策略

#### 4.3.1 数据库备份

```bash
# 创建备份脚本
cat > /usr/local/bin/backup_capp_db.sh << EOF
#!/bin/bash
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/backup/mysql"
mkdir -p \$BACKUP_DIR

mysqldump -u capp_user -p'your_password' capp_db > \$BACKUP_DIR/capp_db_\$DATE.sql

# 压缩备份
gzip \$BACKUP_DIR/capp_db_\$DATE.sql

# 保留最近30天备份
find \$BACKUP_DIR -name "*.sql.gz" -mtime +30 -delete

# 同步到异地存储
rsync -avz \$BACKUP_DIR user@remote-server:/backup/mysql/
EOF

# 设置定时任务
echo "0 2 * * * root /usr/local/bin/backup_capp_db.sh" >> /etc/crontab
```

#### 4.3.2 文件备份

```bash
# 创建文件备份脚本
cat > /usr/local/bin/backup_capp_files.sh << EOF
#!/bin/bash

APP_DIR="/var/www/capp"

mkdir -p \$BACKUP_DIR

# 备份配置文件和上传文件

    \$APP_DIR/config \
    \$APP_DIR/public/uploads

# 保留最近7天备份

EOF

# 设置定时任务
echo "0 3 * * * root /usr/local/bin/backup_capp_files.sh" >> /etc/crontab
```

### 6.2 部署脚本

```bash
#!/bin/bash
# deploy.sh - 生产环境部署脚本

set -e

APP_DIR="/var/www/capp"
BACKUP_DIR="/backup/deploy"

echo "=== 开始部署 ==="

# 1. 备份当前版本
echo "1. 备份当前版本..."
mkdir -p $BACKUP_DIR
tar -czf $BACKUP_DIR/capp_backup_$(date +%Y%m%d_%H%M%S).tar.gz $APP_DIR

# 2. 拉取最新代码
echo "2. 拉取最新代码..."
cd $APP_DIR
git pull origin main

# 3. 安装依赖
echo "3. 安装依赖..."
composer install --no-dev

# 4. 数据库迁移
echo "4. 数据库迁移..."
php bin/migrate.php

# 5. 清理缓存
echo "5. 清理缓存..."
rm -rf $APP_DIR/storage/cache/*

# 6. 重启服务
echo "6. 重启服务..."
systemctl reload php80-php-fpm
systemctl reload nginx

echo "=== 部署完成 ==="
```

### 6.3 回滚流程

```bash
#!/bin/bash
# rollback.sh - 回滚脚本

set -e

APP_DIR="/var/www/capp"
BACKUP_DIR="/backup/deploy"

echo "=== 开始回滚 ==="

# 1. 停止服务
echo "1. 停止服务..."
systemctl stop nginx
systemctl stop php80-php-fpm

# 2. 恢复备份
echo "2. 恢复备份..."
LATEST_BACKUP=$(ls -t $BACKUP_DIR/*.tar.gz | head -1)
tar -xzf $LATEST_BACKUP -C /

# 3. 恢复数据库
echo "3. 恢复数据库..."
mysql -u capp_user -p'your_password' capp_db < /backup/mysql/capp_db_$(date +%Y%m%d).sql

# 4. 启动服务
echo "4. 启动服务..."
systemctl start php80-php-fpm
systemctl start nginx

echo "=== 回滚完成 ==="
```
