### 2.2 业务流程

#### 2.2.1 用户登录流程
```
用户输入账号密码 -> 前端RSA加密 -> POST /api/auth/login
  -> AuthController::login():
     1. 验证码校验
     2. 查询用户信息
     3. password_verify(password, hash)
     4. 检查账号状态(未锁定)
     5. 生成JWT Token
     6. 缓存权限到Redis
     7. 获取菜单树
     8. 更新登录信息、记录日志
     9. 返回Token + 用户信息 + 菜单树
```

#### 2.2.2 权限校验流程
```
请求到达 -> AuthMiddleware验证JWT -> PermissionMiddleware检查权限
  1. 从Redis获取用户权限列表
  2. 检查当前请求需要的permission_code
  3. 超级管理员(ROLE_ADMIN)跳过检查
  4. 权限匹配->放行 / 不匹配->返回403
  -> DataScopeMiddleware数据权限过滤:
     data_scope=1: 无过滤 | =2: WHERE dept_id={用户部门}
     =3: WHERE dept_id IN(部门及子部门) | =4: WHERE creator_id={用户ID}
```

#### 2.2.3 菜单动态加载
```
GET /api/auth/menus -> 获取用户所有角色 -> 查询角色关联菜单权限
  -> 去重合并 -> 构建树形结构(parent_id->children)
  -> 过滤status=1且is_visible=1 -> 按sort_order排序 -> 返回菜单树JSON
```

```php
<?php
// app/controllers/AuthController.php
class AuthController extends BaseController
{
    public function login()
    {
        $username = $this->request->post('username');
        $password = YOUR_PASSWORD'password');
        if (!$this->validateCaptcha($this->request->post('captcha'))) {
            return $this->error('验证码错误');
        }
        $user = $this->userModel->findByUsername($username);
        if (!$user || !password_verify($password, $user['password'])) {
            return $this->error('账号或密码错误');
        }
        if ($user['status'] != 1) return $this->error('账号已停用');

        $token = YOUR_SECRET);
        $this->cacheUserPermissions($user['id']);
        $menus = $this->buildMenuTree($user['id']);
        $this->userModel->updateLoginInfo($user['id'], $this->request->getIp());

        return $this->success([
            'access_token' => $token['access'], 'refresh_token' => $token['refresh'],
            'expires_in' => 7200, 'user' => $this->sanitizeUser($user), 'menus' => $menus,
        ]);
    }

    private function buildMenuTree($userId)
    {
        $roles = $this->userModel->getUserRoles($userId);
        $menuIds = [];
        foreach ($roles as $role) {
            $roleMenus = $this->roleModel->getRoleMenus($role['id']);
            $menuIds = array_merge($menuIds, array_column($roleMenus, 'menu_id'));
        }
        return $this->buildTree($this->menuModel->getMenusByIds(array_unique($menuIds)), 'id', 'parent_id', 'children');
    }

    private function buildTree($items, $idField, $parentIdField, $childrenKey)
    {
        $tree = []; $grouped = [];
        foreach ($items as $item) $grouped[$item[$parentIdField]][] = $item;
        if (isset($grouped[0])) {
            foreach ($grouped[0] as $node) {
                $node[$childrenKey] = $this->buildTreeChildren($node[$idField], $items, $idField, $parentIdField, $childrenKey);
                $tree[] = $node;
            }
        }
        return $tree;
    }

    private function buildTreeChildren($parentId, $items, $idField, $parentIdField, $childrenKey)
    {
        $children = [];
        foreach ($items as $item) {
            if ($item[$parentIdField] == $parentId) {
                $item[$childrenKey] = $this->buildTreeChildren($item[$idField], $items, $idField, $parentIdField, $childrenKey);
                $children[] = $item;
            }
        }
        return $children;
    }
}
```

```php
<?php
// app/middleware/PermissionMiddleware.php
class PermissionMiddleware
{
    public function handle($request, $next, $permissionCode)
    {
        $user = Auth::user();
        if (!$user) return Response::json(['code' => 401, 'message' => '未登录'], 401);
        if ($user['role_code'] === 'ROLE_ADMIN') return $next($request);

        $cacheKey = 'user_perm:' . $user['id'];
        $permissions = Cache::get($cacheKey);
        if (!$permissions) {
            $permissions = $this->loadPermissions($user['id']);
            Cache::set($cacheKey, $permissions, 3600);
        }
        if (!in_array($permissionCode, $permissions)) {
            return Response::json(['code' => 403, 'message' => '权限不足'], 403);
        }
        return $next($request);
    }
}
```

### 3.4 PHP实现要点

```php
<?php
// app/services/ProductService.php
class ProductService
{
    public function createProduct($data, $user)
    {
        $this->db->beginTransaction();
        try {
            $data['product_code'] = $this->generateProductCode($data['product_type']);
            $productId = $this->db->insert('product_info', array_merge($data, [
                'creator_id' => $user['id'], 'creator_name' => $user['real_name'],
            ]));
            if (!empty($data['bom_items'])) $this->createBom($productId, $data['bom_items'], $user);
            $this->db->commit();
            return $productId;
        } catch (\Exception $e) { $this->db->rollBack(); throw $e; }
    }

    private function generateProductCode($productType)
    {
        $prefix = match ($productType) { 1 => 'GM', 2 => 'YL', 3 => 'PART', default => 'PRD' };
        $date = date('Ymd');
        $seq = $this->db->query("SELECT COUNT(*) + 1 FROM product_info WHERE product_code LIKE '{$prefix}{$date}%'")->fetchColumn();
        return sprintf('%s%s%04d', $prefix, $date, $seq);
    }

    public function expandBom($productId, $version = null)
    {
        $vc = $version ? "AND version_no = :version" : "AND is_current = 1";
        $sql = "WITH RECURSIVE bom_tree AS (
            SELECT id, product_id, parent_item_id, material_id, item_code, item_name,
                level, seq_no, quantity, unit, loss_rate, 1 AS depth,
                quantity AS net_quantity, CAST(item_code AS CHAR(1000)) AS path
            FROM product_bom WHERE product_id = :product_id AND parent_item_id IS NULL AND is_deleted = 0 {$vc}
            UNION ALL
            SELECT b.id, b.product_id, b.parent_item_id, b.material_id, b.item_code, b.item_name,
                b.level, b.seq_no, b.quantity, b.unit, b.loss_rate, t.depth + 1,
                ROUND(t.net_quantity * b.quantity * (1 + b.loss_rate / 100), 4),
                CONCAT(t.path, '->', b.item_code)
            FROM product_bom b INNER JOIN bom_tree t ON b.parent_item_id = t.id WHERE b.is_deleted = 0
        ) SELECT * FROM bom_tree ORDER BY path";
        $stmt = $this->db->prepare($sql);
        $stmt->bindValue(':product_id', $productId, PDO::PARAM_INT);
        if ($version) $stmt->bindValue(':version', $version, PDO::PARAM_STR);
        $stmt->execute();
        return $this->buildBomTree($stmt->fetchAll(PDO::FETCH_ASSOC));
    }

    private function buildBomTree($rows)
    {
        $tree = []; $map = [];
        foreach ($rows as $row) { $row['children'] = []; $map[$row['id']] = $row; }
        foreach ($rows as $row) {
            if ($row['parent_item_id'] === null) $tree[] = &$map[$row['id']];
            elseif (isset($map[$row['parent_item_id']])) $map[$row['parent_item_id']]['children'][] = &$map[$row['id']];
        }
        return $tree;
    }

    public function changeVersion($productId, $changeData, $user)
    {
        $this->db->beginTransaction();
        try {
            $current = $this->db->query("SELECT * FROM product_info WHERE id = ? AND is_deleted = 0", [$productId])->fetch(PDO::FETCH_ASSOC);
            $newVersionNo = $this->getNextVersion($current['version_no']);

            unset($newData['id']);

            $this->copyBomToVersion($productId, $newId, $current['version_no'], $newVersionNo);

        } catch (\Exception $e) { $this->db->rollBack(); throw $e; }
    }
```

```php
    public function compareVersions($versionIdA, $versionIdB)
    {
        $routeA = $this->getFullRouteByVersionId($versionIdA);
        $routeB = $this->getFullRouteByVersionId($versionIdB);

        $opsA = array_column($routeA['operations'], null, 'operation_no');
        $opsB = array_column($routeB['operations'], null, 'operation_no');
        foreach ($opsB as $opNo => $opB) {
            if (!isset($opsA[$opNo])) $diff['added'][] = $opB;
            elseif ($this->isOperationChanged($opsA[$opNo], $opB))
                $diff['modified'][] = ['old' => $opsA[$opNo], 'new' => $opB, 'changes' => $this->getOperationChanges($opsA[$opNo], $opB)];
            else $diff['unchanged'][] = $opB;
        }
        foreach ($opsA as $opNo => $opA) { if (!isset($opsB[$opNo])) $diff['deleted'][] = $opA; }
        return $diff;
    }

        foreach (['operation_name', 'operation_type', 'equipment_id', 'fixture_id', 'tool_id', 'work_time', 'operation_desc'] as $f) {

        }
        return false;
    }

        return $changes;
    }
```

```php
<?php

    public function approve($instanceId, $userId, $action, $comment = '')
    {
        $this->db->beginTransaction();
        try {
            $currentNode = $this->getCurrentNode($instanceId);
            if (!$currentNode) throw new \Exception("流程实例不存在或已完成");
            if ($currentNode['approver_id'] != $userId) throw new \Exception("无审批权限");

            $this->db->insert('approval_record', [
                'instance_id' => $instanceId, 'node_id' => $currentNode['id'],
                'approver_id' => $userId, 'action' => $action, 'comment' => $comment,
                'approve_time' => date('Y-m-d H:i:s'),
            ]);

            if ($action === 'approve') {
                $nextNode = $this->getNextNode($instanceId, $currentNode);
                if ($nextNode) {
                    $this->db->update('approval_instance', ['current_node_id' => $nextNode['id']], "id = ?", [$instanceId]);
                    $this->createApprovalNode($instanceId, $nextNode, $userId);
                    $this->sendNotification($instanceId);
                } else {
                    $this->db->update('approval_instance', ['status' => 2, 'end_time' => date('Y-m-d H:i:s')], "id = ?", [$instanceId]);
                    $this->onWorkflowComplete($instanceId);
                }
            } elseif ($action === 'reject') {
                $this->db->update('approval_instance', ['status' => 3, 'end_time' => date('Y-m-d H:i:s')], "id = ?", [$instanceId]);
                $this->onWorkflowRejected($instanceId);
            } elseif ($action === 'return') {
                $this->db->update('approval_instance', ['status' => 4, 'end_time' => date('Y-m-d H:i:s')], "id = ?", [$instanceId]);
            }
            $this->db->commit();
        } catch (\Exception $e) { $this->db->rollBack(); throw $e; }
    }

    private function resolveApprover($nodeDef, $instanceId)
    {
        $instance = $this->db->query("SELECT * FROM approval_instance WHERE id = ?", [$instanceId])->fetch(PDO::FETCH_ASSOC);
        switch ($nodeDef['approver_type']) {
            case 'user': return $nodeDef['approver_id'];
            case 'role':
                return $this->db->query("SELECT ur.user_id FROM sys_user_role ur JOIN sys_role r ON ur.role_id = r.id WHERE r.role_code = ? LIMIT 1", [$nodeDef['approver_role']])->fetchColumn();
            case 'dept_leader':
                $deptId = $this->db->query("SELECT u.dept_id FROM sys_user u WHERE u.id = ?", [$instance['initiator_id']])->fetchColumn();
                return $this->db->query("SELECT leader_id FROM sys_dept WHERE id = ?", [$deptId])->fetchColumn();
            case 'initiator': return $instance['initiator_id'];
            default: throw new \Exception("未知审批人类型");
        }
    }

    private function onWorkflowComplete($instanceId)
    {
        $instance = $this->db->query("SELECT * FROM approval_instance WHERE id = ?", [$instanceId])->fetch(PDO::FETCH_ASSOC);
        switch ($instance['biz_type']) {

                break;
            case 'process_change':
                $this->db->update('process_version', ['status' => 2, 'publish_time' => date('Y-m-d H:i:s')], "route_id = ?", [$instance['biz_id']]);
                break;
        }
        $this->sendCompleteNotification($instanceId);
    }
}
```

## 6. 工时定额模块

### 6.1 功能说明

#### 6.1.1 工时计算

#### 6.1.2 工时定额管理
定额标准维护、定额查询(按产品/工序)、定额调整、定额审批

#### 6.1.3 工时统计
产品工时汇总、工序工时分析(占比)、工时趋势分析(按时间段)、工时对比(不同版本/产品)

#### 6.1.4 工时优化建议
瓶颈识别(工时最长工序)、优化建议(基于历史数据)、合并建议

### 6.2 计算规则

#### 6.2.1 工时构成
**单件总工时 = 准备时间 + 加工时间 + 辅助时间 + 宽放时间**

- **准备时间(Setup Time)**：熟悉图纸、领取毛坯、安装夹具刀具、调整机床、首件试切检验。按工序类型固定值或经验公式
- **加工时间(Machining Time)**：切削加工时间=加工长度/(进给量×转速)、焊接时间=焊缝长度/焊接速度、热处理时间=保温+升温+冷却
- **辅助时间(Auxiliary Time)**：装夹工件、测量检验、更换刀具、清理切屑。加工时间×辅助系数(通常0.15~0.30)
- **宽放时间(Allowance Time)**：生理宽放(2%~5%)、疲劳宽放(2%~8%)、管理宽放(1%~3%)。(准备+加工+辅助)×宽放系数

#### 6.2.2 工时计算公式

| 工序类型 | 计算公式 |
|---------|---------|
| 车削 | T = L / (f × n) + 装夹时间 |
| 铣削 | T = L / (f × n × z) + 装夹时间 |
| 钻孔 | T = L / (f × n) + 定位时间 |
| 磨削 | T = L × B / (f × v) + 装夹时间 |
| 焊接 | T = 焊缝长度 / 焊接速度 + 准备时间 |
| 热处理 | T = 升温时间 + 保温时间 + 冷却时间 |

参数说明：T=加工时间(min)、L=加工长度(mm)、f=进给量(mm/r)、n=主轴转速(r/min)、z=铣刀齿数、B=磨削宽度(mm)、v=磨削速度(mm/min)

```php
<?php
// app/services/WorkTimeService.php
class WorkTimeService
{
    public function calculateOperationTime($operation)
    {
        $setupTime = $this->calculateSetupTime($operation);
        $machiningTime = $this->calculateMachiningTime($operation);
        $auxiliaryTime = $this->calculateAuxiliaryTime($operation, $machiningTime);
        $allowanceTime = $this->calculateAllowanceTime($setupTime, $machiningTime, $auxiliaryTime);
        return [
            'setup_time' => round($setupTime, 2), 'machining_time' => round($machiningTime, 2),
            'auxiliary_time' => round($auxiliaryTime, 2), 'allowance_time' => round($allowanceTime, 2),
            'total_time' => round($setupTime + $machiningTime + $auxiliaryTime + $allowanceTime, 2),
        ];
    }

    private function calculateSetupTime($operation)
    {
        $standard = $this->getSetupTimeStandard($operation['operation_type']);
        return $standard['base_time'] * $this->getComplexityFactor($operation);
    }

    private function calculateMachiningTime($operation)
    {
        $totalTime = 0;
        foreach ($operation['steps'] ?? [] as $step) {
            $params = $step['parameter_json'] ?? [];
            switch ($operation['operation_type']) {
                case 1: $totalTime += $this->calcCuttingTime($params); break;
                case 4: $totalTime += $this->calcHeatTreatmentTime($params); break;
                case 6: $totalTime += $this->calcWeldingTime($params); break;
                default: $totalTime += $step['work_time'] ?? 0;
            }
        }
        return $totalTime;
    }

    private function calcCuttingTime($params)
    {
        $L = $params['cutting_length'] ?? 0; $f = $params['feed_rate'] ?? 0;
        $n = $params['spindle_speed'] ?? 0; $passes = $params['cutting_passes'] ?? 1;
        if ($f <= 0 || $n <= 0) return 0;
        return ($L / ($f * $n)) * $passes;
    }

    private function calcHeatTreatmentTime($params)
    {
        return ($params['heat_up_time'] ?? 0) + ($params['hold_time'] ?? 0) + ($params['cooling_time'] ?? 0);
    }

    private function calcWeldingTime($params)
    {
        $len = $params['weld_length'] ?? 0; $speed = $params['weld_speed'] ?? 0;
        $passes = $params['weld_passes'] ?? 1;
        if ($speed <= 0) return 0;
        return ($len / $speed) * $passes;
    }

    private function calculateAuxiliaryTime($operation, $machiningTime)
    {
        $factors = [1 => 0.20, 2 => 0.25, 3 => 0.15, 4 => 0.10, 5 => 0.20, 6 => 0.25];
        return $machiningTime * ($factors[$operation['operation_type']] ?? 0.20);
    }

    private function calculateAllowanceTime($setupTime, $machiningTime, $auxiliaryTime)
    {
        return ($setupTime + $machiningTime + $auxiliaryTime) * 0.08;
    }

    public function getTotalWorkTime($routeId)
    {
        return $this->db->query(
            "SELECT SUM(setup_time) AS total_setup, SUM(work_time) AS total_machining,
                    SUM(wait_time) AS total_waiting, SUM(transport_time) AS total_transport,
                    SUM(setup_time + work_time + wait_time + transport_time) AS grand_total
             FROM process_operation WHERE route_id = ? AND is_deleted = 0", [$routeId]
        )->fetch(PDO::FETCH_ASSOC);
    }
}
```

### C. 安全设计要点

1. **密码加密**：bcrypt(cost=12)不可逆加密
2. **SQL注入防护**：PDO预处理语句，参数化查询
3. **XSS防护**：输出转义htmlspecialchars()
4. **CSRF防护**：Token验证
5. **文件上传安全**：类型校验+MIME校验+白名单限制
6. **敏感数据加密**：AES-256-CBC加密存储
7. **JWT Token**：access_token(2h)+refresh_token(7d)
8. **操作审计**：完整操作日志记录，保留1年
