class PlmSyncErrorHandler
{
    private $db;
    private $logger;
    private $maxRetries = 3;
    private $retryDelay = 60;

    public function __construct(\PDO $db, $logger = null)
    {
        $this->db = $db;
        $this->logger = $logger;
    }

    public function handleSyncError(string $syncType, string $sourceId, \Exception $e, int $attempt = 1): bool
    {
        $this->logError($syncType, $sourceId, $e, $attempt);

        if ($attempt >= $this->maxRetries) {
            $this->markAsFailed($syncType, $sourceId, $e->getMessage());
            return false;
        }

        if ($this->isRetryableError($e)) {
            sleep($this->retryDelay * $attempt);
            return true;
        }

        $this->markAsFailed($syncType, $sourceId, $e->getMessage());
        return false;
    }

    public function logError(string $syncType, string $sourceId, \Exception $e, int $attempt): void
    {

                ) VALUES (?, 'PLM', ?, ?, ?, ?, NOW())";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$syncType, $sourceId, $e->getMessage(), $e->getTraceAsString(), $attempt]);
    }

    public function markAsFailed(string $syncType, string $sourceId, string $errorMessage): void
    {
        $sql = "INSERT INTO sync_failures (
                    sync_type, source_system, source_id, error_message, status, retry_count, created_at, updated_at
                ) VALUES (?, 'PLM', ?, ?, 'failed', 0, NOW(), NOW())";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$syncType, $sourceId, $errorMessage]);
    }

    private function isRetryableError(\Exception $e): bool
    {
        $msgs = ['Connection timed out', 'Connection refused', 'HTTP 500', 'HTTP 502', 'HTTP 503', 'HTTP 504', 'SSL'];
        foreach ($msgs as $msg) {
            if (stripos($e->getMessage(), $msg) !== false) return true;
        }
        return false;
    }

        return $this->executeRequest('GET', $url);
    }

    private function httpPost(string $endpoint, array $data): array { return $this->executeRequest('POST', $this->baseUrl . $endpoint, $data); }
    private function httpPut(string $endpoint, array $data): array { return $this->executeRequest('PUT', $this->baseUrl . $endpoint, $data); }

    private function executeRequest(string $method, string $url, array $data = null): array
    {
        $timestamp = time();
        $signature = md5($this->appKey . $timestamp . md5($this->appSecret));
        $headers = [
            'X-App-Key: ' . $this->appKey, 'X-Timestamp: ' . $timestamp, 'X-Signature: ' . $signature,
            'Content-Type: application/json', 'Accept: application/json',
        ];

        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_CUSTOMREQUEST, $method);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, $this->timeout);
        curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        if ($data !== null) curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $error = curl_error($ch);

        if (!$result) throw new U8ApiException('Invalid JSON response');
        if (isset($result['code']) && $result['code'] !== 0) throw new U8ApiException("U8 Business Error: {$result['message']}", $result['code']);
        return $result;

        $this->connection = new AMQPStreamConnection($this->config['host'],$this->config['port'],$this->config['username'],$this->config['password'],$this->config['vhost']??'/');
        $this->channel = $this->connection->channel();

    private $errors = [];

    public function validateRouteDispatch(array $data): bool
    {
        $this->errors = [];
        foreach (['route_code','product_code','route_name','version','operations'] as $f) {
            if (!isset($data[$f]) || empty($data[$f])) $this->errors[] = "Missing: {$f}";
        }
        if (isset($data['operations']) && is_array($data['operations'])) {
            foreach ($data['operations'] as $i => $op) {
                foreach (['operation_seq','operation_code','operation_name'] as $f) { if (!isset($op[$f])) $this->errors[] = "Op[{$i}] missing: {$f}"; }
                if (isset($op['setup_time']) && !is_numeric($op['setup_time'])) $this->errors[] = "Op[{$i}] setup_time invalid";
                if (isset($op['run_time']) && !is_numeric($op['run_time'])) $this->errors[] = "Op[{$i}] run_time invalid";
            }
        }
        return empty($this->errors);
    }

    public function validateCompletionReport(array $data): bool
    {
        $this->errors = [];
        foreach (['report_id','work_order','product_code','operation_code','quantity_completed','quantity_qualified','report_time'] as $f) {
            if (!isset($data[$f]) || $data[$f] === '') $this->errors[] = "Missing: {$f}";
        }
        if (isset($data['quantity_completed']) && $data['quantity_completed'] < 0) $this->errors[] = 'quantity_completed < 0';
        if (isset($data['quantity_qualified']) && $data['quantity_qualified'] < 0) $this->errors[] = 'quantity_qualified < 0';
        if (isset($data['quantity_completed']) && isset($data['quantity_qualified']) && $data['quantity_qualified'] > $data['quantity_completed']) $this->errors[] = 'qualified > completed';
        if (isset($data['report_time']) && strtotime($data['report_time']) === false) $this->errors[] = 'Invalid report_time';
        return empty($this->errors);
    }

    public function validateQualityData(array $data): bool

        $this->errors = [];
