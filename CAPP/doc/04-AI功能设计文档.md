## 1. 概述

### 1.1 AI功能定位

CAPP系统的AI功能定位于**智能工艺设计辅助与决策支持**，通过引入大语言模型（LLM）、检索增强生成（RAG）、智能推荐算法等技术，为工业压缩机和压力容器制造企业的工艺设计人员提供全方位的智能化支持。

**核心定位：**

| 定位维度 | 说明 |
|---------|------|
| 辅助角色 | AI作为工艺设计人员的智能助手，不替代人工决策 |
| 知识中枢 | 整合企业工艺知识、标准规范、历史数据，形成统一知识服务 |
| 决策支持 | 基于数据分析和模型推理，提供工艺路线、参数、工时等决策建议 |
| 效率工具 | 减少重复性劳动，提升工艺设计效率50%以上 |

#### 2.2.1 LLM模型选择（支持多种模型接入）

```php
<?php
// app/services/LLMProvider.php

class LLMProvider {
    private $config;
    private $currentModel;
    
    const SUPPORTED_MODELS = [
        'qwen-plus'   => ['provider' => 'aliyun',  'endpoint' => 'dashscope'],
        'qwen-turbo'  => ['provider' => 'aliyun',  'endpoint' => 'dashscope'],
        'deepseek-v3' => ['provider' => 'deepseek', 'endpoint' => 'openai_compat'],
        'ernie-4.0'   => ['provider' => 'baidu',   'endpoint' => 'qianfan'],
        'gpt-4o'      => ['provider' => 'openai',  'endpoint' => 'openai'],
    ];
    
    public function __construct($model = null) {
        $this->config = require APP_PATH . 'config/ai.php';
        $this->currentModel = $model ?? $this->config['default_model'];
    }
    
    public function switchModel($model) {
        if (!isset(self::SUPPORTED_MODELS[$model])) {
            throw new \Exception("不支持的模型: {$model}");
        }
        $this->currentModel = $model;
        return $this;
    }
    
    public function chat($messages, $options = []) {
        $modelConfig = self::SUPPORTED_MODELS[$this->currentModel];
        
        switch ($modelConfig['provider']) {
            case 'aliyun':
                return $this->callAliyunDashScope($messages, $options);
            case 'deepseek':
                return $this->callDeepSeek($messages, $options);
            case 'baidu':
                return $this->callBaiduQianfan($messages, $options);
            case 'openai':
                return $this->callOpenAI($messages, $options);
            default:
                throw new \Exception("不支持的提供商: {$modelConfig['provider']}");
        }
    }
    
    public function chatStream($messages, $callback, $options = []) {
        $modelConfig = self::SUPPORTED_MODELS[$this->currentModel];
        
        switch ($modelConfig['provider']) {
            case 'aliyun':
                return $this->callAliyunDashScopeStream($messages, $callback, $options);
            case 'deepseek':
                return $this->callDeepSeekStream($messages, $callback, $options);
        }
    }
    
    private function callAliyunDashScope($messages, $options) {
        $apiKey = YOUR_SECRET'providers']['aliyun']['api_key'];
        $url = 'https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions';
        
        $payload = [
            'model'    => $this->currentModel,
            'messages' => $messages,
            'temperature' => $options['temperature'] ?? 0.7,
            'max_tokens'  => $options['max_tokens'] ?? 2000,
        ];
        
        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL            => $url,
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => json_encode($payload),
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HTTPHEADER     => [
                'Content-Type: application/json',
                'Authorization: Bearer ' . $apiKey,
            ],
            CURLOPT_TIMEOUT        => $options['timeout'] ?? 60,
        ]);
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        if ($httpCode !== 200) {
            throw new \Exception("LLM API调用失败: HTTP {$httpCode}, {$response}");
        }
        
        $result = json_decode($response, true);
        return [
            'content'   => $result['choices'][0]['message']['content'],
            'usage'     => $result['usage'],
            'model'     => $result['model'],
            'request_id' => $result['request_id'] ?? '',
        ];
    }
    
    private function callAliyunDashScopeStream($messages, $callback, $options) {
        $apiKey = YOUR_SECRET'providers']['aliyun']['api_key'];
        $url = 'https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions';
        
        $payload = [
            'model'    => $this->currentModel,
            'messages' => $messages,
            'stream'   => true,
            'temperature' => $options['temperature'] ?? 0.7,
        ];
        
        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL            => $url,
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => json_encode($payload),
            CURLOPT_RETURNTRANSFER => false,
            CURLOPT_HTTPHEADER     => [
                'Content-Type: application/json',
                'Authorization: Bearer ' . $apiKey,
            ],
            CURLOPT_TIMEOUT        => $options['timeout'] ?? 120,
            CURLOPT_WRITEFUNCTION  => function($ch, $data) use ($callback) {
                $lines = explode("\n", $data);
                foreach ($lines as $line) {
                    $line = trim($line);
                    if (empty($line) || $line === 'data: [DONE]') {
                        continue;
                    }
                    if (str_starts_with($line, 'data: ')) {
                        $json = substr($line, 6);
                        $chunk = json_decode($json, true);
                        if (isset($chunk['choices'][0]['delta']['content'])) {
                            $callback($chunk['choices'][0]['delta']['content']);
                        }
                    }
                }
                return strlen($data);
            },
        ]);
        
        curl_exec($ch);
        curl_close($ch);
    }
}
```

#### 2.2.2 RAG（检索增强生成）架构设计

**RAG流程：**

```
用户问题 → 问题预处理 → 查询改写 → 多路检索（向量+关键词） → 结果融合与重排序 → 上下文构建 → LLM生成回答 → 结果后处理
```

**RAG核心实现：**

```php
<?php
// app/services/RAGService.php

class RAGService {
    private $vectorDB;
    private $llmProvider;
    private $knowledgeDB;
    
    public function __construct() {
        $this->vectorDB = new MilvusClient();
        $this->llmProvider = new LLMProvider();
        $this->knowledgeDB = new KnowledgeModel();
    }
    
    public function query($question, $options = []) {
        // 1. 问题预处理
        $processedQuery = $this->preprocessQuery($question);
        
        // 2. 查询改写
        $queries = $this->rewriteQuery($processedQuery);
        
        // 3. 多路检索
        $vectorResults = $this->vectorSearch($queries, $options);
        $keywordResults = $this->keywordSearch($queries, $options);
        
        // 4. 结果融合与重排序（RRF算法）
        $mergedResults = $this->mergeAndRerank($vectorResults, $keywordResults);
        
        // 5. 构建上下文
        $context = $this->buildContext($mergedResults, $options['top_k'] ?? 5);
        
        // 6. 构建Prompt
        $prompt = $this->buildPrompt($question, $context);
        
        // 7. LLM生成回答
        $response = $this->llmProvider->chat([
            ['role' => 'system', 'content' => $this->getSystemPrompt()],
            ['role' => 'user', 'content' => $prompt],
        ], $options);
        
        return $this->postProcess($response, $context);
    }
    
    private function vectorSearch($queries, $options) {
        $results = [];
        foreach ($queries as $query) {
            $queryVector = $this->embedQuery($query);
            $searchResults = $this->vectorDB->search([
                'collection_name' => 'knowledge_base',
                'vector'          => $queryVector,
                'top_k'           => $options['top_k'] ?? 10,
                'filter'          => $this->buildFilter($options),
                'output_fields'   => ['id', 'content', 'source', 'category', 'metadata'],
            ]);
            $results = array_merge($results, $searchResults);
        }
        return $results;
    }
    
    private function mergeAndRerank($vectorResults, $keywordResults) {
        $scoreMap = [];
        
        foreach ($vectorResults as $rank => $item) {
            $id = $item['id'];
            $scoreMap[$id] = ($scoreMap[$id] ?? 0) + 1 / (60 + $rank + 1);
        }
        
        foreach ($keywordResults as $rank => $item) {
            $id = $item['id'];
            $scoreMap[$id] = ($scoreMap[$id] ?? 0) + 1 / (60 + $rank + 1);
        }
        
        arsort($scoreMap);
        
        $allResults = array_merge($vectorResults, $keywordResults);
        $uniqueResults = [];
        foreach ($allResults as $item) {
            if (!isset($uniqueResults[$item['id']])) {
                $uniqueResults[$item['id']] = $item;
                $uniqueResults[$item['id']]['rrf_score'] = $scoreMap[$item['id']];
            }
        }
        
        usort($uniqueResults, function($a, $b) {
            return $b['rrf_score'] <=> $a['rrf_score'];
        });
        
        return $uniqueResults;
    }
    
    private function buildPrompt($question, $context) {
        $contextText = "";
        foreach ($context as $item) {
            $contextText .= "[知识片段 {$item['index']}]\n";
            $contextText .= "来源: {$item['source']}\n";
            $contextText .= "分类: {$item['category']}\n";
            $contextText .= "内容: {$item['content']}\n\n";
        }
        
        return <<<PROMPT
请根据以下参考知识回答用户问题。如果参考知识不足以回答问题，请明确告知用户。

【参考知识】
{$contextText}

【用户问题】
{$question}

【回答要求】
1. 回答要准确、专业、简洁
2. 引用知识片段时请注明来源
3. 如果涉及工艺参数，请给出具体数值和单位
4. 如果知识之间存在矛盾，请说明并给出建议
PROMPT;
    }
}
```

#### 2.2.3 向量数据库选型（Milvus集成）

```php
<?php
// app/core/MilvusClient.php

class MilvusClient {
    private $host;
    private $port;
    private $client;
    
    public function __construct() {
        $config = require APP_PATH . 'config/milvus.php';
        $this->host = $config['host'];
        $this->port = $config['port'];
        $this->client = new \GuzzleHttp\Client([
            'base_uri' => "http://{$this->host}:{$this->port}/",
            'timeout'  => 30,
        ]);
    }
    
    public function createCollection($collectionName, $schema) {
        $payload = [
            'collection_name' => $collectionName,
            'fields'          => $schema,
        ];
        
        $response = $this->client->post('v1/vector_collections', ['json' => $payload]);
        return json_decode($response->getBody(), true);
    }
    
    public function insert($collectionName, $data) {
        $payload = [
            'collection_name' => $collectionName,
```

```php
        foreach ($phpWord->getSections() as $section) {
            foreach ($section->getElements() as $element) {
                if (method_exists($element, 'getText')) {
                    $text .= $element->getText() . "\n";
                }
            }
        }
        
        return [
            'text'     => $this->cleanText($text),
            'metadata' => [
                'title'  => $phpWord->getDocInfo()->getTitle() ?? '',
                'author' => $phpWord->getDocInfo()->getCreator() ?? '',
            ],
        ];
    }
```

#### 3.1.3 工时智能计算

基于加工特征、材料、设备、批量等因素，智能计算工序工时。

**工时计算模型：**

```
总工时 = 准备工时 + (单件加工工时 + 辅助工时) × 批量 + 等待工时 + 运输工时

单件加工工时 = Σ(各工步加工时间)
工步加工时间 = 加工长度 / (切削速度 × 进给量) × 修正系数

修正系数：材料修正系数 K_material、设备修正系数 K_equipment、精度修正系数 K_precision、批量修正系数 K_batch
```

```php
<?php
```
```php
// app/services/RuleEngine.php

class RuleEngine {
    private $rules;
    
    public function __construct() {
        $this->rules = require APP_PATH . 'config/business_rules.php';
    }
    
    public function evaluate($ruleGroup, $context) {
        $groupRules = $this->rules[$ruleGroup] ?? [];
        
        usort($groupRules, function($a, $b) {
            return $b['priority'] <=> $a['priority'];
        });
        
        $results = ['mandatory' => [], 'recommended' => [], 'violations' => []];
        
        foreach ($groupRules as $rule) {
            $conditionMet = $this->evaluateCondition($rule['condition'], $context);
            
            if ($conditionMet) {
                $action = $this->executeAction($rule['action'], $context);
                
                if (str_starts_with($rule['action'], 'require_')) {
                    $results['mandatory'][] = [
                        'rule_id'  => $rule['id'] ?? '',
                        'message'  => $rule['message'],
                        'priority' => $rule['priority'],
                        'action'   => $action,
                    ];
                } else {
                    $results['recommended'][] = [
                        'rule_id'  => $rule['id'] ?? '',
                        'message'  => $rule['message'],
                        'priority' => $rule['priority'],
                        'action'   => $action,
                    ];
                }
            }
        }
        
        return $results;
    }
    
    private function evaluateCondition($condition, $context) {
        $expression = $this->substituteVariables($condition, $context);
        return $this->safeEvaluate($expression);
    }
    
    private function executeAction($action, $context) {
        preg_match('/(\w+)\((.*)\)/', $action, $matches);
        if (empty($matches)) return null;
        
        $methodName = $matches[1];
        $params = $this->parseParams($matches[2]);
        return call_user_func_array([$this, $methodName], $params);
    }
}
```

```php
            ['key_value' => $encrypted, 'updated_at' => date('Y-m-d H:i:s')]
        );
    }
    
    public function getKey($keyName) {
        $record = $this->keyModel->where('key_name', $keyName)->first();
        if (!$record) throw new \Exception("密钥不存在: {$keyName}");
        return $this->decrypt($record['key_value']);
    }
    
    private function encrypt($data) {
        $key = env('ENCRYPTION_KEY');
        $iv = random_bytes(16);
        $encrypted = openssl_encrypt($data, 'AES-256-CBC', $key, 0, $iv);
        return base64_encode($iv . $encrypted);
    }
    
    private function decrypt($data) {
        $key = env('ENCRYPTION_KEY');
        $data = base64_decode($data);
        $iv = substr($data, 0, 16);
        $encrypted = substr($data, 16);
        return openssl_decrypt($encrypted, 'AES-256-CBC', $key, 0, $iv);
    }
}
```

```php
class AIRateLimitMiddleware {
    private $redis;
    
    const LIMITS = [
        'chat'     => ['requests' => 60, 'window' => 60],   // 每分钟60次

        $limit = self::LIMITS[$type] ?? self::LIMITS['chat'];
        
        $key = "ai:ratelimit:{$type}:{$userId}";
        $current = $this->redis->get($key);
        
        if ($current && $current >= $limit['requests']) {
            return Response::json([
                'code'    => 429,
                'message' => '请求过于频繁，请稍后重试',
                'retry_after' => $limit['window'],
            ], 429);
        }
        
        if ($current) {
            $this->redis->incr($key);
        } else {
            $this->redis->set($key, 1, $limit['window']);
        }
        
        return $next($request);
    }
}
```
