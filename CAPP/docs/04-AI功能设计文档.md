## 1. 概述

### 1.1 AI功能定位

CAPP（计算机辅助工艺设计系统）AI功能模块面向**工业压缩机**和**压力容器**研发、设计、生产企业的核心业务需求，通过引入大语言模型（LLM）、检索增强生成（RAG）、智能决策引擎等AI技术，实现工艺设计的智能化升级。

**AI功能定位**:

| 定位维度 | 说明 |
|---------|------|
| **辅助决策** | 为工艺工程师提供智能参数推荐、工艺路线建议、工时预估等决策支持 |
| **知识赋能** | 将企业沉淀的工艺知识、标准规范、历史案例转化为可检索、可问答的智能知识体系 |
| **效率提升** | 通过AI自动化处理重复性工作，缩短工艺设计周期50%以上 |
| **质量保障** | 利用AI进行工艺方案评审、异常检测，降低工艺设计错误率 |
| **持续进化** | 通过反馈学习机制，不断优化AI模型和推荐准确率 |

#### 2.2.1 LLM模型接入层

```php
// app/services/ai/ModelRouter.php
class ModelRouter {
    private array $models;
    private Redis $cache;

    public function __construct() {
        $this->models = $this->loadModelConfig();
        $this->cache = RedisManager::getInstance();
    }

    public function route(string $intent, string $query): string {
        $cacheKey = "ai:model_route:{$intent}";
        $cached = $this->cache->get($cacheKey);
        if ($cached) {
            return $cached;
        }

        $complexity = $this->estimateComplexity($query);
        
        $modelCode = match (true) {
            $complexity > 0.8 => 'qwen-max',
            $complexity > 0.5 => 'qwen-plus',
            default => 'qwen-turbo',
        };

        $this->cache->setex($cacheKey, 300, $modelCode);
        return $modelCode;
    }

    private function estimateComplexity(string $query): float {
        $length = mb_strlen($query);
        $hasTechnical = preg_match('/工艺|参数|计算|分析|评审|优化/', $query);
        $hasMultiQuestion = substr_count($query, '？') + substr_count($query, '?');
        $score = min(1.0, ($length / 500) + ($hasTechnical ? 0.3 : 0) + ($hasMultiQuestion * 0.2));
        return $score;
    }

    private function loadModelConfig(): array {
        $pdo = Database::getConnection();
        $stmt = $pdo->query("SELECT * FROM ai_model WHERE model_status = 1");
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
}
```

### 2.3 对话管理

#### 2.3.1 会话管理

```php
// app/services/ai/SessionManager.php
class SessionManager {
    private Redis $redis;
    private PDO $db;

    public function __construct() {
        $this->redis = RedisManager::getInstance();
        $this->db = Database::getConnection();
    }

    public function createSession(int $userId, string $title = '新对话'): string {
        $sessionId = 'sess_' . bin2hex(random_bytes(16));
        
        $sessionData = [
            'user_id' => $userId,
            'title' => $title,
            'created_at' => date('Y-m-d H:i:s'),
            'message_count' => 0,
            'last_active_at' => date('Y-m-d H:i:s'),
        ];
        
        $this->redis->hMSet("ai:session:{$sessionId}", $sessionData);
        $this->redis->sAdd("ai:user_sessions:{$userId}", $sessionId);
        $this->redis->expire("ai:session:{$sessionId}", 86400 * 7);
        
        $stmt = $this->db->prepare(
            "INSERT INTO ai_chat_sessions (session_id, user_id, title, created_at) 
             VALUES (?, ?, ?, NOW())"
        );
        $stmt->execute([$sessionId, $userId, $title]);
        
        return $sessionId;
    }

    public function getSessionContext(string $sessionId, int $limit = 10): array {
        $cacheKey = "ai:session_context:{$sessionId}";
        $cached = $this->redis->get($cacheKey);
        if ($cached) {
            return json_decode($cached, true);
        }
        
        $stmt = $this->db->prepare(
            "SELECT user_message, ai_message, created_at 
             FROM ai_chat 
             WHERE session_id = ? 
             ORDER BY created_at DESC 
             LIMIT ?"
        );
        $stmt->execute([$sessionId, $limit]);
        $messages = array_reverse($stmt->fetchAll(PDO::FETCH_ASSOC));
        
        $this->redis->setex($cacheKey, 300, json_encode($messages));
        return $messages;
    }

    public function addMessage(string $sessionId, int $userId, int $modelId, 
                                string $userMessage, string $aiMessage, 
                                array $context = []): void {
        $stmt = $this->db->prepare(
            "INSERT INTO ai_chat 
             (session_id, user_id, model_id, chat_type, user_message, ai_message, 
              context_data, created_at) 
             VALUES (?, ?, ?, 'general', ?, ?, ?, NOW())"
        );
        $stmt->execute([
            $sessionId, $userId, $modelId, 
            $userMessage, $aiMessage, 
            json_encode($context, JSON_UNESCAPED_UNICODE)
        ]);
        
        $this->redis->hSet("ai:session:{$sessionId}", 'last_active_at', date('Y-m-d H:i:s'));
        $this->redis->hIncrBy("ai:session:{$sessionId}", 'message_count', 1);
        $this->redis->del("ai:session_context:{$sessionId}");
    }

    public function getUserSessions(int $userId, int $limit = 20): array {
        $stmt = $this->db->prepare(
```

```php
        $paragraphs = preg_split('/\n\s*\n/', $text);
        $chunks = [];
        $current = '';
        
        foreach ($paragraphs as $para) {

                $current .= "\n" . $para;
            }
        }
        
        if (trim($current) !== '') {
            $chunks[] = trim($current);
        }
        
        return $chunks;
```

```php
        );
```

```php
class AsyncTaskProcessor {
    private Redis $redis;

    public function __construct() {
        $this->redis = RedisManager::getInstance();
    }

    public function enqueue(string $taskType, array $payload): string {
        $taskId = 'task_' . bin2hex(random_bytes(16));
        
        $task = [
            'id' => $taskId,
            'type' => $taskType,
            'payload' => $payload,
            'status' => 'pending',
            'created_at' => time(),
        ];

    }

    public function process(): void {
        while (true) {
            $taskJson = $this->redis->brPop('ai:task_queue', 5);
            if (!$taskJson) {
                continue;
            }

            $task = json_decode($taskJson[1], true);
            $this->executeTask($task);
        }
    }
```

```php
// app/services/ai/RuleEngine.php
class RuleEngine {
    private array $rules;

    public function __construct() {
        $this->rules = $this->loadRules();
    }

    public function evaluate(string $ruleSet, array $features, array $data = []): array {
        $matchedRules = [];
        $result = null;
        $maxConfidence = 0;

        foreach ($this->rules[$ruleSet] ?? [] as $rule) {
            if ($this->matchRule($rule, $features, $data)) {
                $matchedRules[] = $rule;
                if ($rule['priority'] > $maxConfidence) {
                    $maxConfidence = $rule['priority'];
                    $result = $rule['action'];
                }
            }
        }

        return [
            'matched' => !empty($matchedRules),
            'rules' => $matchedRules,
            'result' => $result,
            'confidence' => $maxConfidence,
            'reasoning' => $this->buildReasoning($matchedRules),
        ];
    }

    private function loadRules(): array {
        $pdo = Database::getConnection();
        $stmt = $pdo->query(
            "SELECT * FROM ai_rules WHERE is_enabled = 1 ORDER BY priority DESC"
        );
        
        $rules = [];
        foreach ($stmt->fetchAll(PDO::FETCH_ASSOC) as $row) {
            $rules[$row['rule_set']][] = [
                'id' => $row['id'],
                'name' => $row['rule_name'],
                'condition' => json_decode($row['condition_json'], true),
                'action' => json_decode($row['action_json'], true),
                'priority' => $row['priority'],
            ];
        }
        
        return $rules;
    }

    private function matchRule(array $rule, array $features, array $data): bool {
        $conditions = $rule['condition'];
        
        foreach ($conditions as $condition) {
            $value = $this->resolveValue($condition['field'], $features, $data);
            
            switch ($condition['operator']) {
                case 'eq':
                    if ($value != $condition['value']) return false;
                    break;
                case 'in':
                    if (!in_array($value, $condition['value'])) return false;
                    break;
                case 'gte':
                    if ($value < $condition['value']) return false;
                    break;
                case 'lte':
                    if ($value > $condition['value']) return false;
                    break;
                case 'contains':
                    if (!str_contains((string)$value, $condition['value'])) return false;
                    break;
            }
        }
        
        return true;
    }
}
```

### 4.1 推荐算法

#### 4.1.1 基于相似产品的推荐

**相似度计算公式**:

```
Sim(A, B) = α × Sim_type + β × Sim_material + γ × Sim_size + δ × Sim_precision + ε × Sim_feature

其中:
- Sim_type: 零件类型匹配度（完全匹配=1，同大类=0.7，不同=0）
- Sim_material: 材料相似度（同牌号=1，同类材料=0.6，不同=0.2）
- Sim_size: 尺寸相似度（1 - |size_A - size_B| / max(size_A, size_B)）
- Sim_precision: 精度匹配度（同等级=1，相邻等级=0.7，差2级=0.4）
- Sim_feature: 特征匹配度（Jaccard相似度）
- α=0.25, β=0.25, γ=0.2, δ=0.15, ε=0.15
```

```php
// app/services/ai/SimilarityCalculator.php
class SimilarityCalculator {
    
    public function calculateSimilarity(array $partA, array $partB): float {
        $weights = [
            'type' => 0.25,
            'material' => 0.25,
            'size' => 0.20,
            'precision' => 0.15,
            'feature' => 0.15,
        ];

        $simType = $this->calcTypeSimilarity($partA, $partB);
        $simMaterial = $this->calcMaterialSimilarity($partA, $partB);
        $simSize = $this->calcSizeSimilarity($partA, $partB);
        $simPrecision = $this->calcPrecisionSimilarity($partA, $partB);
        $simFeature = $this->calcFeatureSimilarity($partA, $partB);

        return round(
            $weights['type'] * $simType +
            $weights['material'] * $simMaterial +
            $weights['size'] * $simSize +
            $weights['precision'] * $simPrecision +
            $weights['feature'] * $simFeature,
            4
        );
    }

    private function calcTypeSimilarity(array $a, array $b): float {
        if ($a['product_type'] === $b['product_type']) {
            return 1.0;
        }
        
        $categoryMap = [
            '工业压缩机' => ['压缩机体', '工业', '活塞', '气阀'],
            '压力容器' => ['筒体', '封头', '法兰', '接管'],
        ];
        
        foreach ($categoryMap as $categories) {
            if (in_array($a['product_type'], $categories) && in_array($b['product_type'], $categories)) {
                return 0.7;
            }
        }
        
        return 0.0;
    }

    private function calcMaterialSimilarity(array $a, array $b): float {
        if ($a['material_grade'] === $b['material_grade']) {
            return 1.0;
        }
        
        $materialCategories = [
            'carbon_steel' => ['Q235', '45', '20', '35'],
            'stainless_steel' => ['304', '316L', '321', '304L'],
            'alloy_steel' => ['16Mn', '15CrMo', '12Cr1MoV', '42CrMo'],
        ];
        
        foreach ($materialCategories as $materials) {
            if (in_array($a['material_grade'], $materials) && in_array($b['material_grade'], $materials)) {
                return 0.6;
            }
        }
        
        return 0.2;
    }

    private function calcSizeSimilarity(array $a, array $b): float {
        $sizeA = $a['weight'] ?? $a['diameter'] ?? 0;
        $sizeB = $b['weight'] ?? $b['diameter'] ?? 0;
        
        if ($sizeA == 0 && $sizeB == 0) return 1.0;
        if ($sizeA == 0 || $sizeB == 0) return 0.0;
        
        $maxSize = max($sizeA, $sizeB);
        return 1 - (abs($sizeA - $sizeB) / $maxSize);
    }
}
```

```php
        $this->apiKey = YOUR_SECRET'api_key'];
        $this->timeout = $config['timeout'] ?? 30;
        $this->retryCount = $config['retry_count'] ?? 3;
        $this->client = new HttpClient();
    }

    private function request(string $method, string $path, array $options = []): array {
        $url = "{$this->baseUrl}{$path}";
        $headers = [
            'Authorization' => "Bearer {$this->apiKey}",
            'Content-Type' => 'application/json',
        ];

        $attempt = 0;
        $lastError = null;

        while ($attempt < $this->retryCount) {
            try {
                $response = $this->client->request($method, $url, [
                    'headers' => $headers,
                    'json' => $options['json'] ?? null,
                    'multipart' => $options['multipart'] ?? null,
                    'timeout' => $this->timeout,
                ]);
                return $response;
            } catch (\Exception $e) {
                $lastError = $e;
                $attempt++;
                if ($attempt < $this->retryCount) {
                    usleep(pow(2, $attempt) * 100000);
                }
            }
        }

        throw new \Exception("MaxKB API请求失败: {$lastError->getMessage()}");
    }
```
