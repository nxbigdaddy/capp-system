**核心业务流程说明**:

1. **产品设计阶段**: 设计工程师完成产品设计，创建产品BOM结构
2. **工艺设计阶段**: 工艺工程师根据产品BOM，编制工艺路线、工序、工步及工艺卡片
3. **工艺审批阶段**: 工艺主管对工艺文件进行审核，多级审批通过后发布
4. **生产执行阶段**: 生产车间根据发布的工艺文件组织生产
5. **工艺反馈阶段**: 生产过程中发现问题，反馈至工艺部门
6. **工艺变更阶段**: 根据反馈进行工艺变更，重新走审批流程

## 2. 用户权限管理模块

### 2.1 功能说明

#### 2.1.1 用户管理

| 功能项 | 说明 |

| 密码管理 | 密码强度校验、定期更换提醒、密码找回 |
| 登录日志 | 记录用户登录时间、IP、设备信息 |

#### 2.1.2 角色管理

| 功能项 | 说明 |
|-------|------|
| 角色创建 | 定义角色名称、编码、描述 |
| 角色编辑 | 修改角色信息、调整权限 |
| 角色删除 | 删除未分配用户的角色 |
| 角色分配 | 将角色分配给用户 |
| 权限配置 | 为角色配置菜单权限、操作权限、数据权限 |

#### 2.1.3 权限管理（RBAC模型）

系统采用 **RBAC（Role-Based Access Control）** 模型，实现用户-角色-权限的三层管理：

```
用户 (User) ──N:M──> 角色 (Role) ──N:M──> 权限 (Permission)
```

**权限粒度**:
- **菜单权限**: 控制用户可见的菜单项
- **操作权限**: 控制用户的增删改查操作（add/edit/delete/view/export）
- **数据权限**: 控制用户可访问的数据范围（本人/本部门/全部）

| 部门删除 | 检查关联用户后删除 |

#### 2.4.2 核心PHP代码示例

**登录验证**:

```php
<?php
// auth/login.php
session_start();
require_once '../config/database.php';
require_once '../utils/response.php';
require_once '../utils/security.php';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username'] ?? '');
    $password = YOUR_PASSWORD'password'] ?? '';
    $captcha = $_POST['captcha'] ?? '';
    
    if ($captcha !== $_SESSION['captcha_code']) {
        Response::error(400, '验证码错误');
    }
    
    $db = Database::getInstance();
    $stmt = $db->prepare("SELECT * FROM sys_user WHERE username = ? AND status = 1 AND deleted_at IS NULL");
    $stmt->execute([$username]);
    $user = $stmt->fetch(PDO::FETCH_ASSOC);
    
    if (!$user || !password_verify($password, $user['password'])) {
        Security::logLoginAttempt($username, $_SERVER['REMOTE_ADDR'], false);
        Response::error(401, '用户名或密码错误');
    }
    
    $permissions = Security::getUserPermissions($user['id']);
    
    $_SESSION['user_id'] = $user['id'];
    $_SESSION['username'] = $user['username'];
    $_SESSION['real_name'] = $user['real_name'];
    $_SESSION['permissions'] = $permissions;
    $_SESSION['login_time'] = time();
    
    $stmt = $db->prepare("UPDATE sys_user SET last_login_time = NOW(), last_login_ip = ?, login_count = login_count + 1 WHERE id = ?");
    $stmt->execute([$_SERVER['REMOTE_ADDR'], $user['id']]);
    
    Security::logLoginAttempt($username, $_SERVER['REMOTE_ADDR'], true);
    
    Response::success([
        'user_id' => $user['id'],
        'username' => $user['username'],
        'real_name' => $user['real_name'],
        'permissions' => $permissions
    ], '登录成功');
}
?>
```

**权限校验中间件**:

```php
<?php
// middleware/AuthMiddleware.php
class AuthMiddleware {
    
    public static function checkLogin(): bool {
        if (empty($_SESSION['user_id'])) {
            Response::error(401, '请先登录');
            return false;
        }
        return true;
    }
    
    public static function checkPermission(string $permissionCode): bool {
        self::checkLogin();
        
        $permissions = $_SESSION['permissions'] ?? [];
        
        if (in_array('ROLE_ADMIN', $_SESSION['roles'] ?? [])) {
            return true;
        }
        
        if (!in_array($permissionCode, $permissions)) {
            Response::error(403, '无操作权限');
            return false;
        }
        
        return true;
    }
    
    public static function checkDataPermission(int $dataOwnerId): bool {
```

#### 3.4.2 BOM递归展开PHP实现

```php
<?php
// services/BomService.php
class BomService {
    
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function expandBom(int $bomId, int $maxLevel = 0): array {
        $items = $this->getBomItems($bomId);
        return $this->buildBomTree($items, 0, 1, $maxLevel);
    }
    
    private function getBomItems(int $bomId): array {
        $sql = "SELECT bi.*, m.material_name, m.material_code, m.specification, 
                       m.material_grade, m.unit, m.weight, m.material_type
                FROM bom_item bi
                LEFT JOIN material m ON bi.material_id = m.id
                WHERE bi.bom_id = ?
                ORDER BY bi.sort ASC, bi.id ASC";
        
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$bomId]);
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
    
    private function buildBomTree(array $items, int $parentId, int $currentLevel, int $maxLevel): array {
        $tree = [];
        
        foreach ($items as $item) {
            if ($item['parent_id'] == $parentId) {
                $item['cumulative_qty'] = $this->calculateCumulativeQty($items, $item['id']);
                
                if ($maxLevel == 0 || $currentLevel < $maxLevel) {
                    $children = $this->buildBomTree($items, $item['id'], $currentLevel + 1, $maxLevel);
                    if (!empty($children)) {
                        $item['children'] = $children;
                        $item['has_children'] = true;
                    }
                } else {
                    $item['has_children'] = $this->hasChildren($items, $item['id']);
                }
                
                $tree[] = $item;
            }
        }
        
        return $tree;
    }
    
    private function calculateCumulativeQty(array $items, int $itemId): float {
        $qty = 1;
        $currentId = $itemId;
        
        while (true) {
            $parent = null;
            foreach ($items as $item) {
                if ($item['id'] == $currentId) {
                    $parent = $item;
                    break;
                }
            }
            
            if (!$parent || $parent['parent_id'] == 0) {
                break;
            }
            
            $qty *= $parent['quantity'];
            $currentId = $parent['parent_id'];
        }
        
        return $qty;
    }
    
    private function hasChildren(array $items, int $parentId): bool {
        foreach ($items as $item) {
            if ($item['parent_id'] == $parentId) {
                return true;
            }
        }
        return false;
    }
    
    public function validateBom(int $bomId): array {
        $errors = [];
        $items = $this->getBomItems($bomId);
        
        if ($this->hasCircularReference($items)) {
            $errors[] = 'BOM存在循环引用';
        }
        
        foreach ($items as $item) {
            if ($item['quantity'] <= 0) {
                $errors[] = "物料 {$item['material_code']} 的用量必须大于0";
            }
        }
        
        foreach ($items as $item) {
            if (empty($item['material_name'])) {
                $errors[] = "行号 {$item['item_code']} 关联的物料不存在";
            }
        }
        
        return $errors;
    }
    
    private function hasCircularReference(array $items): bool {
        foreach ($items as $item) {
            $visited = [];
            $currentId = $item['id'];
            
            while ($currentId != 0) {
                if (in_array($currentId, $visited)) {
                    return true;
                }
                $visited[] = $currentId;
                
                $parentId = 0;
                foreach ($items as $parent) {
                    if ($parent['id'] == $currentId) {
                        $parentId = $parent['parent_id'];
                        break;
                    }
                }
                $currentId = $parentId;
            }
        }
        return false;
    }
}
```

## 4. 工艺设计模块（核心）

### 4.1 功能说明

#### 4.1.1 工艺路线设计

工艺路线是产品加工的完整路径，定义从毛坯到成品的全部加工过程。

| 字段 | 说明 |
|-----|------|
| 工艺路线编码 | 按规则自动生成 |
| 关联产品 | 关联产品版本 |
| 路线名称 | 工艺路线名称 |
| 适用批量 | 单件/小批/中批/大批 |
| 路线状态 | 草稿/审核中/已发布/已归档 |
| 编制人 | 工艺工程师 |
| 编制日期 | 创建时间 |

**工业压缩机典型工艺路线**:
```
原材料检验 → 下料 → 锻造 → 粗车 → 热处理 → 精车 → 钻孔 → 研磨 → 清洗 → 检验 → 装配 → 试压 → 终检 → 入库
```

**压力容器典型工艺路线**:
```
材料验收 → 划线 → 下料 → 卷板 → 纵缝焊接 → 环缝焊接 → 热处理 → 无损检测 → 水压试验 → 表面处理 → 涂装 → 检验 → 入库
```

#### 4.1.2 工序设计

工序是工艺路线的基本组成单元：

| 字段 | 说明 |
|-----|------|
| 工序编号 | 如 OP10, OP20 |
| 工序名称 | 工序名称 |
| 工序类型 | 加工/检验/热处理/装配等 |
| 工作中心 | 执行工序的工作中心 |
| 设备 | 使用的设备 |
| 工装夹具 | 使用的工装 |
| 刀具 | 使用的刀具 |
| 量具 | 使用的量具 |
| 准备时间 | 分钟 |
| 单件时间 | 分钟 |
| 工序说明 | 详细描述 |
| 技术要求 | 精度、表面粗糙度等 |

#### 4.1.3 工步设计

工步是工序的细化，描述工序内的具体操作步骤：

| 字段 | 说明 |
|-----|------|
| 工步序号 | 如 1, 2, 3 |
| 工步内容 | 操作描述 |
| 工艺参数 | 转速、进给量、切削深度等 |
| 工装 | 本工步使用的工装 |
| 量具 | 本工步使用的量具 |
| 图示 | 工序简图（支持图片上传） |

#### 4.1.4 工艺卡片编制

工艺卡片是工艺文件的载体，包含：
- 工艺过程卡
- 工序卡
- 工步卡
- 检验卡
- 装配卡

#### 4.1.5 工艺文件管理

- 工艺文件上传、下载
- 文件格式支持：PDF、Word、Excel、CAD图纸
- 文件版本管理
- 文件权限控制

#### 4.1.6 工艺版本管理

与产品版本关联，支持：
- 工艺版本创建
- 版本对比
- 版本回溯
- 版本变更影响分析

```php
    public function compareVersions(int $routeId1, int $routeId2): array {
        $route1 = $this->getOperationSummary($routeId1);
        $route2 = $this->getOperationSummary($routeId2);
        
        $comparison = ['added' => [], 'deleted' => [], 'modified' => [], 'unchanged' => []];
        $opCodes1 = array_column($route1, 'op_code');
        $opCodes2 = array_column($route2, 'op_code');
        
        foreach ($route2 as $op2) {
            if (!in_array($op2['op_code'], $opCodes1)) {
                $comparison['added'][] = $op2;
            }
        }
        
        foreach ($route1 as $op1) {
            if (!in_array($op1['op_code'], $opCodes2)) {
                $comparison['deleted'][] = $op1;
            }
        }
        
        foreach ($route2 as $op2) {
            $opCode = $op2['op_code'];
            if (in_array($opCode, $opCodes1)) {
                $op1 = array_values(array_filter($route1, fn($o) => $o['op_code'] === $opCode))[0];
                
                if ($op1['op_name'] !== $op2['op_name'] || $op1['op_content'] !== $op2['op_content'] || $op1['equipment_id'] !== $op2['equipment_id']) {
                    $comparison['modified'][] = ['old' => $op1, 'new' => $op2];
                } else {
                    $comparison['unchanged'][] = $op2;
                }
            }
        }
        
        return $comparison;
    }
```

## 6. 工时定额模块

### 6.1 功能说明

#### 6.1.1 工时计算

根据工艺参数自动计算各工序的工时定额：

| 计算项 | 说明 | 计算公式 |
|-------|------|---------|
| 准备时间 | 工序开始前的准备工作时间 | 经验值/历史数据 |
| 加工时间 | 实际切削/加工时间 | 根据切削参数计算 |
| 辅助时间 | 装夹、测量、换刀等 | 经验系数 × 加工时间 |
| 宽放时间 | 休息、生理需要等 | (准备+加工+辅助) × 宽放率 |
| 单件时间 | 单个产品的总时间 | 准备时间/批量 + 加工时间 + 辅助时间 + 宽放时间 |

#### 6.1.2 工时定额管理

| 功能项 | 说明 |
|-------|------|
| 定额制定 | 自动计算 + 人工调整 |
| 定额审批 | 工时定额需审批确认 |
| 定额查询 | 按产品/工序查询 |
| 定额调整 | 根据生产反馈调整 |

#### 6.1.3 工时统计

| 统计项 | 说明 |
|-------|------|
| 产品总工时 | 单个产品的全部工序工时汇总 |
| 工序工时分布 | 各工序工时占比分析 |
| 工时趋势 | 历史工时变化趋势 |

- 高工时工序识别
- 工艺参数优化建议

- 工序合并建议

### 6.2 计算规则

#### 6.2.1 准备时间 (T_setup)

```
T_setup = 基础准备时间 + 换刀时间 + 换工装时间 + 首件检验时间
```

| 项目 | 时间(分钟) | 说明 |
|-----|-----------|------|
| 基础准备 | 10-30 | 熟悉图纸、领取物料 |
| 换刀 | 5-15/把 | 根据刀具数量 |
| 换工装 | 10-30 | 根据工装复杂度 |
| 首件检验 | 10-20 | 首件测量检验 |

#### 6.2.2 加工时间 (T_machining)

**车削加工**:
```
T_machining = L / (n × f) × i

其中:
L = 加工长度 (mm)
n = 主轴转速 (rpm)
f = 进给量 (mm/r)
i = 走刀次数
```

**铣削加工**:
```
T_machining = L / (n × f × z) × i

其中:
L = 加工长度 (mm)
n = 主轴转速 (rpm)
f = 每齿进给量 (mm/z)
z = 刀具齿数
i = 走刀次数
```

#### 6.2.3 辅助时间 (T_auxiliary)

```
T_auxiliary = T_machining × K_aux

K_aux 为辅助时间系数，根据工序类型取值:
- 车削: 0.15 ~ 0.25
- 铣削: 0.20 ~ 0.30
- 钻削: 0.10 ~ 0.20
- 磨削: 0.25 ~ 0.40
- 检验: 0.50 ~ 1.00
```

#### 6.2.4 宽放时间 (T_allowance)

```
T_allowance = (T_setup + T_machining + T_auxiliary) × K_allowance

K_allowance 为宽放率，一般取值:
- 一般加工: 0.10 ~ 0.15
- 重体力加工: 0.15 ~ 0.20
- 精密加工: 0.08 ~ 0.12
```

#### 6.2.5 单件时间 (T_unit)

```
T_unit = T_setup / batch_qty + T_machining + T_auxiliary + T_allowance
```

#### 6.3.2 工时计算PHP实现

```php
<?php
// services/TimeQuotaService.php
class TimeQuotaService {
    
    private $db;
    
    private $auxiliaryRates = [
        'turning' => 0.20, 'milling' => 0.25, 'drilling' => 0.15,
        'grinding' => 0.30, 'inspection' => 0.80, 'assembly' => 0.10, 'heat_treat' => 0.05,
    ];
    
    private $allowanceRate = 0.12;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function calculateOperationTime(int $operationId, int $batchQty = 1): array {
        $operation = $this->getOperation($operationId);
        if (!$operation) throw new Exception('工序不存在');
        
        $steps = $this->getSteps($operationId);
        $setupTime = $this->calculateSetupTime($operation, $steps);
        $machiningTime = $this->calculateMachiningTime($steps);
        
        $opType = $operation['op_type'];
        $auxRate = $this->auxiliaryRates[$opType] ?? 0.20;
        $auxiliaryTime = $machiningTime * $auxRate;
        $allowanceTime = ($setupTime + $machiningTime + $auxiliaryTime) * $this->allowanceRate;
        $unitTime = ($setupTime / $batchQty) + $machiningTime + $auxiliaryTime + $allowanceTime;
        
        return [
            'operation_id' => $operationId,
            'setup_time' => round($setupTime, 2),
            'machining_time' => round($machiningTime, 2),
            'auxiliary_time' => round($auxiliaryTime, 2),
            'allowance_time' => round($allowanceTime, 2),
            'unit_time' => round($unitTime, 2),
            'batch_qty' => $batchQty,
            'auxiliary_rate' => $auxRate,
            'allowance_rate' => $this->allowanceRate,
            'calculation_method' => 'auto'
        ];
    }
    
    private function calculateSetupTime(array $operation, array $steps): float {
        $setupTime = 15;
        $toolCount = $this->countTools($steps);
        $setupTime += $toolCount * 10;
        
        if (!empty($operation['fixture_ids'])) {
            $fixtureCount = count(explode(',', $operation['fixture_ids']));
            $setupTime += $fixtureCount * 15;
        }
        $setupTime += 15;
        return $setupTime;
    }
    
    private function calculateMachiningTime(array $steps): float {
        $totalTime = 0;
        foreach ($steps as $step) {
            if (!empty($step['spindle_speed']) && !empty($step['feed_rate'])) {
                $rpm = $this->parseNumericValue($step['spindle_speed']);
                $feed = $this->parseNumericValue($step['feed_rate']);
                if ($rpm > 0 && $feed > 0) {
                    $length = 100;
                    $time = $length / ($rpm * $feed);
                    $totalTime += $time;
                }
            } else {
                $totalTime += 30;
            }
        }
        return $totalTime;
    }
    
    private function countTools(array $steps): int {
        $tools = [];
        foreach ($steps as $step) {
            if (!empty($step['tool'])) $tools[] = $step['tool'];
        }
        return count(array_unique($tools));
    }
    
    private function parseNumericValue(string $value): float {
        preg_match('/([\d.]+)/', $value, $matches);
        return (float)($matches[1] ?? 0);
    }
    
    public function calculateProductTime(int $routeId, int $batchQty = 1): array {
        $sql = "SELECT id FROM process_operation WHERE route_id = ? ORDER BY sort ASC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$routeId]);
        $operationIds = $stmt->fetchAll(PDO::FETCH_COLUMN);
        
        $totalSetup = 0; $totalMachining = 0; $totalUnit = 0;
        $operationTimes = [];
        
        foreach ($operationIds as $opId) {
            $time = $this->calculateOperationTime($opId, $batchQty);
            $operationTimes[] = $time;
            $totalSetup += $time['setup_time'];
            $totalMachining += $time['machining_time'];
            $totalUnit += $time['unit_time'];
        }
        
        return [
            'route_id' => $routeId, 'batch_qty' => $batchQty,
            'total_setup_time' => round($totalSetup, 2),
            'total_machining_time' => round($totalMachining, 2),
            'total_unit_time' => round($totalUnit, 2),
            'operations' => $operationTimes
        ];
    }
    
    public function saveTimeQuota(int $operationId, array $timeData): bool {
        $sql = "INSERT INTO time_quota (operation_id, setup_time, machining_time, auxiliary_time, allowance_time, unit_time, batch_qty, auxiliary_rate, allowance_rate, calculation_method, created_by) 
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                ON DUPLICATE KEY UPDATE setup_time = VALUES(setup_time), machining_time = VALUES(machining_time), auxiliary_time = VALUES(auxiliary_time), allowance_time = VALUES(allowance_time), unit_time = VALUES(unit_time)";
        
        $stmt = $this->db->prepare($sql);
        return $stmt->execute([$operationId, $timeData['setup_time'], $timeData['machining_time'], $timeData['auxiliary_time'], $timeData['allowance_time'], $timeData['unit_time'], $timeData['batch_qty'], $timeData['auxiliary_rate'], $timeData['allowance_rate'], $timeData['calculation_method'], $_SESSION['user_id']]);
    }
    
    private function getOperation(int $operationId): array {
        $sql = "SELECT * FROM process_operation WHERE id = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$operationId]);
        return $stmt->fetch(PDO::FETCH_ASSOC);
    }
    
    private function getSteps(int $operationId): array {
        $sql = "SELECT * FROM process_step WHERE operation_id = ? ORDER BY step_no ASC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$operationId]);
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
}
```

#### 7.3.2 模板引擎实现

```php
<?php
// services/TemplateEngine.php
class TemplateEngine {
    
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function getTemplateParams(int $templateId): array {
        $sql = "SELECT * FROM process_template_param WHERE template_id = ? ORDER BY sort ASC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$templateId]);
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }
    
    public function renderTemplate(int $templateId, array $params): array {
        $sql = "SELECT * FROM process_template_operation WHERE template_id = ? ORDER BY sort ASC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$templateId]);
        $templateOps = $stmt->fetchAll(PDO::FETCH_ASSOC);
        
        $renderedOps = [];
        
        foreach ($templateOps as $tOp) {
            $operation = [
                'op_code' => $tOp['op_code'],
                'op_name' => $this->replaceParams($tOp['op_name'], $params),
                'op_type' => $tOp['op_type'],
                'op_content' => $this->replaceParams($tOp['op_content'], $params),
                'steps' => []
            ];
            
            $sql = "SELECT * FROM process_template_step WHERE template_operation_id = ? ORDER BY step_no ASC";
            $stmt = $this->db->prepare($sql);
            $stmt->execute([$tOp['id']]);
            $templateSteps = $stmt->fetchAll(PDO::FETCH_ASSOC);
            
            foreach ($templateSteps as $tStep) {
                $operation['steps'][] = [
                    'step_no' => $tStep['step_no'],
                    'step_content' => $this->replaceParams($tStep['step_content'], $params),
                    'spindle_speed' => $this->replaceParams($tStep['spindle_speed'], $params),
                    'feed_rate' => $this->replaceParams($tStep['feed_rate'], $params),
                    'cutting_depth' => $this->replaceParams($tStep['cutting_depth'], $params),
                    'surface_roughness' => $this->replaceParams($tStep['surface_roughness'], $params),
                ];
            }
            
            $renderedOps[] = $operation;
        }
        
        return $renderedOps;
    }
    
    private function replaceParams(string $content, array $params): string {
        foreach ($params as $code => $value) {
            $content = str_replace("[$code]", $value, $content);
        }
        $content = preg_replace('/\[[A-Z_]+\]/', '', $content);
        return $content;
    }
    
    public function validateParams(int $templateId, array $params): array {
        $requiredParams = $this->getTemplateParams($templateId);
        $errors = [];
        
        foreach ($requiredParams as $param) {
            if ($param['required'] && empty($params[$param['param_code']])) {
                $errors[] = "参数 {$param['param_name']} ({$param['param_code']}) 为必填项";
            }
        }
        
        return $errors;
    }
    
    public function applyTemplate(int $templateId, array $params, int $productVersionId): int {
        $errors = $this->validateParams($templateId, $params);
        if (!empty($errors)) throw new Exception(implode('; ', $errors));
        
        $renderedOps = $this->renderTemplate($templateId, $params);
        
        $processRoute = new ProcessRoute();
        $routeId = $processRoute->create([
            'route_name' => '基于模板创建',
            'product_version_id' => $productVersionId,
            'template_id' => $templateId
        ]);
        
        foreach ($renderedOps as $op) {
            $opId = $processRoute->addOperation($routeId, [
                'op_name' => $op['op_name'],
                'op_type' => $op['op_type'],
                'op_content' => $op['op_content']
            ]);
            
            foreach ($op['steps'] as $step) {
                $processRoute->addStep($opId, $step);
            }
        }
        
        return $routeId;
    }
}
?>
```

#### 8.3.2 审批引擎实现

```php
<?php
// services/ApprovalEngine.php
class ApprovalEngine {
    
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function startApproval(int $flowId, string $businessType, int $businessId, string $title): int {
        $this->db->beginTransaction();
        
        try {
            $sql = "INSERT INTO approval_instance (flow_id, business_type, business_id, title, status, initiator_id, current_node_id) VALUES (?, ?, ?, ?, 'approving', ?, ?)";
            $stmt = $this->db->prepare($sql);
            $stmt->execute([$flowId, $businessType, $businessId, $title, $_SESSION['user_id'], null]);
            
            $instanceId = (int)$this->db->lastInsertId();
            
            $firstNode = $this->getFirstNode($flowId);
            if (!$firstNode) throw new Exception('流程未配置审批节点');
            
            $this->createTasks($instanceId, $firstNode);
            
            $sql = "UPDATE approval_instance SET current_node_id = ? WHERE id = ?";
            $stmt = $this->db->prepare($sql);
            $stmt->execute([$firstNode['id'], $instanceId]);
            
            $this->db->commit();
            return $instanceId;
        } catch (Exception $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
    
    public function handleApproval(int $taskId, string $action, string $comment = ''): bool {
        $this->db->beginTransaction();
        
        try {
            $task = $this->getTask($taskId);
            if (!$task) throw new Exception('审批任务不存在');
            if ($task['assignee_id'] != $_SESSION['user_id']) throw new Exception('无权审批此任务');
            if ($task['status'] != 'pending') throw new Exception('任务已处理');
            
            $sql = "UPDATE approval_task SET status = ?, action = ?, comment = ?, handled_at = NOW() WHERE id = ?";
            $stmt = $this->db->prepare($sql);
            $stmt->execute([$action === 'approve' ? 'approved' : 'rejected', $action, $comment, $taskId]);
            
            $instanceId = $task['instance_id'];
            
            if ($action === 'reject') {
                $this->rejectInstance($instanceId, $comment);
            } else {
                $this->checkAndProceed($instanceId, $task['node_id']);
            }
            
            $this->db->commit();

        }

            $stmt = $this->db->prepare($sql);
            $stmt->execute([$nextNode['id'], $instanceId]);
        } else {

        }
    }
```

```php
<?php
// services/ConfigService.php
class ConfigService {
    
    private $db;
    private static $cache = [];
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function get(string $key, $default = null) {
        if (isset(self::$cache[$key])) {
            return self::$cache[$key];
        }
        
        $sql = "SELECT config_value, config_type FROM sys_config WHERE config_key = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$key]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$row) return $default;
        
        $value = $this->parseConfigValue($row['config_value'], $row['config_type']);
        self::$cache[$key] = $value;
        return $value;
    }
    
    public function set(string $key, $value): bool {
        $configType = $this->detectConfigType($value);
        $configValue = $this->formatConfigValue($value, $configType);
        
        $sql = "INSERT INTO sys_config (config_key, config_value, config_type) 
                VALUES (?, ?, ?) 
                ON DUPLICATE KEY UPDATE config_value = VALUES(config_value), config_type = VALUES(config_type)";
        $stmt = $this->db->prepare($sql);
        $result = $stmt->execute([$key, $configValue, $configType]);
        
        if ($result) {
            self::$cache[$key] = $value;
        }
        
        return $result;
    }
    
    public function getAll(): array {
        $sql = "SELECT * FROM sys_config ORDER BY config_key ASC";
        $stmt = $this->db->query($sql);
        $configs = $stmt->fetchAll(PDO::FETCH_ASSOC);
        
        $result = [];
        foreach ($configs as $config) {
            $result[$config['config_key']] = $this->parseConfigValue($config['config_value'], $config['config_type']);
        }
        
        return $result;
    }
    
    private function parseConfigValue(string $value, string $type) {
        switch ($type) {
            case 'number': return (float)$value;
            case 'boolean': return $value === '1' || $value === 'true';
            case 'json': return json_decode($value, true);
            default: return $value;
        }
    }
    
    private function detectConfigType($value): string {
        if (is_bool($value)) return 'boolean';
        if (is_int($value) || is_float($value)) return 'number';
        if (is_array($value)) return 'json';
        return 'string';
    }
    
    private function formatConfigValue($value, string $type): string {
        switch ($type) {
            case 'boolean': return $value ? '1' : '0';
            case 'number': return (string)$value;
            case 'json': return json_encode($value);
            default: return (string)$value;
        }
    }
}
```
